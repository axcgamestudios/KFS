   110→  private buildModel(): void {
   111→    const pirate = this.kind === 'pirate';
   112→    const b = new StaticBuilder(new ColliderRegistry(), `${this.id}_visual`);
   113→    const hullKey = pirate ? 'woodPirate' : 'woodShip';
   114→    const sailKey = pirate ? 'sailDark' : 'sail';
   115→    const L = 16;
   116→    const W = 4.3;
   117→
   118→    // --- Hull sections (curved profile, bow at +Z) ---
   119→    const sections = 11;
   120→    for (let i = 0; i < sections; i++) {
   121→      const t = i / (sections - 1); // 0 stern … 1 bow
   122→      const along = (t - 0.5) * L;
   123→      const widthProf = Math.sin(Math.PI * Math.min(1, 0.14 + t * 0.74)) * 0.74 + 0.26;
   124→      const w = W * widthProf;
   125→      const rocker = Math.pow(Math.abs(t - 0.42) * 2, 2.1) * 1.35;
   126→      const hullH = 2.1;
   127→      const py = DECK_Y - 1.05 + rocker * 0.55;
   128→      b.add(new THREE.BoxGeometry(w * 0.96, hullH * 0.5, L / sections + 0.06), hullKey, 0, py - hullH * 0.25, along);
   129→      for (const side of [-1, 1]) {
   130→        b.add(new THREE.BoxGeometry(0.17, hullH, L / sections + 0.05), hullKey, side * (w / 2), py, along);
   131→        b.add(new THREE.BoxGeometry(0.15, hullH * 0.52, L / sections + 0.04), 'wood',
   132→          side * (w / 2 + 0.1), py + hullH * 0.62, along, 0, 1, 1, 1, 0, side * 0.1);
   133→        // Gold/dark trim wale
   134→        b.add(new THREE.BoxGeometry(0.1, 0.14, L / sections + 0.05), pirate ? 'metal' : 'gold',
   135→          side * (w / 2 + 0.16), py + hullH * 0.1, along);
   136→      }
   137→    }
   138→
   139→    // --- Deck platforms (main, quarter, bow) ---
   140→    b.add(new THREE.BoxGeometry(W * 0.88, 0.14, 10.6), 'woodLight', 0, DECK_Y - 0.02, 0.6);
   141→    b.add(new THREE.BoxGeometry(W * 0.8, 1.2, 3.6), hullKey, 0, QUARTER_Y - 0.6, -L / 2 + 2.0);
   142→    b.add(new THREE.BoxGeometry(W * 0.9, 0.14, 3.9), 'woodLight', 0, QUARTER_Y, -L / 2 + 2.0);
   143→    b.add(new THREE.BoxGeometry(W * 0.62, 0.14, 3.0), 'woodLight', 0, DECK_Y - 0.02, L / 2 - 1.9);
   144→    // Rear gallery windows (stern)
   145→    b.add(new THREE.BoxGeometry(W * 0.7, 0.5, 0.12), 'glow', 0, QUARTER_Y - 0.2, -L / 2 - 0.02);
   146→    // Hatch + crates on deck
   147→    b.box('woodDark', 1.1, 0.32, 1.5, 0, DECK_Y + 0.2, 0.6);
   148→    b.box('woodDark', 0.8, 0.7, 0.8, 1.2, DECK_Y + 0.42, -1.6, 0.3);
   149→    if (pirate) {
   150→      b.box('woodDark', 0.8, 0.7, 0.8, -1.2, DECK_Y + 0.42, -0.4, 1.1);
   151→      // Grapple rope coil
   152→      b.add(new THREE.TorusGeometry(0.3, 0.06, 6, 12), 'roofThatch', 0.9, DECK_Y + 0.1, 2.4, 0, 1, 1, 1, Math.PI / 2);
   153→    }
   154→
   155→    // --- Railings ---
   156→    for (const side of [-1, 1]) {
   157→      b.add(new THREE.BoxGeometry(0.09, 0.09, 10.2), 'wood', side * (W / 2 - 0.15), DECK_Y + 0.72, 0.6);
   158→      for (let i = 0; i < 11; i++) {
   159→        const z = -4.6 + i * 0.95;
   160→        b.cylinder('woodDark', 0.045, 0.045, 0.66, 5, side * (W / 2 - 0.15), DECK_Y + 0.36, z);
   161→      }
   162→      b.add(new THREE.BoxGeometry(0.09, 0.09, 3.4), 'wood', side * (W * 0.36), QUARTER_Y + 0.72, -L / 2 + 2.0);
   163→    }
   164→
   165→    // --- Bowsprit ---
   166→    b.add(new THREE.CylinderGeometry(0.08, 0.13, 5.2, 7), 'wood',
   167→      0, DECK_Y + 1.45, L / 2 + 1.4, 0, 1, 1, 1, 0, Math.PI / 2 + 0.12);
   168→
   169→    // --- Masts (main + fore) + yards ---
   170→    const mainZ = -0.8, foreZ = 4.6;
   171→    for (const [mz, mh, mr] of [[mainZ, 15.5, 0.19], [foreZ, 12.0, 0.15]] as Array<[number, number, number]>) {
   172→      b.cylinder('wood', mr, mr + 0.07, mh, 8, 0, DECK_Y - 0.4 + mh / 2, mz);
   173→      b.add(new THREE.BoxGeometry(8.2, 0.13, 0.13), 'woodDark', 0, DECK_Y + mh - 3.4, mz);
   174→      // Shrouds
   175→      for (const side of [-1, 1]) {
   176→        b.add(new THREE.BoxGeometry(0.05, mh - 2.4, 0.05), 'woodDark',
   177→          side * 1.9, DECK_Y + (mh - 2.4) / 2, mz - 0.4, 0, 1, 1, 1, side * 0.1);
   178→      }
   179→    }
   180→    // Crow nest
   181→    b.cylinder('woodDark', 0.55, 0.55, 0.5, 8, 0, DECK_Y + 9.6, mainZ);
   182→
   183→    // --- Wheel + rudder ---
   184→    const wheelG = new THREE.Group();
   185→    const wheelMat = MAT.wood;
   186→    const ring = new THREE.Mesh(new THREE.TorusGeometry(0.42, 0.05, 6, 16), wheelMat);
   187→    wheelG.add(ring);
   188→    for (let i = 0; i < 4; i++) {
   189→      const spoke = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 0.86, 5), wheelMat);
   190→      spoke.rotation.z = (i / 4) * Math.PI;
   191→      wheelG.add(spoke);
   192→    }
   193→    wheelG.position.set(0, QUARTER_Y + 0.55, -L / 2 + 2.6);
   194→    wheelG.rotation.x = Math.PI / 2 - 0.35;
   195→    this.wheel = wheelG;
   196→    this.hull.add(wheelG);
   197→    // Rudder below stern
   198→    b.add(new THREE.BoxGeometry(0.14, 2.2, 0.9), 'woodDark', 0, DECK_Y - 1.4, -L / 2 - 0.3);
   199→
   200→    // --- Cannons: 3 per side (port +X, starboard -X) ---
   201→    const cannonMat = MAT.metal;
   202→    for (const side of [1, -1]) {
   203→      for (const z of [-3.0, -0.6, 1.8]) {
   204→        const c = new THREE.Group();
   205→        const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.13, 1.15, 8), cannonMat);
   206→        barrel.rotation.z = Math.PI / 2;
   207→        barrel.position.x = side * 0.35;
   208→        c.add(barrel);
   209→        const mount = new THREE.Mesh(new THREE.BoxGeometry(0.3, 0.24, 0.34), MAT.woodDark);
   210→        c.add(mount);
   211→        c.position.set(side * (W / 2 - 0.45), DECK_Y + 0.32, z);
   212→        c.rotation.y = side * Math.PI / 2;
   213→        this.hull.add(c);
   214→        this.muzzles.push({ x: side * (W / 2 + 0.35), y: DECK_Y + 0.32, z });
   215→      }
   216→    }
   217→
   218→    // --- Lanterns (stern + bow) ---
   219→    const lantern = (x: number, y: number, z: number) => {
   220→      b.cylinder('metal', 0.14, 0.18, 0.36, 6, x, y, z);
   221→      b.add(new THREE.BoxGeometry(0.2, 0.26, 0.2), 'glow', x, y, z);
   222→    };
   223→    lantern(0, QUARTER_Y + 0.95, -L / 2 - 0.2);
   224→    lantern(0, DECK_Y + 0.5, L / 2 - 0.1);
   225→
   226→    // Merge static hull parts (into the hideable procedural body group)
   227→    b.build(this.bodyG, { shadows: true });
   228→
   229→    // --- Animated sails (separate meshes) ---
   230→    const mkSail = (y: number, z: number, w: number, h: number) => {
   231→      const geo = new THREE.PlaneGeometry(w, h, 6, 4);
   232→      // slight belly curve
   233→      const pos = geo.attributes.position as THREE.BufferAttribute;
   234→      for (let i = 0; i < pos.count; i++) {
   235→        const u = (pos.getX(i) / w) * 2;
   236→        pos.setZ(i, (1 - u * u) * w * 0.09);
   237→      }
   238→      geo.computeVertexNormals();
   239→      const mesh = new THREE.Mesh(geo, MAT[sailKey]);
   240→      mesh.position.set(0, y, z);
   241→      mesh.rotation.y = Math.PI / 2; // face fore-aft
   242→      this.hull.add(mesh);
   243→      return mesh;
   244→    };
   245→    this.sailMain = mkSail(DECK_Y + 10.4, mainZ, 7.6, 5.4);
   246→    this.sailFore = mkSail(DECK_Y + 7.3, foreZ, 5.0, 4.2);
   247→
   248→    // --- Flag at main mast top ---
   249→    const flagGeo = new THREE.PlaneGeometry(1.5, 0.7, 4, 1);
   250→    this.flag = new THREE.Mesh(flagGeo, pirate ? MAT.clothRaven : MAT.clothGold);
   251→    this.flag.position.set(0.76, DECK_Y + 14.6, mainZ - 0.3);
   252→    this.flag.castShadow = false;
   253→    this.hull.add(this.flag);
   254→
   255→    // --- Damage smoke sprites (pooled, hidden until hull low) ---
   256→    const smokeCanvas = document.createElement('canvas');
   257→    smokeCanvas.width = smokeCanvas.height = 64;
   258→    const sctx = smokeCanvas.getContext('2d')!;
   259→    const sgrad = sctx.createRadialGradient(32, 32, 2, 32, 32, 30);
   260→    sgrad.addColorStop(0, 'rgba(60,58,56,0.85)');
   261→    sgrad.addColorStop(1, 'rgba(60,58,56,0)');
   262→    sctx.fillStyle = sgrad;
   263→    sctx.fillRect(0, 0, 64, 64);
   264→    const smokeTex = new THREE.CanvasTexture(smokeCanvas);
   265→    for (let i = 0; i < 3; i++) {
   266→      const sp = new THREE.Sprite(new THREE.SpriteMaterial({
   267→        map: smokeTex, transparent: true, depthWrite: false, opacity: 0,
   268→      }));
   269→      sp.scale.set(2.4, 2.4, 1);
   270→      sp.position.set((i - 1) * 0.9, DECK_Y + 2.4, -1 - i * 1.4);
   271→      this.hull.add(sp);
   272→      this.smoke.push(sp);
   273→    }
   274→
   275→    // --- Stern wake foam strip ---
   276→    const foamGeo = new THREE.PlaneGeometry(W * 0.9, 1);
   277→    foamGeo.rotateX(-Math.PI / 2);
   278→    this.sternFoam = new THREE.Mesh(foamGeo, new THREE.MeshBasicMaterial({
   279→      color: 0xe8f2ee, transparent: true, opacity: 0, depthWrite: false,
   280→    }));
   281→    this.sternFoam.position.set(0, 0.12, -L / 2 - 1.6);
   282→    this.hull.add(this.sternFoam);
   283→  }

   287→  /**
   288→   * Fase 7 — wear the repository's pirate_ship.glb as the ship's body.
   289→   * The model is normalised to the gameplay hull (≈16 m, local +Z = bow,
   290→   * deck plane at DECK_Y) and the procedural body/sails/flag are retired.
   291→   * Cannons, wheel, smoke and wake foam stay (gameplay-anchored visuals).
   292→   */
   293→  attachGlb(model: THREE.Object3D, opts: { sailTint?: number; hullTint?: number } = {}): void {
   294→    if (this.glbHull) return; // already dressed
   295→
   296→    // 1. Strip lights/cameras baked into the Sketchfab export
   297→    const junk: THREE.Object3D[] = [];
   298→    model.traverse((o) => {
   299→      const any = o as unknown as { isLight?: boolean; isCamera?: boolean };
   300→      if (any.isLight || any.isCamera) junk.push(o);
   301→    });
   302→    for (const j of junk) j.parent?.remove(j);
   303→
   304→    // 2. Per-ship material clones when a tint is requested (the AssetSystem
   305→    //    shares materials between every instance of the model)
   306→    if (opts.sailTint !== undefined || opts.hullTint !== undefined) {
   307→      model.traverse((o) => {
   308→        const mesh = o as THREE.Mesh;
   309→        if (!mesh.isMesh) return;
   310→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   311→        const clones = mats.map((m) => (m as THREE.Material).clone());
   312→        for (const c of clones) {
   313→          const std = c as THREE.MeshStandardMaterial;
   314→          if (!std.isMeshStandardMaterial) continue;
   315→          const name = (std.name ?? '').toLowerCase();
   316→          if (opts.sailTint !== undefined && (name.includes('sail') || name.includes('material_0'))) {
   317→            std.color.setHex(opts.sailTint);
   318→          } else if (opts.hullTint !== undefined && (name.includes('mat.1') || name.includes('main'))) {
   319→            std.color.lerp(new THREE.Color(opts.hullTint), 0.5);
   320→          }
   321→        }
   322→        mesh.material = clones.length === 1 ? clones[0] : clones;
   323→      });
   324→    }
   325→
   326→    // 3. Normalise: fit the longest horizontal axis to ~17 m (the hull spans
   327→    //    ≈16 m + bowsprit), flip so the GLB bow (-Z, verified from the
   328→    //    bowsprit/jib geometry) matches the ship convention (+Z = bow).
   329→    model.updateWorldMatrix(true, true);
   330→    const rawBox = new THREE.Box3().setFromObject(model);
   331→    const rawSize = rawBox.getSize(new THREE.Vector3());
   332→    const s = 17.2 / Math.max(1e-6, Math.max(rawSize.x, rawSize.z));
   333→
   334→    const wrap = new THREE.Group();
   335→    wrap.name = 'glb_hull';
   336→    wrap.add(model);
   337→    model.rotation.y = Math.PI; // bow to +Z
   338→    model.scale.multiplyScalar(s);
   339→
   340→    // 4. Align: centre on X/Z and put the export's DESIGNED WATERLINE at the
   341→    //    buoyancy surface (y ≈ 0). Measured offline (Fase 9, world-space
   342→    //    vertex probe): the deck plane — where Mat.6/Mat.7 furniture sits —
   343→    //    lands at raw Y ≈ 128 and the keel at −649. The old −60 waterline
   344→    //    left only 0.68 m of freeboard: the ships rode badly sunken.
   345→    //    −149 restores a seamanlike 1.1 m freeboard / 1.83 m draft.
   346→    wrap.updateWorldMatrix(true, true);
   347→    const box = new THREE.Box3().setFromObject(model);
   348→    const center = box.getCenter(new THREE.Vector3());
   349→    model.position.set(0, 0, 0);
   350→    const WATERLINE_RAW = -149; // deck furniture raw 128 − 1.0 m / s
   351→    wrap.position.set(-center.x, -WATERLINE_RAW * s, -center.z);
   352→
   353→    // 5. Retire the procedural body & sails — keep cannons, wheel, smoke, foam
   354→    this.bodyG.visible = false;
   355→    this.sailMain.visible = false;
   356→    this.sailFore.visible = false;
   357→    this.flag.visible = false;
   358→    this.hull.add(wrap);
   359→    this.glbHull = wrap;
   360→
   361→    // Fase 11 — fit the colliders + walkable deck to the REAL hull: the
   362→    // export is wider/longer than the procedural hull the base layout was
   363→    // tuned on, which is why the player used to walk through the sides.
   364→    wrap.updateWorldMatrix(true, true);
   365→    const realBox = new THREE.Box3().setFromObject(wrap);
   366→    const halfW = Math.max(2.35, realBox.max.x - realBox.min.x) / 2;
   367→    const halfL = Math.max(8.4, realBox.max.z - realBox.min.z) / 2;
   368→    this.colSX = Math.min(1.85, halfW / 2.35);
   369→    this.colSZ = Math.min(1.22, halfL / 8.4);
   370→    this.deckHalfW = Math.max(1.62, Math.min(3.1, halfW - 0.52));
   371→    this.deckBowZ = Math.max(5.9, halfL - 0.8);
   372→    this.deckSternZ = Math.min(-4.7, -halfL + 3.2);
   373→    this.quarterHalfW = Math.max(1.5, this.deckHalfW - 0.16);
   374→    // quarterdeck (aft): from the stern extreme up to just abaft the main deck
   375→    this.quarterSternZ = Math.min(-7.9, -halfL + 0.7);
   376→    this.quarterBowZ = this.deckSternZ + 0.35;
   377→    this.colSynced = false; // force an AABB refresh with the new scales
   378→  }

   388→  /** Ship-local ↔ world transforms. Local +Z = bow. */
   389→  toWorld(lx: number, ly: number, lz: number, out = new THREE.Vector3()): THREE.Vector3 {
   390→    const s = Math.sin(this.heading);
   391→    const c = Math.cos(this.heading);
   392→    return out.set(
   393→      this.pos.x + lx * c + lz * s,
   394→      ly + this.group.position.y,
   395→      this.pos.z - lx * s + lz * c,
   396→    );
   397→  }
   398→
   399→  toLocal(wx: number, wz: number, out = new THREE.Vector2()): THREE.Vector2 {
   400→    const dx = wx - this.pos.x;
   401→    const dz = wz - this.pos.z;
   402→    const s = Math.sin(this.heading);
   403→    const c = Math.cos(this.heading);
   404→    return out.set(dx * c - dz * s, dx * s + dz * c);
   405→  }

   407→  /** Deck platform height at ship-local coords, or null when off-deck.
   408→   *  Fase 11 — the bounds follow the REAL hull (GLB ships are wider). */
   409→  deckHeightAt(lx: number, lz: number): number | null {
   410→    if (Math.abs(lx) > this.deckHalfW) return null;
   411→    if (lz >= this.deckSternZ && lz <= this.deckBowZ) return DECK_Y + this.group.position.y;       // main + bow deck
   412→    if (lz >= this.quarterSternZ && lz <= this.quarterBowZ && Math.abs(lx) < this.quarterHalfW) return QUARTER_Y + this.group.position.y; // quarterdeck (aft)
   413→    return null;
   414→  }

   420→      maxX: this.deckHalfW - 0.08,
   421→      minZ: this.quarterBowZ,
   422→      maxZ: this.deckBowZ - 0.2,
   423→    };

   426→  get wheelWorld(): THREE.Vector3 {
   427→    return this.toWorld(0, QUARTER_Y + 0.15, -5.2);
   428→  }
   429→
   430→  get crippled(): boolean {
   431→    return this.hullHp < this.maxHullHp * 0.35 && !this.sunk;
   432→  }
   433→
   434→  // ------------------------------------------------------------- physics
   435→
   436→  /** Wind factor 0.12..1 by angle between heading and wind blowing direction. */
   437→  windFactor(windDir: number): number {
   438→    const cosA = Math.sin(this.heading) * Math.sin(windDir) + Math.cos(this.heading) * Math.cos(windDir);
   439→    const ease = this.perks.windEase ?? 1; // nav_viento raises ease → adverse wind hurts less
   440→    const base = 0.16 + 0.84 * Math.pow(cosA * 0.5 + 0.5, Math.max(0.6, 1.45 / ease));
   441→    return clamp(base, 0.12, 1);
   442→  }
   443→
   444→  get maxSpeed(): number {
   445→    return (this.kind === 'caravel' ? 8.6 : 7.4) * (this.perks.speedMult ?? 1) * (this.crippled ? 0.55 : 1);
   446→  }
   447→
   448→  update(
   449→    dt: number,
   450→    water: WaterSystem,
   451→    terrain: TerrainHeight,
   452→    windDir: number,
   453→  ): void {
   454→    this.t += dt;
   455→
   456→    if (this.sunk) {
   457→      this.removable = true;
   458→      return;
   459→    }
   460→
   461→    // --- Sinking animation ---
   462→    if (this.sinking > 0) {
   463→      this.sinking = Math.min(1, this.sinking + dt / 14);
   464→      this.group.position.y -= dt * (0.35 + this.sinking * 0.8);
   465→      this.hull.rotation.z += dt * 0.06;
   466→      this.hull.rotation.x -= dt * 0.03;
   467→      this.speed = damp(this.speed, 0, 1.2, dt);
   468→      if (this.sinking >= 1) {
   469→        this.sunk = true;
   470→        this.group.visible = false;
   471→      }
   472→      this.syncTransform();
   473→      return;
   474→    }
   475→
   476→    // --- Grounding: caravels need ~1.8 m of water ---
   477→    const depth = terrain.height(this.pos.x, this.pos.z);
   478→    if (depth > -1.7 && this.speed > 0.1) {
   479→      this.speed = 0;
   480→    }
   481→
   482→    // --- Sails & wind ---
   483→    const wf = this.windFactor(windDir);
   484→    const sailK = this.sailState / 2;
   485→    const target = this.anchored || this.grappled ? 0 : sailK * this.maxSpeed * wf;
   486→    const accel = target > this.speed ? 0.32 : 0.8;
   487→    this.speed = damp(this.speed, target, accel, dt);
   488→
   489→    // --- Rudder & heading ---
   490→    this.rudder = damp(this.rudder, this.rudderInput, 3.2, dt);
   491→    if (!this.frozen && Math.abs(this.speed) > 0.15) {
   492→      const turnRate = 0.62 * (0.35 + 0.65 * Math.abs(this.speed) / this.maxSpeed);
   493→      this.heading += this.rudder * turnRate * dt;
   494→    }
   495→
   496→    // --- Integrate ---
   497→    if (!this.frozen) {
   498→      this.pos.x += Math.sin(this.heading) * this.speed * dt;
   499→      this.pos.z += Math.cos(this.heading) * this.speed * dt;
   500→      this.pos.x = clamp(this.pos.x, -1010, 1010);
   501→      this.pos.z = clamp(this.pos.z, -1010, 1010);
   502→    }
   503→
   504→    // --- Buoyancy (4-point sampling of the same Gerstner waves as the shader) ---
   505→    if (!this.frozen) {
   506→      const pBow = this.toWorld(0, 0, 6.4);
   507→      const pStern = this.toWorld(0, 0, -6.4);
   508→      const pPort = this.toWorld(1.9, 0, 0);
   509→      const pStar = this.toWorld(-1.9, 0, 0);
   510→      const hBow = water.waveHeightAt(pBow.x, pBow.z);
   511→      const hStern = water.waveHeightAt(pStern.x, pStern.z);
   512→      const hPort = water.waveHeightAt(pPort.x, pPort.z);
   513→      const hStar = water.waveHeightAt(pStar.x, pStar.z);
   514→      const avg = (hBow + hStern + hPort + hStar) / 4;
   515→      // Fase 9 — rides a touch high: +0.2 m keeps the scuppers clear of the
   516→      // swell crests (the sunken-ship look was mostly water burying the deck)
   517→      this.group.position.y = damp(this.group.position.y, avg + 0.2, 5, dt);
   518→      const targetPitch = Math.atan2(hStern - hBow, 12.8);
   519→      const targetRoll = Math.atan2(hPort - hStar, 3.8) + this.rudder * clamp(this.speed / this.maxSpeed, 0, 1) * 0.09;
   520→      this.hull.rotation.x = dampAngle(this.hull.rotation.x, targetPitch, 4, dt);
   521→      this.hull.rotation.z = dampAngle(this.hull.rotation.z, targetRoll, 4, dt);
   522→    }
   523→
   524→    this.syncTransform();
   525→
   526→    // --- Sails visual state ---
   527→    const sailScale = [0.1, 0.62, 1.0][this.sailState];
   528→    const billow = 1 + (this.speed / Math.max(1, this.maxSpeed)) * 0.18 + Math.sin(this.t * 1.7) * 0.03;
   529→    this.sailMain.scale.set(sailScale * billow, sailScale, 1);
   530→    this.sailFore.scale.set(sailScale * billow, sailScale, 1);
   531→    // Sails trim slightly toward the wind
   532→    const relWind = Math.atan2(Math.sin(windDir - this.heading), Math.cos(windDir - this.heading));
   533→    this.sailMain.rotation.x = clamp(relWind * 0.1, -0.2, 0.2);
   534→    this.sailFore.rotation.x = clamp(relWind * 0.12, -0.25, 0.25);
   535→
   536→    // --- Flag flutter ---
   537→    this.flag.rotation.y = Math.sin(this.t * 6) * 0.22 + relWind * 0.4;
   538→    this.flag.scale.x = 0.7 + Math.abs(Math.sin(this.t * 2.3)) * 0.3;
   539→
   540→    // --- Wheel spins with the rudder ---
   541→    this.wheel.rotation.z = this.rudder * 1.9;
   542→
   543→    // --- Damage smoke when crippled ---
   544→    if (this.crippled) {
   545→      this.smokeTimer += dt;
   546→      for (let i = 0; i < this.smoke.length; i++) {
   547→        const sp = this.smoke[i];
   548→        const k = 1 - (this.smokeTimer * 0.4 + i * 0.33) % 1;
   549→        sp.material.opacity = k * 0.5;
   550→        sp.position.y = DECK_Y + 2.2 + (1 - k) * 2.6;
   551→        sp.scale.setScalar(1.8 + (1 - k) * 2.2);
   552→      }
   553→    } else {
   554→      for (const sp of this.smoke) sp.material.opacity = 0;
   555→    }
   556→
   557→    // --- Wake foam ---
   558→    const spd = Math.abs(this.speed);
   559→    (this.sternFoam.material as THREE.MeshBasicMaterial).opacity = clamp(spd / 3, 0, 0.55);
   560→    this.sternFoam.scale.set(1 + spd * 0.12, 1, 2.4 + spd * 0.9);
   561→    if (spd > 0.2) {
   562→      this.sternFoam.position.z = -this.length / 2 - 1.4 - spd * 0.25;
   563→    }
   564→
   565→    // --- Hull creaks while underway (audio callback, throttled) ---
   566→    if (spd > 2.5 && Math.random() < dt * 0.12) {
   567→      this.onCreak?.(this.group.position);
   568→    }
   569→
   570→    // --- Cannon cooldowns ---
   571→    if (this.cdPort > 0) this.cdPort -= dt / (this.perks.cannonCdMult ?? 1);
   572→    if (this.cdStar > 0) this.cdStar -= dt / (this.perks.cannonCdMult ?? 1);
   573→  }
   574→
   575→  /** Fase 11 — public: cinematics and Game moves ships deliberately. */
   576→  syncTransform(): void {
   577→    this.group.position.x = this.pos.x;
   578→    this.group.position.z = this.pos.z;
   579→    this.group.rotation.y = this.heading;
   580→  }
   581→
   582→  /** Broadside spawn descriptors for one side (null if on cooldown). */
   583→  fire(side: 'port' | 'starboard'): CannonShot[] | null {
   584→    if (side === 'port' ? this.cdPort > 0 : this.cdStar > 0) return null;
   585→    if (side === 'port') this.cdPort = this.cannonCd;
   586→    else this.cdStar = this.cannonCd;
   587→    const dirSign = side === 'port' ? 1 : -1;
   588→    const out: CannonShot[] = [];
   589→    for (const m of this.muzzles.filter((mm) => (side === 'port' ? mm.x > 0 : mm.x < 0))) {
   590→      const from = this.toWorld(m.x, m.y, m.z);
   591→      // Broadside direction: perpendicular to heading, plus slight forward bias
   592→      const dx = Math.cos(this.heading) * dirSign + Math.sin(this.heading) * 0.18;
   593→      const dz = -Math.sin(this.heading) * dirSign + Math.cos(this.heading) * 0.18;
   594→      out.push({ from, dir: new THREE.Vector3(dx, 0.16, dz).normalize(), side });
   595→    }
   596→    return out;
   597→  }
   598→
   599→  /** Muzzle world position for flash placement. */
   600→  muzzleFlashPos(side: 'port' | 'starboard', index: number): THREE.Vector3 {
   601→    const arr = this.muzzles.filter((mm) => (side === 'port' ? mm.x > 0 : mm.x < 0));
   602→    const m = arr[index % arr.length];
   603→    return this.toWorld(m.x, m.y, m.z);
   604→  }
   605→
   606→  applyDamage(amount: number): void {
   607→    if (this.sinking > 0 || this.sunk) return;
   608→    this.hullHp = Math.max(0, this.hullHp - amount);
   609→    if (this.hullHp <= 0) {
   610→      this.sinking = 0.001;
   611→    }
   612→  }

   614→  serialize() {
   615→    return {
   616→      x: this.pos.x, z: this.pos.z, heading: this.heading,
   617→      hullHp: this.hullHp, sailState: this.sailState,
   618→      anchored: this.anchored, sunk: this.sunk, sinking: this.sinking,
   619→    };
   620→  }

   622→  // ---------------------------------------------------------- collisions
   623→
   624→  /**
   625→   * Fase 9 — solid hull: an AABB composition that follows the ship.
   626→   *
   627→   * The PDF demanded real collisions: no walking through the hull, no
   628→   * swimming through it, deck still walkable. The volumes mirror the hull
   629→   * shape in 6 local boxes (port/starboard bulwarks in two segments so the
   630→   * tapered bow is respected, bow cap, stern cap) — all of them OUTSIDE the
   631→   // walkable deck bounds so boarding/crew combat is untouched.
   632→   *
   633→   * Re-sync only when the ship moved ≥ 0.35 m or turned ≥ 1.5° (and only
   634→   * when a listener is within 140 m) — the registry update is not free.
   635→   */
   636→  private static readonly COLLIDER_PARTS: Array<{ x0: number; x1: number; z0: number; z1: number }> = [
   637→    // port side, stern half (widest)
   638→    { x0: 1.5, x1: 2.35, z0: -8.1, z1: 0.4 },
   639→    // port side, bow half (tapered)
   640→    { x0: 1.5, x1: 2.35, z0: 0.4, z1: 5.6 },
   641→    // starboard mirror
   642→    { x0: -2.35, x1: -1.5, z0: -8.1, z1: 0.4 },
   643→    { x0: -2.35, x1: -1.5, z0: 0.4, z1: 5.6 },
   644→    // bow cap (curved stem approximated)
   645→    { x0: -1.35, x1: 1.35, z0: 5.6, z1: 8.4 },
   646→    // stern cap (transom)
   647→    { x0: -1.5, x1: 1.5, z0: -8.4, z1: -8.1 },
   648→  ];
   649→
   650→  private colReg: ColliderRegistry | null = null;
   651→  private colSynced = false;
   652→  private lastColX = NaN;
   653→  private lastColZ = NaN;
   654→  private lastColHeading = NaN;
   655→  private lastColY = NaN;
   656→
   657→  /** Fase 11 — collider/deck scale derived from the REAL hull (GLB ships are
   658→   *  wider and longer than the procedural hull the AABBs were tuned on;
   659→   *  without this the player walks through the visible hull sides). */
   660→  colSX = 1;
   661→  colSZ = 1;
   662→  deckHalfW = 1.62;
   663→  deckBowZ = 5.9;
   664→  deckSternZ = -4.7;
   665→  quarterHalfW = 1.5;
   666→  quarterSternZ = -4.3;
   667→  quarterBowZ = -7.9;
   668→
   669→  /** Bind the game's collision world (called once by the NavalSystem). */
   670→  bindColliderRegistry(registry: ColliderRegistry): void {
   671→    this.colReg = registry;
   672→    this.colSynced = false;
   673→    this.lastColX = NaN;
   674→  }
   675→
   676→  /**
   677→   * Refresh the hull AABBs when (and only when) the ship moved enough.
   678→   * @param refX/refZ listener position (player) — far ships skip the sync.
   679→   */
   680→  syncColliders(refX: number, refZ: number): void {
   681→    if (!this.colReg || this.sinking > 0 || this.sunk) {
   682→      if (this.colSynced) this.clearColliders();
   683→      return;
   684→    }
   685→    const dx = this.pos.x - refX;
   686→    const dz = this.pos.z - refZ;
   687→    const far = dx * dx + dz * dz > 140 * 140;
   688→    if (far) {
   689→      if (this.colSynced) this.clearColliders(); // out of range: free the cells
   690→      return;
   691→    }
   692→    const moved = !this.colSynced
   693→      || Math.abs(this.pos.x - this.lastColX) > 0.35
   694→      || Math.abs(this.pos.z - this.lastColZ) > 0.35
   695→      || Math.abs(this.heading - this.lastColHeading) > 0.026
   696→      || Math.abs(this.group.position.y - this.lastColY) > 0.5;
   697→    if (!moved) return;
   698→
   699→    this.clearColliders();
   700→    const gy = this.group.position.y;
   701→    const y0 = gy - 1.15;               // below the waterline: swimmers bump it
   702→    const y1 = gy + DECK_Y + 0.72;      // bulwark/rail height
   703→    const sinH = Math.sin(this.heading);
   704→    const cosH = Math.cos(this.heading);
   705→    for (let i = 0; i < Ship.COLLIDER_PARTS.length; i++) {
   706→      const p = Ship.COLLIDER_PARTS[i];
   707→      // Fase 11 — the parts scale to the REAL hull (GLB ships are wider
   708→      // than the procedural hull the base layout was tuned on)
   709→      const sx = this.colSX;
   710→      const sz = this.colSZ;
   711→      // local -> world for the 4 corners, then take the AABB
   712→      let minX = Infinity, maxX = -Infinity, minZ = Infinity, maxZ = -Infinity;
   713→      for (const [lx, lz] of [
   714→        [p.x0 * sx, p.z0 * sz], [p.x1 * sx, p.z0 * sz], [p.x0 * sx, p.z1 * sz], [p.x1 * sx, p.z1 * sz],
   715→      ] as Array<[number, number]>) {
   716→        // toWorld(): x' = x·cos + z·sin, z' = −x·sin + z·cos
   717→        const wx = this.pos.x + lx * cosH + lz * sinH;
   718→        const wz = this.pos.z - lx * sinH + lz * cosH;
   719→        if (wx < minX) minX = wx;
   720→        if (wx > maxX) maxX = wx;
   721→        if (wz < minZ) minZ = wz;
   722→        if (wz > maxZ) maxZ = wz;
   723→      }
   724→      this.colReg.add(`ship_${this.id}_${i}`, {
   725→        kind: 'box', minX, maxX, minZ, maxZ, minY: y0, maxY: y1,
   726→      });
   727→    }
   728→    this.colSynced = true;
   729→    this.lastColX = this.pos.x;
   730→    this.lastColZ = this.pos.z;
   731→    this.lastColHeading = this.heading;
   732→    this.lastColY = gy;
   733→  }
   734→
   735→  private clearColliders(): void {
   736→    if (!this.colReg) return;
   737→    for (let i = 0; i < Ship.COLLIDER_PARTS.length; i++) {
   738→      this.colReg.remove(`ship_${this.id}_${i}`);
   739→    }
   740→    this.colSynced = false;
   741→  }
   742→
   743→  deserialize(d: ReturnType<Ship['serialize']>): void {
   744→    this.pos.set(d.x, 0, d.z);
   745→    this.heading = d.heading;
   746→    this.hullHp = d.hullHp;
   747→    this.sailState = d.sailState;
   748→    this.anchored = d.anchored;
   749→    this.sinking = d.sinking;
   750→    this.sunk = d.sunk;
   751→    this.group.visible = !d.sunk;
   752→    this.speed = 0;
   753→    this.syncTransform();
   754→    this.group.position.y = 0.12;
   755→  }
   756→
   757→  dispose(): void {
   758→    this.clearColliders(); // Fase 9 — free the hull AABBs
   759→    this.group.parent?.remove(this.group);
   760→    this.group.traverse((obj) => {
   761→      if ((obj as THREE.Mesh).isMesh) {
   762→        (obj as THREE.Mesh).geometry.dispose();
   763→      }
   764→    });
   765→  }
