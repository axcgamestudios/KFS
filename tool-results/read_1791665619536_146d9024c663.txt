    42→function box(w: number, h: number, d: number, color: number, rough = 0.85): THREE.Mesh {
    43→  const m = new THREE.Mesh(
    44→    new THREE.BoxGeometry(w, h, d),
    45→    new THREE.MeshStandardMaterial({ color, roughness: rough, metalness: 0.02 }),
    46→  );
    47→  m.castShadow = true;
    48→  return m;
    49→}

   249→  private buildWeapon(kind: WeaponKind, hip: THREE.Group, hand: THREE.Group): void {
   250→    const steel = new THREE.MeshStandardMaterial({ color: 0xaab0b6, roughness: 0.35, metalness: 0.85 });
   251→    const darkSteel = new THREE.MeshStandardMaterial({ color: 0x767c82, roughness: 0.45, metalness: 0.75 });
   252→    const guardM = new THREE.MeshStandardMaterial({ color: 0xc9a227, roughness: 0.4, metalness: 0.7 });
   253→    const woodM = new THREE.MeshStandardMaterial({ color: 0x4a3624, roughness: 0.85 });
   254→    const gripM = new THREE.MeshStandardMaterial({ color: 0x3a2a1a, roughness: 0.9 });
   255→
   256→    if (kind === 'sword') {
   257→      const mk = (parent: THREE.Group) => {
   258→        const blade = new THREE.Mesh(new THREE.BoxGeometry(0.045, 0.78, 0.012), steel);
   259→        blade.position.y = 0.52;
   260→        blade.castShadow = true;
   261→        parent.add(blade);
   262→        const tip = new THREE.Mesh(new THREE.ConeGeometry(0.033, 0.12, 4), steel);
   263→        tip.position.y = 0.96;
   264→        parent.add(tip);
   265→        const guard = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.035, 0.045), guardM);
   266→        guard.position.y = 0.13;
   267→        parent.add(guard);
   268→        const grip = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.022, 0.22, 6), gripM);
   269→        grip.position.y = 0.0;
   270→        parent.add(grip);
   271→        const pommel = new THREE.Mesh(new THREE.SphereGeometry(0.032, 6, 5), guardM);
   272→        pommel.position.y = -0.12;
   273→        parent.add(pommel);
   274→      };
   275→      mk(hip);
   276→      mk(hand);
   277→      return;
   278→    }
   279→
   280→    if (kind === 'axe') {
   281→      const mk = (parent: THREE.Group) => {
   282→        const haft = new THREE.Mesh(new THREE.CylinderGeometry(0.024, 0.028, 0.92, 7), woodM);
   283→        haft.position.y = 0.42;
   284→        haft.castShadow = true;
   285→        parent.add(haft);
   286→        // Axe head: two flaring blades joined at the socket
   287→        const headCore = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.16, 0.1), darkSteel);
   288→        headCore.position.set(0, 0.82, 0);
   289→        parent.add(headCore);
   290→        for (const s of [-1, 1]) {
   291→          const blade = new THREE.Mesh(new THREE.CylinderGeometry(0.15, 0.05, 0.03, 3), darkSteel);
   292→          blade.rotation.set(0, 0, -s * Math.PI / 2);
   293→          blade.rotation.x = Math.PI / 2;
   294→          blade.position.set(s * 0.13, 0.82, 0);
   295→          blade.castShadow = true;
   296→          parent.add(blade);
   297→        }
   298→        const grip = new THREE.Mesh(new THREE.CylinderGeometry(0.026, 0.026, 0.18, 7), gripM);
   299→        grip.position.y = 0.02;
   300→        parent.add(grip);
   301→        const pommel = new THREE.Mesh(new THREE.SphereGeometry(0.03, 6, 5), guardM);
   302→        pommel.position.y = -0.06;
   303→        parent.add(pommel);
   304→      };
   305→      mk(hip);
   306→      mk(hand);
   307→      return;
   308→    }
   309→
   310→    // spear
   311→    const mk = (parent: THREE.Group) => {
   312→      const shaft = new THREE.Mesh(new THREE.CylinderGeometry(0.021, 0.023, 1.72, 7), woodM);
   313→      shaft.position.y = 0.55;
   314→      shaft.castShadow = true;
   315→      parent.add(shaft);
   316→      const socket = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.024, 0.1, 7), guardM);
   317→      socket.position.y = 1.42;
   318→      parent.add(socket);
   319→      const tip = new THREE.Mesh(new THREE.ConeGeometry(0.045, 0.3, 6), steel);
   320→      tip.position.y = 1.6;
   321→      tip.castShadow = true;
   322→      parent.add(tip);
   323→      const butt = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.016, 0.12, 6), darkSteel);
   324→      butt.position.y = -0.32;
   325→      parent.add(butt);
   326→    };
   327→    mk(hip);
   328→    mk(hand);
   329→  }

   331→  private buildBow(parent: THREE.Group): void {
   332→    const woodM = new THREE.MeshStandardMaterial({ color: 0x5a4128, roughness: 0.75 });
   333→    // Arc of the bow: 5 short segments following a curve
   334→    const segs = 6;
   335→    for (let i = 0; i < segs; i++) {
   336→      const k = i / (segs - 1);          // 0..1 along the bow
   337→      const ang = (k - 0.5) * 1.9;       // sweep
   338→      const seg = new THREE.Mesh(new THREE.CylinderGeometry(0.014, 0.017, 0.19, 5), woodM);
   339→      seg.position.set(Math.cos(ang) * 0.12 - 0.12, Math.sin(ang) * 0.52, 0);
   340→      seg.rotation.z = -ang + Math.PI / 2;
   341→      seg.castShadow = true;
   342→      parent.add(seg);
   343→    }
   344→    // String
   345→    const stringM = new THREE.MeshStandardMaterial({ color: 0xd8d2c0, roughness: 0.9 });
   346→    const str = new THREE.Mesh(new THREE.CylinderGeometry(0.004, 0.004, 1.04, 4), stringM);
   347→    str.position.set(-0.085, 0, 0);
   348→    parent.add(str);
   349→  }

   351→  private buildShield(parent: THREE.Group): void {
   352→    // Heater shield: extruded 2D silhouette + iron boss + scale bands
   353→    const shape = new THREE.Shape();
   354→    shape.moveTo(-0.26, 0.32);
   355→    shape.quadraticCurveTo(0, 0.4, 0.26, 0.32);
   356→    shape.quadraticCurveTo(0.3, 0.05, 0.2, -0.18);
   357→    shape.quadraticCurveTo(0.1, -0.36, 0, -0.44);
   358→    shape.quadraticCurveTo(-0.1, -0.36, -0.2, -0.18);
   359→    shape.quadraticCurveTo(-0.3, 0.05, -0.26, 0.32);
   360→    const geo = new THREE.ExtrudeGeometry(shape, { depth: 0.035, bevelEnabled: true, bevelThickness: 0.012, bevelSize: 0.012, bevelSegments: 1 });
   361→    const woodM = new THREE.MeshStandardMaterial({ color: 0x4c3a26, roughness: 0.8 });
   362→    const shield = new THREE.Mesh(geo, woodM);
   363→    shield.castShadow = true;
   364→    parent.add(shield);
   365→    // Iron boss
   366→    const boss = new THREE.Mesh(
   367→      new THREE.SphereGeometry(0.075, 10, 8, 0, Math.PI * 2, 0, Math.PI / 2),
   368→      new THREE.MeshStandardMaterial({ color: 0x8a9198, roughness: 0.35, metalness: 0.85 }),
   369→    );
   370→    boss.rotation.x = Math.PI / 2;
   371→    boss.position.z = 0.03;
   372→    boss.castShadow = true;
   373→    parent.add(boss);
   374→    // Scale bands (dragon-scale look, tide blue)
   375→    const scaleM = new THREE.MeshStandardMaterial({ color: 0x2e5a5e, roughness: 0.5, metalness: 0.35 });
   376→    for (let row = 0; row < 3; row++) {
   377→      for (let i = 0; i < 4; i++) {
   378→        const scale = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.055, 0.012, 6), scaleM);
   379→        scale.rotation.x = Math.PI / 2;
   380→        scale.position.set(-0.15 + i * 0.1 + (row % 2) * 0.05, 0.2 - row * 0.19, 0.032);
   381→        parent.add(scale);
   382→      }
   383→    }
   384→  }

   424→  private refreshWeaponVisibility(): void {
   425→    for (const kind of ['sword', 'axe', 'spear'] as WeaponKind[]) {
   426→      const hip = this.hipSlots[kind];
   427→      const hand = this.handSlots[kind];
   428→      if (hip) hip.visible = this.weapon === kind && this.weaponEquippedVisible;
   429→      if (hand) hand.visible = false; // shown only mid-swing by update()
   430→    }
   431→  }

   433→  /** Face target yaw (world). */
   434→  faceYaw(yaw: number): void {
   435→    this.root.rotation.y = yaw;
   436→  }
   437→
   438→  update(p: AnimParams): void {
   439→    this.t += p.dt;
   440→    const walkFreq = p.swimming ? 2.2 : p.speed > 4.5 ? 9.2 : 6.4;
   441→    if (p.moving) this.phase += p.dt * walkFreq;
   442→    const ph = this.phase;
   443→
   444→    // Damage flash decay (discrete hit feedback)
   445→    if (this.flashT > 0) {
   446→      this.flashT = Math.max(0, this.flashT - p.dt * 3.2);
   447→      const e = this.flashT * 0.55;
   448→      for (const m of this.flashMaterials) m.emissive.setRGB(e, e * 0.12, e * 0.08);
   449→    }
   450→
   451→    const resetPose = () => {
   452→      this.legL.rotation.x = 0; this.legR.rotation.x = 0;
   453→      this.kneeL.rotation.x = 0; this.kneeR.rotation.x = 0;
   454→      this.armL.rotation.x = 0; this.armR.rotation.x = 0;
   455→      this.armL.rotation.z = 0.06; this.armR.rotation.z = -0.06;
   456→      this.elbowL.rotation.x = 0; this.elbowR.rotation.x = 0;
   457→      this.chest.rotation.x = 0; this.chest.rotation.y = 0;
   458→      this.hips.position.y = 0.94; this.head.rotation.x = 0; this.head.rotation.y = 0;
   459→      this.root.rotation.x = 0;
   460→    };
   461→    resetPose();
   462→
   463→    // --- Dodge roll: full-body tumble, overrides everything else ---
   464→    const rollT = p.rollT ?? 0;
   465→    if (rollT > 0) {
   466→      this.root.rotation.x = rollT * Math.PI * 2;
   467→      this.hips.position.y = 0.94 - 0.32 * Math.sin(rollT * Math.PI);
   468→      this.legL.rotation.x = -1.9; this.legR.rotation.x = -1.7;
   469→      this.kneeL.rotation.x = -1.9; this.kneeR.rotation.x = -1.7;
   470→      this.armL.rotation.x = -1.4; this.armR.rotation.x = -1.2;
   471→      this.elbowL.rotation.x = -1.6; this.elbowR.rotation.x = -1.6;
   472→      this.chest.rotation.x = 0.7;
   473→      this.head.rotation.x = 0.5;
   474→      if (this.cape) this.cape.rotation.x = 1.2;
   475→      this.updateWeaponPose(p, false);
   476→      return;
   477→    }
   478→
   479→    if (p.swimming) {
   480→      // Breaststroke: body pitched forward, sweeping arms, flutter kick
   481→      this.hips.position.y = 0.94;
   482→      this.root.rotation.x = 0; // pitch handled by Player on a parent group
   483→      const sweep = Math.sin(ph * 2.2);
   484→      this.armL.rotation.x = -1.2 + sweep * 1.1;
   485→      this.armR.rotation.x = -1.2 + sweep * 1.1;
   486→      this.armL.rotation.z = 0.35 + sweep * 0.3;
   487→      this.armR.rotation.z = -0.35 - sweep * 0.3;
   488→      this.elbowL.rotation.x = -0.5;
   489→      this.elbowR.rotation.x = -0.5;
   490→      this.legL.rotation.x = 0.25 + Math.sin(ph * 6) * 0.3;
   491→      this.legR.rotation.x = 0.25 + Math.sin(ph * 6 + Math.PI) * 0.3;
   492→      this.kneeL.rotation.x = -0.2;
   493→      this.kneeR.rotation.x = -0.2;
   494→      this.head.rotation.x = -0.5;
   495→      if (this.cape) this.cape.rotation.x = 0.9;
   496→      this.updateWeaponPose(p, false);
   497→      return;
   498→    }
   499→
   500→    if (p.riding) {
   501→      // Fase 4 — seat pose: legs wrap the saddle, hands hold the reins,
   502→      // torso leans into the gait with a two-phase bounce.
   503→      const bounce = p.rideBounce ?? 0;
   504→      const bob = Math.sin(bounce * Math.PI * 2);
   505→      this.hips.position.y = 0.94 + Math.abs(bob) * 0.05;
   506→      this.root.rotation.x = 0;
   507→      // Legs bent outward around the horse's barrel
   508→      this.legL.rotation.x = -0.85;
   509→      this.legR.rotation.x = -0.85;
   510→      this.kneeL.rotation.x = -1.35;
   511→      this.kneeR.rotation.x = -1.35;
   512→      this.legL.rotation.z = 0.28;
   513→      this.legR.rotation.z = -0.28;
   514→      // Arms forward on the reins
   515→      this.armL.rotation.x = -0.55 + bob * 0.06;
   516→      this.armR.rotation.x = -0.55 + bob * 0.06;
   517→      this.armL.rotation.z = 0.18;
   518→      this.armR.rotation.z = -0.18;
   519→      this.elbowL.rotation.x = -0.5;
   520→      this.elbowR.rotation.x = -0.5;
   521→      this.chest.rotation.x = 0.08 + bob * 0.03;
   522→      this.head.rotation.x = -0.08;
   523→      if (this.cape) this.cape.rotation.x = 0.35 + Math.abs(bob) * 0.12;
   524→      this.updateWeaponPose(p, false);
   525→      return;
   526→    }
   527→
   528→    if (!p.grounded) {
   529→      // Airborne pose
   530→      this.armL.rotation.x = -0.7; this.armR.rotation.x = -0.5;
   531→      this.armL.rotation.z = 0.5; this.armR.rotation.z = -0.5;
   532→      this.legL.rotation.x = -0.5; this.legR.rotation.x = 0.35;
   533→      this.kneeL.rotation.x = -0.7; this.kneeR.rotation.x = -0.3;
   534→      this.chest.rotation.x = 0.12;
   535→      if (this.cape) this.cape.rotation.x = 0.55 + Math.sin(this.t * 6) * 0.06;
   536→      this.updateWeaponPose(p, false);
   537→      return;
   538→    }
   539→
   540→    // --- Bow draw aim pose (overrides walk arms while aiming) ---
   541→    const bowDraw = p.bowDraw ?? 0;
   542→    if (bowDraw > 0) {
   543→      const sway = Math.sin(this.t * 1.8) * 0.015;
   544→      this.armL.rotation.x = -1.45 + sway;
   545→      this.armL.rotation.z = 0.12;
   546→      this.elbowL.rotation.x = -0.1;
   547→      this.armR.rotation.x = -1.2 + bowDraw * 0.25 + sway;
   548→      this.armR.rotation.z = -0.3;
   549→      this.elbowR.rotation.x = -0.25 - bowDraw * 1.15;
   550→      this.chest.rotation.y = -0.22;
   551→      this.chest.rotation.x = 0.05;
   552→      this.head.rotation.y = 0.12;
   553→      if (p.moving) {
   554→        const amp = 0.28;
   555→        this.legL.rotation.x = Math.sin(ph) * amp;
   556→        this.legR.rotation.x = Math.sin(ph + Math.PI) * amp;
   557→        this.hips.position.y = 0.94 + Math.abs(Math.sin(ph)) * 0.03;
   558→      } else {
   559→        this.kneeL.rotation.x = -0.12;
   560→        this.kneeR.rotation.x = -0.18;
   561→        this.hips.position.y = 0.94 - bowDraw * 0.03;
   562→      }
   563→      if (this.cape) this.cape.rotation.x = 0.18 + Math.sin(this.t * 2) * 0.03;
   564→      this.updateWeaponPose(p, true);
   565→      return;
   566→    }
   567→
   568→    if (p.moving) {
   569→      const runK = Math.min(1, p.speed / 6.0);
   570→      const amp = 0.5 + runK * 0.42;
   571→      this.legL.rotation.x = Math.sin(ph) * amp;
   572→      this.legR.rotation.x = Math.sin(ph + Math.PI) * amp;
   573→      this.kneeL.rotation.x = Math.max(0, -Math.sin(ph - 0.6)) * (0.5 + runK * 0.5);
   574→      this.kneeR.rotation.x = Math.max(0, -Math.sin(ph + Math.PI - 0.6)) * (0.5 + runK * 0.5);
   575→      this.armL.rotation.x = Math.sin(ph + Math.PI) * (0.35 + runK * 0.45);
   576→      this.armR.rotation.x = Math.sin(ph) * (0.35 + runK * 0.45);
   577→      this.armL.rotation.z = 0.08; this.armR.rotation.z = -0.08;
   578→      this.elbowL.rotation.x = -0.25 - runK * 0.3;
   579→      this.elbowR.rotation.x = -0.25 - runK * 0.3;
   580→      // Body bob & forward lean
   581→      this.hips.position.y = 0.94 + Math.abs(Math.sin(ph)) * 0.045 * (0.5 + runK);
   582→      this.chest.rotation.x = 0.05 + runK * 0.14;
   583→      this.chest.rotation.y = Math.sin(ph) * 0.06;
   584→      if (this.cape) this.cape.rotation.x = 0.12 + runK * 0.5 + Math.sin(this.t * 5) * 0.05;
   585→    } else {
   586→      // Idle: breathing, subtle sway, weight shift
   587→      const breathe = Math.sin(this.t * 1.6);
   588→      this.chest.rotation.x = 0.015 + breathe * 0.012;
   589→      this.chest.scale.y = 1 + breathe * 0.008;
   590→      this.armL.rotation.x = breathe * 0.03;
   591→      this.armR.rotation.x = -breathe * 0.03;
   592→      this.armL.rotation.z = 0.07 + breathe * 0.012;
   593→      this.armR.rotation.z = -0.07 - breathe * 0.012;
   594→      this.head.rotation.y = Math.sin(this.t * 0.5) * 0.14;
   595→      this.hips.position.y = 0.94 + breathe * 0.006;
   596→      if (p.talking) {
   597→        this.head.rotation.y = Math.sin(this.t * 2.2) * 0.06;
   598→        this.armR.rotation.x = Math.sin(this.t * 3.1) * 0.22 - 0.15;
   599→        this.elbowR.rotation.x = -0.45;
   600→      }
   601→      if (this.cape) this.cape.rotation.x = 0.06 + Math.sin(this.t * 1.2) * 0.02;
   602→    }
   603→
   604→    // --- Block pose overlay (right mouse hold) ---
   605→    const blockT = p.blockT ?? 0;
   606→    if (blockT > 0) {
   607→      const k = Math.min(1, blockT);
   608→      if (this.shieldGroup.visible) {
   609→        // Shield forward, braced
   610→        this.armL.rotation.x = THREE.MathUtils.lerp(this.armL.rotation.x, -1.15, k);
   611→        this.armL.rotation.z = THREE.MathUtils.lerp(this.armL.rotation.z, 0.3, k);
   612→        this.elbowL.rotation.x = THREE.MathUtils.lerp(this.elbowL.rotation.x, -0.85, k);
   613→        this.armR.rotation.x = THREE.MathUtils.lerp(this.armR.rotation.x, -0.35, k);
   614→        this.elbowR.rotation.x = THREE.MathUtils.lerp(this.elbowR.rotation.x, -1.5, k);
   615→      } else {
   616→        // Sword held vertical in front
   617→        this.armR.rotation.x = THREE.MathUtils.lerp(this.armR.rotation.x, -0.95, k);
   618→        this.armR.rotation.z = THREE.MathUtils.lerp(this.armR.rotation.z, -0.25, k);
   619→        this.elbowR.rotation.x = THREE.MathUtils.lerp(this.elbowR.rotation.x, -1.35, k);
   620→        this.armL.rotation.x = THREE.MathUtils.lerp(this.armL.rotation.x, -0.5, k);
   621→        this.elbowL.rotation.x = THREE.MathUtils.lerp(this.elbowL.rotation.x, -1.0, k);
   622→      }
   623→      this.chest.rotation.x = THREE.MathUtils.lerp(this.chest.rotation.x, 0.12, k);
   624→      this.hips.position.y -= k * 0.06;
   625→    }
   626→
   627→    // --- Attack overlays ---
   628→    if (p.attackT > 0) {
   629→      const k = p.attackT;
   630→      let armX: number, armZ: number, chestY: number, elbow = -0.15;
   631→      const kind = p.attackKind ?? 'light1';
   632→      if (kind === 'heavy') {
   633→        // Overhead smash: big windup up-back, crouch, then slam down
   634→        let ax: number, az: number, cy: number, hipY = 0;
   635→        if (k < 0.42) {
   636→          const u = k / 0.42;
   637→          ax = THREE.MathUtils.lerp(0, -2.7, u);
   638→          az = THREE.MathUtils.lerp(-0.08, -0.35, u);
   639→          cy = THREE.MathUtils.lerp(0, -0.42, u);
   640→          hipY = -0.1 * u;
   641→        } else if (k < 0.68) {
   642→          const u = (k - 0.42) / 0.26;
   643→          ax = THREE.MathUtils.lerp(-2.7, 0.85, u);
   644→          az = THREE.MathUtils.lerp(-0.35, 0, u);
   645→          cy = THREE.MathUtils.lerp(-0.42, 0.38, u);
   646→          hipY = -0.1 * (1 - u);
   647→        } else {
   648→          const u = (k - 0.68) / 0.32;
   649→          ax = THREE.MathUtils.lerp(0.85, 0, u);
   650→          az = THREE.MathUtils.lerp(0, -0.06, u);
   651→          cy = THREE.MathUtils.lerp(0.38, 0, u);
   652→        }
   653→        this.armR.rotation.x = ax;
   654→        this.armR.rotation.z = az;
   655→        this.chest.rotation.y = cy;
   656→        this.hips.position.y += hipY;
   657→        this.chest.rotation.x = 0.18 * Math.sin(k * Math.PI);
   658→        this.elbowR.rotation.x = -0.5 + Math.max(0, k - 0.42) * 1.6;
   659→      } else if (kind === 'thrust') {
   660→        // Spear jab: pull back, lunge forward
   661→        if (k < 0.35) {
   662→          const u = k / 0.35;
   663→          this.armR.rotation.x = THREE.MathUtils.lerp(0, -0.5, u);
   664→          this.elbowR.rotation.x = THREE.MathUtils.lerp(-0.15, -1.7, u);
   665→          this.chest.rotation.y = THREE.MathUtils.lerp(0, 0.35, u);
   666→        } else if (k < 0.6) {
   667→          const u = (k - 0.35) / 0.25;
   668→          this.armR.rotation.x = THREE.MathUtils.lerp(-0.5, -1.5, u);
   669→          this.elbowR.rotation.x = THREE.MathUtils.lerp(-1.7, -0.05, u);
   670→          this.chest.rotation.y = THREE.MathUtils.lerp(0.35, -0.3, u);
   671→          this.chest.rotation.x = 0.15 * u;
   672→        } else {
   673→          const u = (k - 0.6) / 0.4;
   674→          this.armR.rotation.x = THREE.MathUtils.lerp(-1.5, 0, u);
   675→          this.elbowR.rotation.x = THREE.MathUtils.lerp(-0.05, -0.15, u);
   676→          this.chest.rotation.y = THREE.MathUtils.lerp(-0.3, 0, u);
   677→        }
   678→      } else if (kind === 'light2') {
   679→        // Backhand: windup right, sweep left
   680→        if (k < 0.35) {
   681→          const u = k / 0.35;
   682→          armX = THREE.MathUtils.lerp(0, -1.9, u);
   683→          armZ = THREE.MathUtils.lerp(-0.08, 0.55, u);
   684→          chestY = THREE.MathUtils.lerp(0, 0.4, u);
   685→        } else if (k < 0.75) {
   686→          const u = (k - 0.35) / 0.4;
   687→          armX = THREE.MathUtils.lerp(-1.9, 0.4, u);
   688→          armZ = THREE.MathUtils.lerp(0.55, -0.85, u);
   689→          chestY = THREE.MathUtils.lerp(0.4, -0.45, u);
   690→        } else {
   691→          const u = (k - 0.75) / 0.25;
   692→          armX = THREE.MathUtils.lerp(0.4, 0, u);
   693→          armZ = THREE.MathUtils.lerp(-0.85, -0.06, u);
   694→          chestY = THREE.MathUtils.lerp(-0.45, 0, u);
   695→        }
   696→        this.armR.rotation.x = armX!;
   697→        this.armR.rotation.z = armZ!;
   698→        this.chest.rotation.y = chestY!;
   699→        this.elbowR.rotation.x = elbow;
   700→      } else if (kind === 'bowRelease') {
   701→        // String snap: right hand flies forward off the cheek
   702→        const u = Math.min(1, k * 2.2);
   703→        this.armL.rotation.x = -1.45;
   704→        this.elbowL.rotation.x = -0.1;
   705→        this.armR.rotation.x = THREE.MathUtils.lerp(-1.45, -1.2, u);
   706→        this.elbowR.rotation.x = THREE.MathUtils.lerp(-1.4, -0.2, u);
   707→        this.chest.rotation.y = -0.22;
   708→      } else {
   709→        // light1 — classic diagonal slash (windup → slash → recover)
   710→        if (k < 0.35) {
   711→          const u = k / 0.35;
   712→          armX = THREE.MathUtils.lerp(this.armR.rotation.x, -2.3, u);
   713→          armZ = THREE.MathUtils.lerp(-0.08, -0.9, u);
   714→          chestY = THREE.MathUtils.lerp(0, -0.4, u);
   715→        } else if (k < 0.75) {
   716→          const u = (k - 0.35) / 0.4;
   717→          armX = THREE.MathUtils.lerp(-2.3, 0.5, u);
   718→          armZ = THREE.MathUtils.lerp(-0.9, 0.55, u);
   719→          chestY = THREE.MathUtils.lerp(-0.4, 0.45, u);
   720→        } else {
   721→          const u = (k - 0.75) / 0.25;
   722→          armX = THREE.MathUtils.lerp(0.5, 0, u);
   723→          armZ = THREE.MathUtils.lerp(0.55, -0.06, u);
   724→          chestY = THREE.MathUtils.lerp(0.45, 0, u);
   725→        }
   726→        this.armR.rotation.x = armX!;
   727→        this.armR.rotation.z = armZ!;
   728→        this.chest.rotation.y = chestY!;
   729→        this.elbowR.rotation.x = elbow;
   730→      }
   731→    }
   732→
   733→    this.updateWeaponPose(p, p.attackT > 0);
   734→  }

   736→  /** Show hip/hand weapon copies + bow according to the current context. */
   737→  private updateWeaponPose(p: AnimParams, attacking: boolean): void {
   738→    const hip = this.hipSlots[this.weapon];
   739→    const showHip = this.weaponEquippedVisible && this.weapon !== 'none';
   740→    if (hip) hip.visible = showHip && !attacking;
   741→    // Hand copy: visible only mid-swing for melee weapons
   742→    if (this.weapon !== 'none' && this.weapon !== 'bow') {
   743→      const hand = this.handSlots[this.weapon];
   744→      if (hand) hand.visible = attacking;
   745→    }
   746→    // Bow: visible only while aiming/drawing (held in the left hand)
   747→    const bowDraw = p.bowDraw ?? 0;
   748→    this.bowInHand.visible = this.weapon === 'bow' && (bowDraw > 0 || (p.attackKind === 'bowRelease' && p.attackT > 0));
   749→  }
   750→
   751→  dispose(): void {
   752→    this.root.traverse((obj) => {
   753→      if ((obj as THREE.Mesh).isMesh) {
   754→        const m = obj as THREE.Mesh;
   755→        m.geometry.dispose();
   756→        (m.material as THREE.Material).dispose();
   757→      }
   758→    });
   759→  }
