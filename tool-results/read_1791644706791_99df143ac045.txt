   389→  buildSaveData(): SaveData {
   390→    return {
   391→      version: 1,
   392→      timestamp: Date.now(),

   394→      time: this.world!.sky.serialize(),

   399→  autosave(): void {
   400→    if (!this.world || !this.player) return;
   401→    if (SaveSystem.autosave(this.buildSaveData())) {
   402→      useGameStore.getState().setHasSave(true);
   403→    }
   404→  }

   406→  saveToSlot(slot: number): boolean {
   407→    const ok = SaveSystem.save(slot, this.buildSaveData());
   408→    if (ok) {
   409→      useGameStore.getState().setHasSave(true);
   410→      useGameStore.getState().notify('save', 'Partida guardada', `Ranura ${slot + 1} · Día ${this.world!.sky.dayCount}`);
   411→      this.audio.playEvent('unlock');
   412→    }
   413→    return ok;
   414→  }
   415→
   416→  loadFromSlot(slot: number): boolean {
   417→    const data = SaveSystem.load(slot);
   418→    if (!data || !this.world || !this.player) return false;
   419→    this.state.deserialize(data.gameState);
   420→    this.player.deserialize(data.player);
   421→    this.world.sky.deserialize(data.time);
   422→    this.dead = false;
   423→    this.restoreWorldFromState();
   424→    this.applyStats();
   425→    this.syncProgression();
   426→    const store = useGameStore.getState();
   427→    store.setPhase('playing');
   428→    this.input.requestLock();
   429→    store.notify('save', 'Partida cargada', `Día ${data.time.dayCount} · Nivel ${data.gameState.level}`);
   430→    return true;
   431→  }

   433→  /** Sync chests/herbs visuals with loaded state. */
   434→  private restoreWorldFromState(): void {
   435→    const w = this.world!;
   436→    for (const chest of w.chests) {
   437→      const opened = this.state.openedChests.has(chest.id);
   438→      if (opened && !chest.handles.isOpen()) chest.handles.open();
   439→      if (!opened && chest.handles.isOpen()) {
   440→        chest.handles.lid.rotation.x = 0;
   441→      }
   442→    }
   443→    for (const herb of w.herbs) {
   444→      herb.mesh.visible = !this.state.pickedHerbs.has(herb.id);
   445→    }
   446→    this.player!.model.setEquippedState(!!this.state.equippedWeapon);
   447→  }

   451→  private applyStats(): void {
   452→    const p = this.player!;
   453→    const s = this.state;
   454→    let maxHp = 100;
   455→    let maxSt = 100;
   456→    let speedMult = 1;
   457→    if (s.learnedSkills.has('comb_piel')) maxHp += 20;
   458→    if (s.learnedSkills.has('sup_corazon')) maxHp += 25;
   459→    if (s.learnedSkills.has('comb_vigor')) maxSt += 20;
   460→    if (s.learnedSkills.has('comb_vigor2')) maxSt += 20;
   461→    if (s.learnedSkills.has('sup_zancada')) speedMult = 1.08;
   462→    p.stats.maxHealth = maxHp;
   463→    p.stats.maxStamina = maxSt;
   464→    p.stats.walkSpeed = 3.6 * speedMult;
   465→    p.stats.runSpeed = 6.6 * speedMult;
   466→    p.stats.swimSpeed = 2.3 * speedMult;
   467→    p.health = Math.min(p.health, maxHp);
   468→    p.stamina = Math.min(p.stamina, maxSt);
   469→  }

   479→    });

   484→  private update(dt: number): void {
   485→    if (!this.world || !this.player || !this.rig) return;
   486→    const store = useGameStore.getState();
   487→    const w = this.world;
   488→    const p = this.player;
   489→    windTime.value += dt;
   490→
   491→    switch (store.phase) {
   492→      case 'menu': {
   493→        this.menuOrbit += dt * 0.021;
   494→        const r = 150;
   495→        const cx = 20 + Math.cos(this.menuOrbit) * r;
   496→        const cz = 200 + Math.sin(this.menuOrbit) * r;
   497→        const cy = w.terrainHeight.height(cx, cz) + 42;
   498→        this.rig.setManual(new THREE.Vector3(cx, cy, cz), new THREE.Vector3(20, 12, 170));
   499→        w.sky.dayFraction = 0.78;
   500→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   501→        w.water.update(dt, this.engine.camera);
   502→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   503→        w.gulls.update(dt);
   504→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   505→        w.terrain.update(cx, cz);
   506→        w.vegetation.update();
   507→        break;
   508→      }
   509→      case 'intro': {
   510→        this.updateIntro(dt);
   511→        w.sky.tick(dt);
   512→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   513→        w.water.update(dt, this.engine.camera);
   514→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   515→        w.gulls.update(dt);
   516→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   517→        break;
   518→      }
   519→      case 'playing': {
   520→        // Toggle overlays (works whether or not one is already open)
   521→        const toggle = this.input.menuTogglePressed();
   522→        if (toggle && !store.dialogue) {
   523→          const map: Record<string, 'inventory' | 'map' | 'journal' | 'skills'> = {
   524→            inventory: 'inventory', map: 'map', journal: 'journal', skills: 'skills',
   525→          };
   526→          const target = map[toggle];
   527→          if (store.overlay === target) {
   528→            store.closeOverlay();
   529→            this.input.requestLock();
   530→          } else {
   531→            store.openOverlay(target);
   532→            this.input.releaseLock();
   533→          }
   534→        }
   535→        if (!store.overlay) {
   536→          this.updatePlaying(dt, store);
   537→        } else {
   538→          // Overlays pause input but keep rendering a frozen frame
   539→          w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   540→        }
   541→        break;
   542→      }
   543→      case 'paused': {
   544→        // frozen — nothing updates
   545→        break;
   546→      }
   547→      default:
   548→        break;
   549→    }
   550→    this.input.endFrame();
   551→  }

   553→  private updateIntro(dt: number): void {
   554→    const store = useGameStore.getState();
   555→    // Skip
   556→    if (this.input.anyPressed('Space', 'Escape', 'Enter', 'KeyE')) {
   557→      this.finishIntro();
   558→      return;
   559→    }
   560→    this.introT += dt;
   561→    let shot = 0;
   562→    let t0 = 0;
   563→    for (let i = 0; i < INTRO_SHOTS.length; i++) {
   564→      if (this.introT < t0 + INTRO_SHOTS[i].dur) {
   565→        shot = i;
   566→        break;
   567→      }
   568→      t0 += INTRO_SHOTS[i].dur;
   569→      shot = i;
   570→    }
   571→    const totalDur = INTRO_SHOTS.reduce((a, s) => a + s.dur, 0);
   572→    if (this.introT >= totalDur) {
   573→      this.finishIntro();
   574→      return;
   575→    }
   576→    const s = INTRO_SHOTS[shot];
   577→    const k = smoothstep(0, 1, clamp((this.introT - t0) / s.dur, 0, 1));
   578→    const pos = s.from.clone().lerp(s.to, k);
   579→    const look = s.lookFrom.clone().lerp(s.lookTo, k);
   580→    this.rig!.setManual(pos, look);
   581→    // Only push to the store when the shot changes (avoids per-frame renders)
   582→    if (this.lastIntroShot !== shot) {
   583→      this.lastIntroShot = shot;
   584→      store.setIntro({
   585→        text: s.text,
   586→        shot,
   587→        total: INTRO_SHOTS.length,
   588→        title: shot === INTRO_SHOTS.length - 1,
   589→      });
   590→    }
   591→  }

   593→  private updatePlaying(dt: number, store: ReturnType<typeof useGameStore.getState>): void {
   594→    const w = this.world!;
   595→    const p = this.player!;
   596→    const rig = this.rig!;
   597→    const dialogueActive = !!store.dialogue;
   598→
   599→    // --- Death / respawn ---
   600→    if (p.health <= 0 && !this.dead) {
   601→      this.dead = true;
   602→      const sp = w.village.spawnPoint;
   603→      p.teleport(sp.x, 1.5, sp.z);
   604→      p.health = Math.floor(p.stats.maxHealth * 0.5);
   605→      store.notify('info', 'Las olas te devuelven a la orilla', 'Despiertas en el muelle, empapado pero vivo.');
   606→      this.audio.playEvent('splash');
   607→    }
   608→
   609→    // --- Clock ---
   610→    const newDay = w.sky.tick(dt);
   611→    if (newDay) {
   612→      store.notify('info', `Día ${w.sky.dayCount} en Aldmor`, formatClock(w.sky.dayFraction));
   613→    }
   614→
   615→    // --- Camera input & player ---
   616→    rig.handleInput(this.input, dt);
   617→    p.update(dt, this.input, rig.yawForward, w.terrainHeight, this.colliders, dialogueActive || this.dead);
   618→
   619→    // --- Attack (LMB) ---
   620→    if (!dialogueActive && this.input.leftPressed && p.attackT === 0 && this.state.equippedWeapon) {
   621→      p.attackT = 0.001;
   622→      p.onAttackSwing?.();
   623→      this.lastSwingHit = false;
   624→    }
   625→    if (p.attackT > 0 && !this.lastSwingHit && p.attackT >= 0.45) {
   626→      this.lastSwingHit = true;
   627→      // Training dummy hit check
   628→      const d = w.dummyPos;
   629→      const dx = d.x - p.position.x;
   630→      const dz = d.z - p.position.z;
   631→      const dist = Math.hypot(dx, dz);
   632→      if (dist < 2.3) {
   633→        const facing = Math.sin(p.yaw) * dx + Math.cos(p.yaw) * dz;
   634→        if (facing > 0) {
   635→          this.audio.playEvent('damage');
   636→          this.state.giveXp(2);
   637→        }
   638→      }
   639→    }
   640→
   641→    // --- Lantern ---
   642→    if (!dialogueActive && this.input.pressed('lantern')) {
   643→      p.toggleLantern();
   644→      this.audio.playEvent('pickup');
   645→    }
   646→
   647→    // --- World systems ---
   648→    rig.update(dt, p.position, this.input.isDown('sprint') && !dialogueActive, w.terrainHeight, this.colliders);
   649→    w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   650→    w.water.update(dt, this.engine.camera);
   651→    w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   652→    w.terrain.update(p.position.x, p.position.z);
   653→    w.vegetation.update();
   654→    this.refreshGrass(p.position.x, p.position.z);
   655→    w.gulls.update(dt);
   656→    w.village.torches.update(dt, w.sky.nightFactor, p.position);
   657→    updateChests(dt, w.chests.map((c) => c.handles));
   658→
   659→    // Waterfall shimmer
   660→    const sheetMat = w.waterfallSheet.material as THREE.MeshStandardMaterial;
   661→    sheetMat.opacity = 0.48 + Math.sin(windTime.value * 2.2) * 0.08;
   662→    const foamMat = w.waterfallFoam.material as THREE.MeshBasicMaterial;
   663→    foamMat.opacity = 0.42 + Math.sin(windTime.value * 3.1) * 0.12;
   664→
   665→    // NPC updates
   666→    for (const npc of this.npcs) {
   667→      npc.update(dt, w.sky.dayFraction, w.terrainHeight, this.colliders, p.position);
   668→    }
   669→
   670→    // --- Interaction scan ---
   671→    this.updateInteractions(store, dialogueActive);
   672→
   673→    // --- Discoveries ---
   674→    this.discoveryTimer -= dt;
   675→    if (this.discoveryTimer <= 0) {
   676→      this.discoveryTimer = 0.5;
   677→      this.checkDiscoveries(store);
   678→    }
   679→
   680→    // --- Herb glow pulse ---
   681→    for (const herb of w.herbs) {
   682→      if (herb.mesh.visible) {
   683→        const petals = herb.mesh.children[herb.mesh.children.length - 1] as THREE.Mesh;
   684→        const m = petals.material as THREE.MeshStandardMaterial;
   685→        m.emissiveIntensity = 0.55 + Math.sin(windTime.value * 2 + herb.pos.x) * 0.25;
   686→      }
   687→    }
   688→
   689→    // --- Audio ambience ---
   690→    const terrainHNow = w.terrainHeight.height(p.position.x, p.position.z);
   691→    this.audio.update(dt, {
   692→      altitude: Math.max(0, p.position.y),
   693→      shoreDist: clamp(terrainHNow, 0, 160),
   694→      dayFactor: w.sky.dayFactor,
   695→      waterfallDist: p.position.distanceTo(w.waterfallBasin),
   696→    });
   697→
   698→    // --- Heart of Oak slow regeneration ---
   699→    if (this.state.learnedSkills.has('sup_corazon') && p.health < p.stats.maxHealth) {
   700→      p.health = Math.min(p.stats.maxHealth, p.health + 0.6 * dt);
   701→    }
   702→
   703→    // --- Music intensity decay ---
   704→    this.audio.musicIntensity += (this.musicIntensityTarget - this.audio.musicIntensity) * dt * 0.3;
   705→    this.musicIntensityTarget += (0.15 - this.musicIntensityTarget) * dt * 0.05;
   706→
   707→    // --- HUD sync ---
   708→    this.hudTimer -= dt;
   709→    if (this.hudTimer <= 0) {
   710→      this.hudTimer = 0.15;
   711→      (window as unknown as { __kotfsYaw?: number }).__kotfsYaw = rig.yaw;
   712→      (window as unknown as { __kotfsPlayer?: { x: number; y: number; z: number } }).__kotfsPlayer = {
   713→        x: p.position.x, y: p.position.y, z: p.position.z,
   714→      };
   715→      const tracker = this.state.currentTracker();
   716→      const clickHint =
   717→        !this.input.pointerLocked && !dialogueActive
   718→          ? 'Haz clic para tomar el control'
   719→          : null;
   720→      store.setHud({
   721→        health: p.health,
   722→        maxHealth: p.stats.maxHealth,
   723→        stamina: p.stamina,
   724→        maxStamina: p.stats.maxStamina,
   725→        clock: formatClock(w.sky.dayFraction),
   726→        dayCount: w.sky.dayCount,
   727→        questTracker: tracker,
   728→        fps: this.engine.fps,
   729→        prompt: clickHint ?? store.prompt,
   730→      });
   731→      this.pollQuestEvents();
   732→    }
   733→
   734→    // --- Map exploration ---
   735→    this.mapTimer -= dt;
   736→    if (this.mapTimer <= 0) {
   737→      this.mapTimer = 2;
   738→      this.markExplored(p.position.x, p.position.z, 3);
   739→    }
   740→
   741→    // --- Playtime & autosave ---
   742→    this.state.playtimeSeconds += dt;
   743→    this.autosaveTimer -= dt;
   744→    if (this.autosaveTimer <= 0) {
   745→      this.autosaveTimer = 120;
   746→      this.autosave();
   747→    }
   748→
   749→    // --- Tutorials ---
   750→    if (!this.state.hasFlag('tut_night') && w.sky.dayFactor < 0.2) {
   751→      this.state.setFlag('tut_night');
   752→      store.notify('info', 'Cae la noche', 'Pulsa F para encender tu farol. Las antorchas del pueblo iluminan los caminos.');
   753→    }
   754→    if (!this.state.hasFlag('tut_stamina') && p.stamina < 25) {
   755→      this.state.setFlag('tut_stamina');
   756→      store.notify('info', 'Resistencia baja', 'Deja de correr para recuperarla. Mejórala con habilidades (K).');
   757→    }
   758→  }

   773→  private updateInteractions(store: ReturnType<typeof useGameStore.getState>, dialogueActive: boolean): void {
   774→    const w = this.world!;
   775→    const p = this.player!;
   776→    let best: { prompt: string; run: () => void } | null = null;
   777→    let bestDist = Infinity;
   778→
   779→    if (!dialogueActive) {
   780→      for (const it of w.interactables) {
   781→        if (it.available && !it.available()) continue;
   782→        const d = it.pos.distanceTo(p.position);
   783→        if (d < it.radius && d < bestDist) {
   784→          const prompt = typeof it.prompt === 'string' ? it.prompt : it.prompt;
   785→          if (!prompt) continue;
   786→          bestDist = d;
   787→          best = { prompt, run: it.run };
   788→        }
   789→      }
   790→      // NPCs
   791→      for (const npc of this.npcs) {
   792→        const d = npc.position.distanceTo(p.position);
   793→        if (d < 2.6 && d < bestDist) {
   794→          bestDist = d;
   795→          best = {
   796→            prompt: `Hablar con ${npc.name}`,
   797→            run: () => {
   798→              npc.talkTo(p.position);
   799→              if (!this.dialogue.start(npc.def.dialogueId)) {
   800→                this.dialogue.startLore(npc.name, ['…']);
   801→              }
   802→              this.audio.playUi('click');
   803→            },
   804→          };
   805→        }
   806→      }
   807→    }
   808→
   809→    const promptText = best ? `[E] ${best.prompt}` : null;
   810→    if (promptText !== store.prompt) {
   811→      store.setHud({ prompt: promptText });
   812→    }
   813→    if (best && !dialogueActive && this.input.pressed('interact')) {
   814→      best.run();
   815→    }
   816→  }

   818→  private checkDiscoveries(store: ReturnType<typeof useGameStore.getState>): void {
   819→    const w = this.world!;
   820→    const p = this.player!;
   821→    for (const poi of POIS) {
   822→      if (this.state.discoveredPOIs.has(poi.id)) continue;
   823→      const d = Math.hypot(p.position.x - poi.x, p.position.z - poi.z);
   824→      if (d < poi.radius) {
   825→        this.state.discoveredPOIs.add(poi.id);
   826→        this.state.giveXp(poi.xp);
   827→        this.state.addJournal({
   828→          id: `poi_${poi.id}`,
   829→          kind: 'discovery',
   830→          title: poi.name,
   831→          text: poi.journal,
   832→        });
   833→        this.audio.playEvent('discovery');
   834→        this.musicIntensityTarget = 0.45;
   835→        store.setDiscovery({ title: poi.name, sub: `Lugar descubierto · +${poi.xp} XP` });
   836→        setTimeout(() => {
   837→          if (useGameStore.getState().discovery?.title === poi.name) {
   838→            useGameStore.getState().setDiscovery(null);
   839→          }
   840→        }, 4200);
   841→        this.markExplored(poi.x, poi.z, 6);
   842→      }
   843→    }
   844→  }

   846→  private markExplored(x: number, z: number, radiusCells: number): void {
   847→    const size = 128;
   848→    if (this.state.exploredMap.length !== size * size) {
   849→      this.state.exploredMap = new Array(size * size).fill(false);
   850→    }
   851→    const cx = Math.floor(((x + 1024) / 2048) * size);
   852→    const cz = Math.floor(((z + 1024) / 2048) * size);
   853→    for (let dz = -radiusCells; dz <= radiusCells; dz++) {
   854→      for (let dx = -radiusCells; dx <= radiusCells; dx++) {
   855→        const ix = cx + dx;
   856→        const iz = cz + dz;
   857→        if (ix < 0 || iz < 0 || ix >= size || iz >= size) continue;
   858→        this.state.exploredMap[iz * size + ix] = true;
   859→      }
   860→    }
   861→  }
