   484→   484→  private update(dt: number): void {
   485→   485→    if (!this.world || !this.player || !this.rig) return;
   486→   486→    const store = useGameStore.getState();
   487→   487→    const w = this.world;
   488→   488→    const p = this.player;
   489→   489→    windTime.value += dt;
   490→   490→
   491→   491→    switch (store.phase) {
   492→   492→      case 'menu': {
   493→   493→        this.menuOrbit += dt * 0.021;
   494→   494→        const r = 150;
   495→   495→        const cx = 20 + Math.cos(this.menuOrbit) * r;
   496→   496→        const cz = 200 + Math.sin(this.menuOrbit) * r;
   497→   497→        const cy = w.terrainHeight.height(cx, cz) + 42;
   498→   498→        this.rig.setManual(new THREE.Vector3(cx, cy, cz), new THREE.Vector3(20, 12, 170));
   499→   499→        w.sky.dayFraction = 0.78;
   500→   500→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   501→   501→        w.water.update(dt, this.engine.camera);
   502→   502→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   503→   503→        w.gulls.update(dt);
   504→   504→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   505→   505→        w.terrain.update(cx, cz);
   506→   506→        w.vegetation.update();
   507→   507→        break;
   508→   508→      }
   509→   509→      case 'intro': {
   510→   510→        this.updateIntro(dt);
   511→   511→        w.sky.tick(dt);
   512→   512→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   513→   513→        w.water.update(dt, this.engine.camera);
   514→   514→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   515→   515→        w.gulls.update(dt);
   516→   516→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   517→   517→        break;
   518→   518→      }
   519→   519→      case 'playing': {
   520→   520→        // Toggle overlays (works whether or not one is already open)
   521→   521→        const toggle = this.input.menuTogglePressed();
   522→   522→        if (toggle && !store.dialogue) {
   523→   523→          const map: Record<string, 'inventory' | 'map' | 'journal' | 'skills'> = {
   524→   524→            inventory: 'inventory', map: 'map', journal: 'journal', skills: 'skills',
   525→   525→          };
   526→   526→          const target = map[toggle];
   527→   527→          if (store.overlay === target) {
   528→   528→            store.closeOverlay();
   529→   529→            this.input.requestLock();
   530→   530→          } else {
   531→   531→            store.openOverlay(target);
   532→   532→            this.input.releaseLock();
   533→   533→          }
   534→   534→        }
   535→   535→        if (!store.overlay) {
   536→   536→          this.updatePlaying(dt, store);
   537→   537→        } else {
   538→   538→          // Overlays pause input but keep rendering a frozen frame
   539→   539→          w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   540→   540→        }
   541→   541→        break;
   542→   542→      }
   543→   543→      case 'paused': {
   544→   544→        // frozen — nothing updates
   545→   545→        break;
   546→   546→      }
   547→   547→      default:
   548→   548→        break;
   549→   549→    }
   550→   550→    this.input.endFrame();
   551→   551→  }

   553→   553→  private updateIntro(dt: number): void {
   554→   554→    const store = useGameStore.getState();
   555→   555→    // Skip
   556→   556→    if (this.input.anyPressed('Space', 'Escape', 'Enter', 'KeyE')) {
   557→   557→      this.finishIntro();
   558→   558→      return;
   559→   559→    }
   560→   560→    this.introT += dt;
   561→   561→    let shot = 0;
   562→   562→    let t0 = 0;
   563→   563→    for (let i = 0; i < INTRO_SHOTS.length; i++) {
   564→   564→      if (this.introT < t0 + INTRO_SHOTS[i].dur) {
   565→   565→        shot = i;
   566→   566→        break;
   567→   567→      }
   568→   568→      t0 += INTRO_SHOTS[i].dur;
   569→   569→      shot = i;
   570→   570→    }
   571→   571→    const totalDur = INTRO_SHOTS.reduce((a, s) => a + s.dur, 0);
   572→   572→    if (this.introT >= totalDur) {
   573→   573→      this.finishIntro();
   574→   574→      return;
   575→   575→    }
   576→   576→    const s = INTRO_SHOTS[shot];
   577→   577→    const k = smoothstep(0, 1, clamp((this.introT - t0) / s.dur, 0, 1));
   578→   578→    const pos = s.from.clone().lerp(s.to, k);
   579→   579→    const look = s.lookFrom.clone().lerp(s.lookTo, k);
   580→   580→    this.rig!.setManual(pos, look);
   581→   581→    // Only push to the store when the shot changes (avoids per-frame renders)
   582→   582→    if (this.lastIntroShot !== shot) {
   583→   583→      this.lastIntroShot = shot;
   584→   584→      store.setIntro({
   585→   585→        text: s.text,
   586→   586→        shot,
   587→   587→        total: INTRO_SHOTS.length,
   588→   588→        title: shot === INTRO_SHOTS.length - 1,
   589→   589→      });
   590→   590→    }
   591→   591→  }

   593→   593→  private updatePlaying(dt: number, store: ReturnType<typeof useGameStore.getState>): void {
   594→   594→    const w = this.world!;
   595→   595→    const p = this.player!;
   596→   596→    const rig = this.rig!;
   597→   597→    const dialogueActive = !!store.dialogue;
   598→   598→
   599→   599→    // --- Death / respawn ---
   600→   600→    if (p.health <= 0 && !this.dead) {
   601→   601→      this.dead = true;
   602→   602→      const sp = w.village.spawnPoint;
   603→   603→      p.teleport(sp.x, 1.5, sp.z);
   604→   604→      p.health = Math.floor(p.stats.maxHealth * 0.5);
   605→   605→      store.notify('info', 'Las olas te devuelven a la orilla', 'Despiertas en el muelle, empapado pero vivo.');
   606→   606→      this.audio.playEvent('splash');
   607→   607→    }
   608→   608→
   609→   609→    // --- Clock ---
   610→   610→    const newDay = w.sky.tick(dt);
   611→   611→    if (newDay) {
   612→   612→      store.notify('info', `Día ${w.sky.dayCount} en Aldmor`, formatClock(w.sky.dayFraction));
   613→   613→    }
   614→   614→
   615→   615→    // --- Camera input & player ---
   616→   616→    rig.handleInput(this.input, dt);
   617→   617→    p.update(dt, this.input, rig.yawForward, w.terrainHeight, this.colliders, dialogueActive || this.dead);
   618→   618→
   619→   619→    // --- Attack (LMB) ---
   620→   620→    if (!dialogueActive && this.input.leftPressed && p.attackT === 0 && this.state.equippedWeapon) {
   621→   621→      p.attackT = 0.001;
   622→   622→      p.onAttackSwing?.();
   623→   623→      this.lastSwingHit = false;
   624→   624→    }
   625→   625→    if (p.attackT > 0 && !this.lastSwingHit && p.attackT >= 0.45) {
   626→   626→      this.lastSwingHit = true;
   627→   627→      // Training dummy hit check
   628→   628→      const d = w.dummyPos;
   629→   629→      const dx = d.x - p.position.x;
   630→   630→      const dz = d.z - p.position.z;
   631→   631→      const dist = Math.hypot(dx, dz);
   632→   632→      if (dist < 2.3) {
   633→   633→        const facing = Math.sin(p.yaw) * dx + Math.cos(p.yaw) * dz;
   634→   634→        if (facing > 0) {
   635→   635→          this.audio.playEvent('damage');
   636→   636→          this.state.giveXp(2);
   637→   637→        }
   638→   638→      }
   639→   639→    }
   640→   640→
   641→   641→    // --- Lantern ---
   642→   642→    if (!dialogueActive && this.input.pressed('lantern')) {
   643→   643→      p.toggleLantern();
   644→   644→      this.audio.playEvent('pickup');
   645→   645→    }
   646→   646→
   647→   647→    // --- World systems ---
   648→   648→    rig.update(dt, p.position, this.input.isDown('sprint') && !dialogueActive, w.terrainHeight, this.colliders);
   649→   649→    w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   650→   650→    w.water.update(dt, this.engine.camera);
   651→   651→    w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   652→   652→    w.terrain.update(p.position.x, p.position.z);
   653→   653→    w.vegetation.update();
   654→   654→    this.refreshGrass(p.position.x, p.position.z);
   655→   655→    w.gulls.update(dt);
   656→   656→    w.village.torches.update(dt, w.sky.nightFactor, p.position);
   657→   657→    updateChests(dt, w.chests.map((c) => c.handles));
   658→   658→
   659→   659→    // Waterfall shimmer
   660→   660→    const sheetMat = w.waterfallSheet.material as THREE.MeshStandardMaterial;
   661→   661→    sheetMat.opacity = 0.48 + Math.sin(windTime.value * 2.2) * 0.08;
   662→   662→    const foamMat = w.waterfallFoam.material as THREE.MeshBasicMaterial;
   663→   663→    foamMat.opacity = 0.42 + Math.sin(windTime.value * 3.1) * 0.12;
   664→   664→
   665→   665→    // NPC updates
   666→   666→    for (const npc of this.npcs) {
   667→   667→      npc.update(dt, w.sky.dayFraction, w.terrainHeight, this.colliders, p.position);
   668→   668→    }
   669→   669→
   670→   670→    // --- Interaction scan ---
   671→   671→    this.updateInteractions(store, dialogueActive);
   672→   672→
   673→   673→    // --- Discoveries ---
   674→   674→    this.discoveryTimer -= dt;
   675→   675→    if (this.discoveryTimer <= 0) {
   676→   676→      this.discoveryTimer = 0.5;
   677→   677→      this.checkDiscoveries(store);
   678→   678→    }
   679→   679→
   680→   680→    // --- Herb glow pulse ---
   681→   681→    for (const herb of w.herbs) {
   682→   682→      if (herb.mesh.visible) {
   683→   683→        const petals = herb.mesh.children[herb.mesh.children.length - 1] as THREE.Mesh;
   684→   684→        const m = petals.material as THREE.MeshStandardMaterial;
   685→   685→        m.emissiveIntensity = 0.55 + Math.sin(windTime.value * 2 + herb.pos.x) * 0.25;
   686→   686→      }
   687→   687→    }
   688→   688→
   689→   689→    // --- Audio ambience ---
   690→   690→    const terrainHNow = w.terrainHeight.height(p.position.x, p.position.z);
   691→   691→    this.audio.update(dt, {
   692→   692→      altitude: Math.max(0, p.position.y),
   693→   693→      shoreDist: clamp(terrainHNow, 0, 160),
   694→   694→      dayFactor: w.sky.dayFactor,
   695→   695→      waterfallDist: p.position.distanceTo(w.waterfallBasin),
   696→   696→    });
   697→   697→
   698→   698→    // --- Heart of Oak slow regeneration ---
   699→   699→    if (this.state.learnedSkills.has('sup_corazon') && p.health < p.stats.maxHealth) {
   700→   700→      p.health = Math.min(p.stats.maxHealth, p.health + 0.6 * dt);
   701→   701→    }
   702→   702→
   703→   703→    // --- Music intensity decay ---
   704→   704→    this.audio.musicIntensity += (this.musicIntensityTarget - this.audio.musicIntensity) * dt * 0.3;
   705→   705→    this.musicIntensityTarget += (0.15 - this.musicIntensityTarget) * dt * 0.05;
   706→   706→
   707→   707→    // --- HUD sync ---
   708→   708→    this.hudTimer -= dt;
   709→   709→    if (this.hudTimer <= 0) {
   710→   710→      this.hudTimer = 0.15;
   711→   711→      (window as unknown as { __kotfsYaw?: number }).__kotfsYaw = rig.yaw;
   712→   712→      (window as unknown as { __kotfsPlayer?: { x: number; y: number; z: number } }).__kotfsPlayer = {
   713→   713→        x: p.position.x, y: p.position.y, z: p.position.z,
   714→   714→      };
   715→   715→      const tracker = this.state.currentTracker();
   716→   716→      const clickHint =
   717→   717→        !this.input.pointerLocked && !dialogueActive
   718→   718→          ? 'Haz clic para tomar el control'
   719→   719→          : null;
   720→   720→      store.setHud({
   721→   721→        health: p.health,
   722→   722→        maxHealth: p.stats.maxHealth,
   723→   723→        stamina: p.stamina,
   724→   724→        maxStamina: p.stats.maxStamina,
   725→   725→        clock: formatClock(w.sky.dayFraction),
   726→   726→        dayCount: w.sky.dayCount,
   727→   727→        questTracker: tracker,
   728→   728→        fps: this.engine.fps,
   729→   729→        prompt: clickHint ?? store.prompt,
   730→   730→      });
   731→   731→      this.pollQuestEvents();
   732→   732→    }
   733→   733→
   734→   734→    // --- Map exploration ---
   735→   735→    this.mapTimer -= dt;
   736→   736→    if (this.mapTimer <= 0) {
   737→   737→      this.mapTimer = 2;
   738→   738→      this.markExplored(p.position.x, p.position.z, 3);
   739→   739→    }
   740→   740→
   741→   741→    // --- Playtime & autosave ---
   742→   742→    this.state.playtimeSeconds += dt;
   743→   743→    this.autosaveTimer -= dt;
   744→   744→    if (this.autosaveTimer <= 0) {
   745→   745→      this.autosaveTimer = 120;
   746→   746→      this.autosave();
   747→   747→    }
   748→   748→
   749→   749→    // --- Tutorials ---
   750→   750→    if (!this.state.hasFlag('tut_night') && w.sky.dayFactor < 0.2) {
   751→   751→      this.state.setFlag('tut_night');
   752→   752→      store.notify('info', 'Cae la noche', 'Pulsa F para encender tu farol. Las antorchas del pueblo iluminan los caminos.');
   753→   753→    }
   754→   754→    if (!this.state.hasFlag('tut_stamina') && p.stamina < 25) {
   755→   755→      this.state.setFlag('tut_stamina');
   756→   756→      store.notify('info', 'Resistencia baja', 'Deja de correr para recuperarla. Mejórala con habilidades (K).');
   757→   757→    }
   758→   758→  }

   773→   773→  private updateInteractions(store: ReturnType<typeof useGameStore.getState>, dialogueActive: boolean): void {
   774→   774→    const w = this.world!;
   775→   775→    const p = this.player!;
   776→   776→    let best: { prompt: string; run: () => void } | null = null;
   777→   777→    let bestDist = Infinity;
   778→   778→
   779→   779→    if (!dialogueActive) {
   780→   780→      for (const it of w.interactables) {
   781→   781→        if (it.available && !it.available()) continue;
   782→   782→        const d = it.pos.distanceTo(p.position);
   783→   783→        if (d < it.radius && d < bestDist) {
   784→   784→          const prompt = typeof it.prompt === 'string' ? it.prompt : it.prompt;
   785→   785→          if (!prompt) continue;
   786→   786→          bestDist = d;
   787→   787→          best = { prompt, run: it.run };
   788→   788→        }
   789→   789→      }
   790→   790→      // NPCs
   791→   791→      for (const npc of this.npcs) {
   792→   792→        const d = npc.position.distanceTo(p.position);
   793→   793→        if (d < 2.6 && d < bestDist) {
   794→   794→          bestDist = d;
   795→   795→          best = {
   796→   796→            prompt: `Hablar con ${npc.name}`,
   797→   797→            run: () => {
   798→   798→              npc.talkTo(p.position);
   799→   799→              if (!this.dialogue.start(npc.def.dialogueId)) {
   800→   800→                this.dialogue.startLore(npc.name, ['…']);
   801→   801→              }
   802→   802→              this.audio.playUi('click');
   803→   803→            },
   804→   804→          };
   805→   805→        }
   806→   806→      }
   807→   807→    }
   808→   808→
   809→   809→    const promptText = best ? `[E] ${best.prompt}` : null;
   810→   810→    if (promptText !== store.prompt) {
   811→   811→      store.setHud({ prompt: promptText });
   812→   812→    }
   813→   813→    if (best && !dialogueActive && this.input.pressed('interact')) {
   814→   814→      best.run();
   815→   815→    }
   816→   816→  }

   818→   818→  private checkDiscoveries(store: ReturnType<typeof useGameStore.getState>): void {
   819→   819→    const w = this.world!;
   820→   820→    const p = this.player!;
   821→   821→    for (const poi of POIS) {
   822→   822→      if (this.state.discoveredPOIs.has(poi.id)) continue;
   823→   823→      const d = Math.hypot(p.position.x - poi.x, p.position.z - poi.z);
   824→   824→      if (d < poi.radius) {
   825→   825→        this.state.discoveredPOIs.add(poi.id);
   826→   826→        this.state.giveXp(poi.xp);
   827→   827→        this.state.addJournal({
   828→   828→          id: `poi_${poi.id}`,
   829→   829→          kind: 'discovery',
   830→   830→          title: poi.name,
   831→   831→          text: poi.journal,
   832→   832→        });
   833→   833→        this.audio.playEvent('discovery');
   834→   834→        this.musicIntensityTarget = 0.45;
   835→   835→        store.setDiscovery({ title: poi.name, sub: `Lugar descubierto · +${poi.xp} XP` });
   836→   836→        setTimeout(() => {
   837→   837→          if (useGameStore.getState().discovery?.title === poi.name) {
   838→   838→            useGameStore.getState().setDiscovery(null);
   839→   839→          }
   840→   840→        }, 4200);
   841→   841→        this.markExplored(poi.x, poi.z, 6);
   842→   842→      }
   843→   843→    }

   846→   846→  private markExplored(x: number, z: number, radiusCells: number): void {
   847→   847→    const size = 128;
   848→   848→    if (this.state.exploredMap.length !== size * size) {
   849→   849→      this.state.exploredMap = new Array(size * size).fill(false);
   850→   850→    }
   851→   851→    const cx = Math.floor(((x + 1024) / 2048) * size);
   852→   852→    const cz = Math.floor(((z + 1024) / 2048) * size);
   853→   853→    for (let dz = -radiusCells; dz <= radiusCells; dz++) {
   854→   854→      for (let dx = -radiusCells; dx <= radiusCells; dx++) {
   855→   855→        const ix = cx + dx;
   856→   856→        const iz = cz + dz;
   857→   857→        if (ix < 0 || iz < 0 || ix >= size || iz >= size) continue;
   858→   858→        this.state.exploredMap[iz * size + ix] = true;
   859→   859→      }
   860→   860→    }
   861→   861→  }
