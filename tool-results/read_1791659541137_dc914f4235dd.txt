   685→  loadFromSlot(slot: number): boolean {
   686→    const data = SaveSystem.load(slot);
   687→    if (!data || !this.world || !this.player || !this.naval) return false;
   688→    this.state.deserialize(data.gameState);
   689→    this.player.deserialize(data.player);
   690→    this.world.sky.deserialize(data.time);
   691→    this.dead = false;
   692→
   693→    // Naval restoration: only when the Colibrí was already repaired
   694→    if (this.state.hasFlag('ship_repaired')) {
   695→      this.world.village.setStaticShip(false);
   696→      if (data.naval) {
   697→        this.naval.deserialize(data.naval, true);
   698→      } else {
   699→        // v1 save that somehow has the flag — spawn at dock
   700→        const dock = this.world.dockSpawn;
   701→        this.naval.spawnPlayerShip(dock.x, dock.z, dock.heading);

   705→      this.world.village.setStaticShip(true);
   706→      this.world.streamer.register(this.world.village.staticShip, this.world.village.shipPos.x, this.world.village.shipPos.z, 30);
   707→      this.naval.resetAll();
   708→      if (data.naval) this.naval.deserialize(data.naval, false);
   709→    }
   710→
   711→    // Sailing state after load: resume at the helm or on foot
   712→    this.player.platformProbe = undefined;
   713→    this.rig!.setShipMode(null);
   714→    if (this.naval.playerSailing && this.naval.playerShip) {
   715→      this.naval.playerBoardsHelm();
   716→      this.attachPlayerToWheel();
   717→      this.rig!.setShipMode(this.naval.playerShip);

   719→

   722→    this.syncProgression();
   723→    // Fase 3: align dragons & weapon visuals with the loaded save
   724→    this.dragons?.syncWithState();
   725→    this.combat?.refreshVisuals();
   726→    this.audio.bossIntensity = 0;
   727→    const store = useGameStore.getState();
   728→    store.setPhase('playing');
   729→    this.input.requestLock();
   730→    store.notify('save', 'Partida cargada', `Día ${data.time.dayCount} · Nivel ${data.gameState.level}`);
   731→    return true;

   785→  private update(dt: number): void {
   786→    if (!this.world || !this.player || !this.rig) return;
   787→    const store = useGameStore.getState();
   788→    const w = this.world;
   789→    const p = this.player;
   790→    windTime.value += dt;
   791→
   792→    switch (store.phase) {
   793→      case 'menu': {
   794→        this.menuOrbit += dt * 0.021;
   795→        const r = 150;
   796→        const cx = 20 + Math.cos(this.menuOrbit) * r;
   797→        const cz = 200 + Math.sin(this.menuOrbit) * r;
   798→        const cy = w.terrainHeight.height(cx, cz) + 42;
   799→        this.rig.setManual(new THREE.Vector3(cx, cy, cz), new THREE.Vector3(20, 12, 170));
   800→        w.sky.dayFraction = 0.78;
   801→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   802→        w.water.update(dt, this.engine.camera);
   803→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   804→        w.gulls.update(dt);
   805→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   806→        w.terrain.update(cx, cz);
   807→        w.vegetation.update();
   808→        break;
   809→      }
   810→      case 'intro': {
   811→        this.updateIntro(dt);
   812→        w.sky.tick(dt);
   813→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   814→        w.water.update(dt, this.engine.camera);
   815→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   816→        w.gulls.update(dt);
   817→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   818→        break;
   819→      }
   820→      case 'playing': {
   821→        // Toggle overlays (works whether or not one is already open)
   822→        const toggle = this.input.menuTogglePressed();
   823→        if (toggle && !store.dialogue) {
   824→          const map: Record<string, 'inventory' | 'map' | 'journal' | 'skills'> = {
   825→            inventory: 'inventory', map: 'map', journal: 'journal', skills: 'skills',
   826→          };
   827→          const target = map[toggle];
   828→          if (store.overlay === target) {
   829→            store.closeOverlay();
   830→            this.input.requestLock();
   831→          } else {
   832→            store.openOverlay(target);
   833→            this.input.releaseLock();
   834→          }
   835→        }
   836→        if (!store.overlay) {
   837→          this.updatePlaying(dt, store);
   838→        } else {
   839→          // Overlays pause input but keep rendering a frozen frame
   840→          w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   841→        }
   842→        break;
   843→      }
   844→      case 'paused': {
   845→        // frozen — nothing updates
   846→        break;
   847→      }
   848→      default:
   849→        break;
   850→    }
   851→    this.input.endFrame();
   852→  }

   854→  private updateIntro(dt: number): void {
   855→    const store = useGameStore.getState();
   856→    // Skip
   857→    if (this.input.anyPressed('Space', 'Escape', 'Enter', 'KeyE')) {
   858→      this.finishIntro();
   859→      return;
   860→    }
   861→    this.introT += dt;
   862→    let shot = 0;
   863→    let t0 = 0;
   864→    for (let i = 0; i < INTRO_SHOTS.length; i++) {
   865→      if (this.introT < t0 + INTRO_SHOTS[i].dur) {
   866→        shot = i;
   867→        break;
   868→      }
   869→      t0 += INTRO_SHOTS[i].dur;
   870→      shot = i;
   871→    }
   872→    const totalDur = INTRO_SHOTS.reduce((a, s) => a + s.dur, 0);
   873→    if (this.introT >= totalDur) {
   874→      this.finishIntro();
   875→      return;
   876→    }
   877→    const s = INTRO_SHOTS[shot];
   878→    const k = smoothstep(0, 1, clamp((this.introT - t0) / s.dur, 0, 1));
   879→    const pos = s.from.clone().lerp(s.to, k);
   880→    const look = s.lookFrom.clone().lerp(s.lookTo, k);
   881→    this.rig!.setManual(pos, look);
   882→    // Only push to the store when the shot changes (avoids per-frame renders)
   883→    if (this.lastIntroShot !== shot) {
   884→      this.lastIntroShot = shot;
   885→      store.setIntro({
   886→        text: s.text,
   887→        shot,
   888→        total: INTRO_SHOTS.length,
   889→        title: shot === INTRO_SHOTS.length - 1,
   890→      });
   891→    }
   892→  }

   907→  private updatePlaying(dt: number, store: ReturnType<typeof useGameStore.getState>): void {
   908→    const w = this.world!;
   909→    const p = this.player!;
   910→    const rig = this.rig!;
   911→    const naval = this.naval!;
   912→    const dialogueActive = !!store.dialogue;
   913→    const sailing = naval.playerSailing && !!naval.playerShip;
   914→
   915→    // --- Naval layer: wind, fleets, cannonballs, boarding, effects ---
   916→    const sailingHud: SailingHud | null = naval.update(dt, p.position);
   917→
   918→    // --- Fase 3: dragons + combat VFX (both on foot and at sea) ---
   919→    this.dragons?.update(
   920→      dt, p.position, !this.dead, sailing,
   921→      sailing && naval.playerShip ? naval.playerShip.pos : null,
   922→    );
   923→    this.vfx?.update(dt);
   924→
   925→    // --- Death / respawn ---
   926→    if (p.health <= 0 && !this.dead) {
   927→      this.dead = true;
   928→      // Dragons lose interest and return to their lairs at full strength
   929→      this.dragons?.resetCombat();
   930→      this.audio.bossIntensity = 0;
   931→      if (naval.boardingActive) {
   932→        naval.abortBoarding();
   933→        p.platformProbe = undefined;
   934→        rig.setShipMode(null);
   935→      }
   936→      const sp = w.village.spawnPoint;
   937→      p.teleport(sp.x, 1.5, sp.z);
   938→      p.health = Math.floor(p.stats.maxHealth * 0.5);
   939→      store.notify('info', 'Las olas te devuelven a la orilla', 'Despiertas en el muelle, empapado pero vivo.');
   940→      this.audio.playEvent('splash');
   941→    }
   942→
   943→    // --- Clock ---
   944→    const newDay = w.sky.tick(dt);
   945→    if (newDay) {
   946→      store.notify('info', `Día ${w.sky.dayCount} en Aldmor`, formatClock(w.sky.dayFraction));
   947→    }
   948→
   949→    if (sailing && !naval.boardingActive) {
   950→      // ================= SAILING MODE — the player holds the wheel =================

  1174→  private updateInteractions(store: ReturnType<typeof useGameStore.getState>, dialogueActive: boolean): void {
  1175→    const w = this.world!;
  1176→    const p = this.player!;
  1177→    let best: { prompt: string; run: () => void } | null = null;
  1178→    let bestDist = Infinity;
  1179→
  1180→    if (!dialogueActive) {
  1181→      // Naval prompts have priority while aboard / at sea
  1182→      const navalPrompt = this.naval?.currentPrompt(p.position) ?? null;
  1183→      if (navalPrompt) {
  1184→        best = { prompt: navalPrompt.prompt, run: navalPrompt.run };
  1185→        bestDist = 0;
  1186→      } else if (!this.naval?.playerSailing) {
  1187→        for (const it of w.interactables) {
  1188→          if (it.available && !it.available()) continue;
  1189→          const d = it.pos.distanceTo(p.position);
  1190→          if (d < it.radius && d < bestDist) {
  1191→            const prompt = typeof it.prompt === 'string' ? it.prompt : it.prompt;
  1192→            if (!prompt) continue;
  1193→            bestDist = d;
  1194→            best = { prompt, run: it.run };
  1195→          }
  1196→        }
  1197→        // NPCs
  1198→        for (const npc of this.npcs) {
  1199→          const d = npc.position.distanceTo(p.position);
  1200→          if (d < 2.6 && d < bestDist) {
  1201→            bestDist = d;
  1202→            best = {
  1203→              prompt: `Hablar con ${npc.name}`,
  1204→              run: () => {
  1205→                npc.talkTo(p.position);
  1206→                if (!this.dialogue.start(npc.def.dialogueId)) {
  1207→                  this.dialogue.startLore(npc.name, ['…']);
  1208→                }
  1209→                this.audio.playUi('click');
  1210→              },
  1211→            };
  1212→          }
  1213→        }
  1214→      }
  1215→    }
  1216→
  1217→    const promptText = best ? `[E] ${best.prompt}` : null;
  1218→    if (promptText !== store.prompt) {
  1219→      store.setHud({ prompt: promptText });
  1220→    }
  1221→    if (best && !dialogueActive && this.input.pressed('interact')) {
  1222→      best.run();
  1223→    }
  1224→  }

  1254→  private markExplored(x: number, z: number, radiusCells: number): void {
  1255→    const size = 128;
  1256→    if (this.state.exploredMap.length !== size * size) {
  1257→      this.state.exploredMap = new Array(size * size).fill(false);
  1258→    }
  1259→    const cx = Math.floor(((x + 1024) / 2048) * size);
  1260→    const cz = Math.floor(((z + 1024) / 2048) * size);
  1261→    for (let dz = -radiusCells; dz <= radiusCells; dz++) {
  1262→      for (let dx = -radiusCells; dx <= radiusCells; dx++) {
  1263→        const ix = cx + dx;
  1264→        const iz = cz + dz;
  1265→        if (ix < 0 || iz < 0 || ix >= size || iz >= size) continue;
  1266→        this.state.exploredMap[iz * size + ix] = true;
  1267→      }
  1268→    }
  1269→  }
