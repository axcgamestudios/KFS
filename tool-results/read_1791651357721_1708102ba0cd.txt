   692→   692→  private update(dt: number): void {
   693→   693→    if (!this.world || !this.player || !this.rig) return;
   694→   694→    const store = useGameStore.getState();
   695→   695→    const w = this.world;
   696→   696→    const p = this.player;
   697→   697→    windTime.value += dt;
   698→   698→
   699→   699→    switch (store.phase) {
   700→   700→      case 'menu': {
   701→   701→        this.menuOrbit += dt * 0.021;
   702→   702→        const r = 150;
   703→   703→        const cx = 20 + Math.cos(this.menuOrbit) * r;
   704→   704→        const cz = 200 + Math.sin(this.menuOrbit) * r;
   705→   705→        const cy = w.terrainHeight.height(cx, cz) + 42;
   706→   706→        this.rig.setManual(new THREE.Vector3(cx, cy, cz), new THREE.Vector3(20, 12, 170));
   707→   707→        w.sky.dayFraction = 0.78;
   708→   708→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   709→   709→        w.water.update(dt, this.engine.camera);
   710→   710→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   711→   711→        w.gulls.update(dt);
   712→   712→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   713→   713→        w.terrain.update(cx, cz);
   714→   714→        w.vegetation.update();
   715→   715→        break;
   716→   716→      }
   717→   717→      case 'intro': {
   718→   718→        this.updateIntro(dt);
   719→   719→        w.sky.tick(dt);
   720→   720→        w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   721→   721→        w.water.update(dt, this.engine.camera);
   722→   722→        w.water.setSky(w.sky.sunDir, w.sky.sunColor, w.sky.zenithColor, w.sky.horizonColor, w.sky.nightFactor);
   723→   723→        w.gulls.update(dt);
   724→   724→        w.village.torches.update(dt, w.sky.nightFactor, this.engine.camera.position);
   725→   725→        break;
   726→   726→      }
   727→   727→      case 'playing': {
   728→   728→        // Toggle overlays (works whether or not one is already open)
   729→   729→        const toggle = this.input.menuTogglePressed();
   730→   730→        if (toggle && !store.dialogue) {
   731→   731→          const map: Record<string, 'inventory' | 'map' | 'journal' | 'skills'> = {
   732→   732→            inventory: 'inventory', map: 'map', journal: 'journal', skills: 'skills',
   733→   733→          };
   734→   734→          const target = map[toggle];
   735→   735→          if (store.overlay === target) {
   736→   736→            store.closeOverlay();
   737→   737→            this.input.requestLock();
   738→   738→          } else {
   739→   739→            store.openOverlay(target);
   740→   740→            this.input.releaseLock();
   741→   741→          }
   742→   742→        }
   743→   743→        if (!store.overlay) {
   744→   744→          this.updatePlaying(dt, store);
   745→   745→        } else {
   746→   746→          // Overlays pause input but keep rendering a frozen frame
   747→   747→          w.sky.update(this.engine.camera, p.position, this.engine.quality.shadowsEnabled);
   748→   748→        }
   749→   749→        break;
   750→   750→      }
   751→   751→      case 'paused': {
   752→   752→        // frozen — nothing updates
   753→   753→        break;
   754→   754→      }
   755→   755→      default:
   756→   756→        break;
   757→   757→    }
   758→   758→    this.input.endFrame();
   759→   759→  }

   761→   761→  private updateIntro(dt: number): void {
   762→   762→    const store = useGameStore.getState();
   763→   763→    // Skip
   764→   764→    if (this.input.anyPressed('Space', 'Escape', 'Enter', 'KeyE')) {
   765→   765→      this.finishIntro();
   766→   766→      return;
   767→   767→    }
   768→   768→    this.introT += dt;
   769→   769→    let shot = 0;
   770→   770→    let t0 = 0;
   771→   771→    for (let i = 0; i < INTRO_SHOTS.length; i++) {
   772→   772→      if (this.introT < t0 + INTRO_SHOTS[i].dur) {
   773→   773→        shot = i;
   774→   774→        break;
   775→   775→      }
   776→   776→      t0 += INTRO_SHOTS[i].dur;
   777→   777→      shot = i;
   778→   778→    }
   779→   779→    const totalDur = INTRO_SHOTS.reduce((a, s) => a + s.dur, 0);
   780→   780→    if (this.introT >= totalDur) {
   781→   781→      this.finishIntro();
   782→   782→      return;
   783→   783→    }
   784→   784→    const s = INTRO_SHOTS[shot];
   785→   785→    const k = smoothstep(0, 1, clamp((this.introT - t0) / s.dur, 0, 1));
   786→   786→    const pos = s.from.clone().lerp(s.to, k);
   787→   787→    const look = s.lookFrom.clone().lerp(s.lookTo, k);
   788→   788→    this.rig!.setManual(pos, look);
   789→   789→    // Only push to the store when the shot changes (avoids per-frame renders)
   790→   790→    if (this.lastIntroShot !== shot) {
   791→   791→      this.lastIntroShot = shot;
   792→   792→      store.setIntro({
   793→   793→        text: s.text,
   794→   794→        shot,
   795→   795→        total: INTRO_SHOTS.length,
   796→   796→        title: shot === INTRO_SHOTS.length - 1,
   797→   797→      });
   798→   798→    }
   799→   799→  }

   801→   801→  private updatePlaying(dt: number, store: ReturnType<typeof useGameStore.getState>): void {
   802→   802→    const w = this.world!;
   803→   803→    const p = this.player!;
   804→   804→    const rig = this.rig!;
   805→   805→    const naval = this.naval!;
   806→   806→    const dialogueActive = !!store.dialogue;
   807→   807→    const sailing = naval.playerSailing && !!naval.playerShip;
   808→   808→
   809→   809→    // --- Naval layer: wind, fleets, cannonballs, boarding, effects ---
   810→   810→    const sailingHud: SailingHud | null = naval.update(dt, p.position);
   811→   811→
   812→   812→    // --- Death / respawn ---
   813→   813→    if (p.health <= 0 && !this.dead) {
   814→   814→      this.dead = true;
   815→   815→      if (naval.boardingActive) {
   816→   816→        naval.abortBoarding();
   817→   817→        p.platformProbe = undefined;
