    44→/** One boarding pirate (walks the enemy deck in ship-local coordinates). */
    45→class BoardingCrew {
    46→  readonly model: CharacterModel;
    47→  readonly group = new THREE.Group();
    48→  hp: number;
    49→  readonly maxHp: number;
    50→  readonly isCaptain: boolean;
    51→  readonly name: string;
    52→  lx: number;           // ship-local X
    53→  lz: number;           // ship-local Z
    54→  yaw = 0;
    55→  attackT = 0;
    56→  attackCd = 0;
    57→  dead = false;
    58→  deathT = 0;
    59→  /** Fase 9 — stagger timer: > 0 means reeling (no move, no attacks). */
    60→  hurtT = 0;
    61→  /** Fase 9 — approach bearing around the player (radians, ship-local). */
    62→  ringAngle = 0;

    64→  constructor(captain: boolean, lx: number, lz: number, palette: CharacterPalette) {
    65→    this.isCaptain = captain;
    66→    this.name = captain ? 'Capitana Morwen' : 'Corsario del Cuervo';
    67→    this.hp = captain ? 110 : 46;
    68→    this.maxHp = this.hp;
    69→    this.lx = lx;
    70→    this.lz = lz;
    71→    this.model = new CharacterModel(palette);
    72→    this.model.setEquippedState(true);
    73→    this.group.add(this.model.root);
    74→  }
    75→
    76→  updateAnim(dt: number): void {
    77→    if (this.dead) {
    78→      this.deathT += dt;
    79→      const k = Math.min(1, this.deathT / 0.7);
    80→      this.group.rotation.x = -k * Math.PI / 2;
    81→      this.group.position.y -= dt * 0.25 * Math.min(1, this.deathT);
    82→      if (this.deathT > 4) this.group.visible = false;
    83→      return;
    84→    }
    85→    this.model.update({
    86→      dt,
    87→      speed: this.moving ? (this.isCaptain ? 1.5 : 1.8) : 0,
    88→      moving: this.moving,
    89→      grounded: true,
    90→      swimming: false,
    91→      attackT: this.attackT,
    92→    });
    93→  }
    94→
    95→  moving = false;
    96→
    97→  get hpFrac(): number {
    98→    return this.hp / this.maxHp;
    99→  }

   248→  spawnAdminShip(x: number, z: number): THREE.Vector3 {
   249→    // find the nearest deep-water spot (ring scan)
   250→    let bestX = x, bestZ = z, bestD = Infinity;
   251→    for (let r = 6; r <= 90; r += 6) {
   252→      for (let a = 0; a < 12; a++) {
   253→        const ang = (a / 12) * Math.PI * 2;
   254→        const sx = x + Math.cos(ang) * r;
   255→        const sz = z + Math.sin(ang) * r;
   256→        if (this.terrain.height(sx, sz) < -2.2) {
   257→          const d = Math.hypot(sx - x, sz - z);
   258→          if (d < bestD) { bestD = d; bestX = sx; bestZ = sz; }
   259→        }
   260→      }
   261→      if (bestD < Infinity) break;
   262→    }

   264→      id: `admin_${Date.now() % 100000}`,
   265→      name: 'Navío Errante', kind: 'pirate', isPlayer: false,
   266→      x: bestX, z: bestZ, heading: Math.random() * Math.PI * 2,

   271→    this.extraShips.push(ship);
   272→    this.dressShip(ship);
   273→    return new THREE.Vector3(bestX, 0, bestZ);
   274→  }

   342→  /** Fase 11 — nearest sailable ship (the Colibrí always; admin ships only
   343→   *  while at anchor). Enemy AI ships are NEVER candidates. */
   344→  helmCandidate(playerWorld: THREE.Vector3): Ship | null {
   345→    let best: Ship | null = null;
   346→    let bestD = Infinity;
   347→    const consider = (s: Ship, isPlayer: boolean) => {
   348→      if (s.sunk || s.sinking > 0) return;
   349→      const d = Math.hypot(s.pos.x - playerWorld.x, s.pos.z - playerWorld.z);
   350→      if (d < 13 && d < bestD && (isPlayer || s.anchored)) {
   351→        best = s;
   352→        bestD = d;
   353→      }
   354→    };
   355→    if (this.playerShip) consider(this.playerShip, true);
   356→    for (const s of this.extraShips) consider(s, false);
   357→    return best;
   358→  }

   413→  firePlayer(side: 'port' | 'starboard'): void {
   414→    const ship = this.activeShip ?? this.playerShip;
   415→    if (!ship || !this.playerSailing || this.boarding) return;
   416→    const shots = ship.fire(side);
   417→    if (!shots) {
   418→      if (this.t - this.lastDenyT > 1.2) {
   419→        this.lastDenyT = this.t;
   420→        this.cb.notify('info', 'Cañones recargando…', 'La bordada aún no está lista.');
   421→      }
   422→      return;
   423→    }
   424→    this.executeBroadside(ship, shots, true);
   425→  }

   427→  private lastDenyT = -10;

   429→  private executeBroadside(ship: Ship, shots: CannonShot[], fromPlayer: boolean): void {
   430→    for (let i = 0; i < shots.length; i++) {
   431→      const shot = shots[i];
   432→      const spread = fromPlayer ? 0.035 : 0.085;
   433→      const dir = shot.dir.clone();
   434→      dir.x += (Math.random() - 0.5) * spread * 2;
   435→      dir.z += (Math.random() - 0.5) * spread * 2;
   436→      dir.y += 0.05 + Math.random() * 0.05;
   437→      dir.normalize();
   438→      const speed = 32 + Math.random() * 3;
   439→      this.spawnBall(shot.from, dir.multiplyScalar(speed), fromPlayer);
   440→      // Muzzle flash + gun smoke (Fase 7: real flame burst at the muzzle)
   441→      getFlameFX().burst(shot.from, 0.9, 3, 0.75);
   442→      this.flashes.spawn(shot.from, 1.2, 0.14, undefined, { grow: 3 });
   443→      this.smoke.spawn(shot.from.clone().add(new THREE.Vector3(0, 0.4, 0)), 1.8, 1.6,
   444→        new THREE.Vector3(dir.x * 2, 0.6, dir.z * 2), { grow: 1.4 });
   445→      if (i === 0) this.cb.sound('cannon');
   446→    }
   447→  }

   453→    mesh.visible = true;

   455→  }

   463→  canBoard(): EnemyBrain | null {
   464→    if (this.boarding || !this.playerShip) return null;
   465→    for (const brain of this.enemies) {
   466→      if (brain.questBoarded) continue;

   469→        if (d < 17) return brain;
   470→      }
   471→    }
   472→    return null;
   473→  }

   475→  startBoarding(brain: EnemyBrain): void {
   476→    if (this.boarding || !this.playerShip) return;
   477→    const enemy = brain.ship;
   478→    // Grapple: both ships freeze, sails down
   479→    enemy.grappled = true;
   480→    enemy.frozen = true;
   481→    enemy.sailState = 0;
   482→    enemy.rudderInput = 0;
   483→    this.playerShip.grappled = true;
   484→    this.playerShip.frozen = true;
   485→    this.playerShip.sailState = 0;
   486→    this.playerShip.rudderInput = 0;
   487→    brain.mode = 'grappled';
   488→    brain.questBoarded = true;
   489→
   490→    // Crew: the quest ship gets Morwen + two corsairs; raiders get three corsairs
   491→    const corsairPalette: CharacterPalette = {
   492→      skin: 0xb08a64, hair: 0x1c1410, tunic: 0x3a2a30, sleeves: 0x57424a, pants: 0x2c2428, boots: 0x1e1812,
   493→    };

   495→      skin: 0xc9a17a, hair: 0x2a1e2a, tunic: 0x5a1f28, sleeves: 0x7a3a3a, pants: 0x2c2024, boots: 0x1e1616, cape: 0x1c1c22,
   496→    };

   534→  /** Called by CombatSystem when a melee swing lands. Returns true if hit.
   535→   *  Fase 3: weapon reach/arc are honored (spear pikes across the deck). */
   536→  crewHitCheck(
   537→    playerWorld: THREE.Vector3, playerYaw: number, damage: number,
   538→    reach = 2.35, arc = Math.PI * 0.85,
   539→  ): boolean {
   540→    if (!this.boarding) return false;
   541→    const enemy = this.boarding.brain.ship;
   542→    let hitAny = false;
   543→    for (const c of this.boarding.crew) {
   544→      if (c.dead) continue;
   545→      const cw = enemy.toWorld(c.lx, 0, c.lz);
   546→      const dx = cw.x - playerWorld.x;
   547→      const dz = cw.z - playerWorld.z;
   548→      const dist = Math.hypot(dx, dz);
   549→      if (dist < reach) {
   550→        const facing = Math.sin(playerYaw) * dx + Math.cos(playerYaw) * dz;
   551→        // Facing within the weapon's half-arc (side distance test)
   552→        const side = Math.abs(dx * Math.cos(playerYaw) - dz * Math.sin(playerYaw));
   553→        if (facing > -0.25 && side < Math.tan(arc / 2) * Math.max(0.8, facing) + 0.5) {
   554→          c.hp -= damage;
   555→          hitAny = true;
   556→          this.cb.sound('hit_hull');
   557→          this.flashes.spawn(cw.clone().setY(cw.y + 1.2), 0.5, 0.1, undefined, { grow: 2 });
   558→          // Fase 9 — hit reaction: reel back along the blow direction
   559→          // (the captain has poise; the corsairs get bullied)
   560→          if (c.hp > 0) {
   561→            c.hurtT = Math.max(c.hurtT, c.isCaptain ? 0.28 : 0.5);
   562→            const klen = Math.hypot(dx, dz) || 1;
   563→            const push = c.isCaptain ? 0.35 : 0.8;
   564→            const nlx = c.lx + (dx / klen) * push;
   565→            const nlz = c.lz + (dz / klen) * push;
   566→            const b = enemy.deckBounds;
   567→            c.lx = clamp(nlx, b.minX + 0.2, b.maxX - 0.2);
   568→            c.lz = clamp(nlz, b.minZ + 0.6, b.maxZ - 0.4);
   569→          }
   570→          if (c.hp <= 0) {
   571→            c.dead = true;
   572→            c.moving = false;
   573→            if (c.isCaptain) {
   574→              this.cb.notify('quest', '¡Morwen ha caído!', 'El Cuervo de Tres Cabezas ha perdido a su capitana.');
   575→            }
   576→          }
   577→          break;
   578→        }
   579→      }
   580→    }
   581→    return hitAny;
   582→  }
