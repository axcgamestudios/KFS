    10→/** Shared time uniform injected into all wind-swaying materials. */
    11→export const windTime = { value: 0 };
    12→/** Fase 4 — global wind gust multiplier (weather-driven). */
    13→export const windGust = { value: 1 };

    15→/** Patch a MeshStandardMaterial so its vertices sway with the wind. */
    16→export function patchWind(material: THREE.Material, opts: { strength: number; minHeight: number; maxHeight: number }): void {
    17→  material.onBeforeCompile = (shader) => {
    18→    shader.uniforms.uTime = windTime;
    19→    shader.uniforms.uWindStrength = windGust;
    20→    shader.vertexShader = shader.vertexShader
    21→      .replace('#include <common>', '#include <common>\nuniform float uTime;\nuniform float uWindStrength;')
    22→      .replace(
    23→        '#include <begin_vertex>',
    24→        /* glsl */ `
    25→        #include <begin_vertex>
    26→        #ifdef USE_INSTANCING
    27→          float windPhase = instanceMatrix[3][0] * 0.08 + instanceMatrix[3][2] * 0.11;
    28→          float swayK = smoothstep(${opts.minHeight.toFixed(2)}, ${opts.maxHeight.toFixed(2)}, position.y) * ${opts.strength.toFixed(3)} * uWindStrength;
    29→          transformed.x += (sin(uTime * 1.35 + windPhase) + 0.5 * sin(uTime * 2.3 + windPhase * 1.7)) * 0.5 * swayK;
    30→          transformed.z += (cos(uTime * 1.05 + windPhase * 1.3) + 0.4 * sin(uTime * 2.9 + windPhase)) * 0.4 * swayK;
    31→        #endif
    32→        `,
    33→      );
    34→  };
    35→  material.customProgramCacheKey = () => `wind_${opts.strength}_${opts.minHeight}_${opts.maxHeight}`;
    36→}

    38→/** Set a vertex color attribute on a geometry (with subtle per-vertex noise). */
    39→function colorize(geo: THREE.BufferGeometry, color: THREE.Color, variation = 0.05, seed = 1): void {
    40→  const count = geo.attributes.position.count;
    41→  const colors = new Float32Array(count * 3);
    42→  const rand = mulberry32(seed);
    43→  for (let i = 0; i < count; i++) {
    44→    const v = 1 + (rand() - 0.5) * 2 * variation;
    45→    colors[i * 3] = clamp(color.r * v, 0, 1);
    46→    colors[i * 3 + 1] = clamp(color.g * v, 0, 1);
    47→    colors[i * 3 + 2] = clamp(color.b * v, 0, 1);
    48→  }
    49→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
    50→}

    52→function transformGeo(geo: THREE.BufferGeometry, x: number, y: number, z: number, sx = 1, sy = 1, sz = 1, rotY = 0, rotZ = 0): THREE.BufferGeometry {
    53→  const m = new THREE.Matrix4();
    54→  const q = new THREE.Quaternion().setFromEuler(new THREE.Euler(0, rotY, rotZ));
    55→  m.compose(new THREE.Vector3(x, y, z), q, new THREE.Vector3(sx, sy, sz));
    56→  const g = geo.index ? geo.toNonIndexed() : geo;
    57→  return g.applyMatrix4(m);
    58→}

    60→/**
    61→ * Fase 12 — trees are TWO textured parts (bark trunk + leaf canopy) for every
    62→ * species, with organically noise-displaced canopies so they read as leaf
    63→ * clusters instead of angular blobs. Species:
    64→ * 0 oak · 1 pine · 2 birch · 3 maple (golden) · 4 spruce · 5 dead/ancient
    65→ */
    66→function displacedBlob(radius: number, seed: number, amp = 0.24, squashY = 0.9): THREE.BufferGeometry {
    67→  const g = new THREE.IcosahedronGeometry(radius, 1);
    68→  const pos = g.attributes.position as THREE.BufferAttribute;
    69→  // Hash the position (not the index): shared corners displace identically → no cracks.
    70→  for (let i = 0; i < pos.count; i++) {
    71→    const x = pos.getX(i), y = pos.getY(i), z = pos.getZ(i);
    72→    const h = (frac(Math.sin(x * 12.9898 + y * 78.233 + z * 37.719) * 43758.5453) * 2 - 1) * amp;
    73→    const h2 = (frac(Math.sin(x * 93.989 + y * 47.233 + z * 17.719) * 24634.6345) * 2 - 1) * amp * 0.55;
    74→    pos.setXYZ(i, x * (1 + h), y * (1 + h2) * squashY, z * (1 + h));
    75→  }
    76→  g.computeVertexNormals();
    77→  return g;
    78→}
    79→const frac = (v: number) => v - Math.floor(v);

   233→/** Procedural billboard texture: a soft tree silhouette (impostor LOD). */
   234→function makeBillboardTexture(): THREE.Texture {
   235→  const c = document.createElement('canvas');
   236→  c.width = 128;
   237→  c.height = 128;
   238→  const ctx = c.getContext('2d')!;
   239→  ctx.clearRect(0, 0, 128, 128);
   240→  // Trunk
   241→  ctx.fillStyle = 'rgba(70,52,36,0.95)';
   242→  ctx.fillRect(60, 70, 8, 56);
   243→  // Canopy blobs (layered, soft edges via radial gradients)
   244→  const blob = (x: number, y: number, r: number, a: number) => {
   245→    const g = ctx.createRadialGradient(x, y, r * 0.1, x, y, r);
   246→    g.addColorStop(0, `rgba(255,255,255,${a})`);
   247→    g.addColorStop(0.75, `rgba(255,255,255,${a * 0.85})`);
   248→    g.addColorStop(1, 'rgba(255,255,255,0)');
   249→    ctx.fillStyle = g;
   250→    ctx.beginPath();
   251→    ctx.arc(x, y, r, 0, Math.PI * 2);
   252→    ctx.fill();
   253→  };
   254→  blob(64, 46, 40, 0.95);
   255→  blob(40, 58, 26, 0.85);
   256→  blob(88, 58, 26, 0.85);
   257→  blob(64, 26, 24, 0.9);
   258→  blob(46, 38, 20, 0.8);
   259→  blob(82, 38, 20, 0.8);
   260→  const tex = new THREE.CanvasTexture(c);
   261→  tex.colorSpace = THREE.SRGBColorSpace;
   262→  return tex;
   263→}

   276→function buildRockGeometry(): THREE.BufferGeometry {
   277→  const g = new THREE.IcosahedronGeometry(1, 1);
   278→  const pos = g.attributes.position as THREE.BufferAttribute;
   279→  const rand = mulberry32(777);
   280→  for (let i = 0; i < pos.count; i++) {
   281→    pos.setXYZ(
   282→      i,
   283→      pos.getX(i) * (0.8 + rand() * 0.4),
   284→      pos.getY(i) * (0.55 + rand() * 0.3),
   285→      pos.getZ(i) * (0.8 + rand() * 0.4),
   286→    );
   287→  }
   288→  g.computeVertexNormals();
   289→  colorize(g, new THREE.Color(0x6e6a62), 0.16, 60);
   290→  return g;

   403→      metalness: 0,
   404→      side: THREE.DoubleSide,
   405→    });
   406→    this.billboardGeo = buildBillboardGeometry();
   407→
   408→    this.rockMat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.95, metalness: 0.02 });
   409→    this.rockGeo = buildRockGeometry();
   410→
   411→    const built = buildTreeGeometries();
   412→    this.speciesParts = built;
   413→    this.treeGeos = built.map(([trunk, canopy]) => {
   414→      const g = mergeGeometries([trunk, canopy])!;
   415→      g.clearGroups();
   416→      g.addGroup(0, trunk.attributes.position.count, 0);
   417→      g.addGroup(trunk.attributes.position.count, canopy.attributes.position.count, 1);
   418→      return g;
   419→    });
   420→
   421→    // Fase 6 — prado de briznas individuales (sustituye a los matas de planos)
   422→    this.grassField = new GrassField(terrain, scene, {
   423→      density: grassDensity,
   424→      timeUniform: windTime,
   425→      gustUniform: windGust,
   426→    });
   427→    // Fase 9 — capa lejana: mattas por todo el mundo visible (misma ventana
   428→    // de exclusiónes y mismo viento que el prado fino)
   429→    this.farGrass = new FarGrass(terrain, scene, {
   430→      density: Math.min(1, grassDensity * 1.1),
   431→      timeUniform: windTime,
   432→      gustUniform: windGust,
   433→    });
   434→  }
   435→
   436→  setColliderSinks(
   437→    add: (x: number, z: number, r: number, key: string) => void,
   438→    removePrefix: (prefix: string) => void,
   439→  ): void {
   440→    this.treeColliderSink = add;
   441→    this.treeColliderRemove = removePrefix;
   442→  }
   443→
   444→  setGrassDensity(d: number): void {
   445→    if (Math.abs(d - this.grassDensity) < 0.001) return;
   446→    this.grassDensity = d;
   447→    this.grassField.setDensity(d);
   448→    this.farGrass.setDensity(Math.min(1, d * 1.1));
   449→  }

   451→  /** Fase 6 — el GrassField se autogestiona por frame en update(); esta
   452→   *  entrada forcea una comprobación inmediata de la ventana de tiles. */
   453→  refreshGrassNear(px: number, pz: number): void {
   454→    this.grassField.update(px, pz);
   455→  }

   457→  onChunkScatter(scatter: ChunkScatter): void {
   458→    const trees: TreeInstance[] = [];
   459→    const rocks: RockInstance[] = [];
   460→    const m = new THREE.Matrix4();
   461→    const q = new THREE.Quaternion();
   462→    const euro = new THREE.Euler();
   463→    const pos = new THREE.Vector3();
   464→    const scl = new THREE.Vector3();
   465→
   466→    for (const t of scatter.trees) {
   467→      euro.set(0, t.rot, 0);
   468→      q.setFromEuler(euro);
   469→      pos.set(t.x, t.y - 0.15, t.z);
   470→      scl.set(t.scale, t.scale * (0.9 + (t.x % 1) * 0.25), t.scale);
   471→      m.compose(pos, q, scl);
   472→      const c = new THREE.Color().setHSL(0.28 + ((t.z * 13.7) % 1) * 0.04, 0.32, 0.5 + ((t.x * 7.3) % 1) * 0.08);
   473→      trees.push({ species: t.species, matrix: m.clone(), color: c });
   474→      this.treeColliderSink?.(t.x, t.z, 0.42 * t.scale, `tree:${scatter.key}:${trees.length}`);
   475→    }
   476→    for (const r of scatter.rocks) {
   477→      euro.set(0, r.rot, 0);
   478→      q.setFromEuler(euro);
   479→      pos.set(r.x, r.y - r.scale * 0.35, r.z);
   480→      scl.set(r.scale, r.scale, r.scale);
   481→      m.compose(pos, q, scl);
   482→      rocks.push({ matrix: m.clone(), color: new THREE.Color().setHSL(0.08, 0.05, 0.42 + ((r.x * 3.1) % 1) * 0.12) });
   483→    }
   484→    this.chunkData.set(scatter.key, { trees, rocks });
   485→    this.markClusterDirty(scatter.cx, scatter.cz, scatter.key);
   486→  }
   487→
   488→  onChunkRemoved(key: string): void {
   489→    const data = this.chunkData.get(key);
   490→    if (!data) return;
   491→    this.chunkData.delete(key);
   492→    this.treeColliderRemove?.(`tree:${key}:`);
   493→    this.markClusterDirtyByKey(key);
   494→  }

   498→  private clusterKey(ccx: number, ccz: number): string {
   499→    return `${ccx},${ccz}`;
   500→  }
   501→
   502→  private markClusterDirty(cx: number, cz: number, chunkKey: string): void {
   503→    const ccx = Math.floor(cx / CLUSTER_CHUNKS);
   504→    const ccz = Math.floor(cz / CLUSTER_CHUNKS);
   505→    const key = this.clusterKey(ccx, ccz);
   506→    let cluster = this.clusters.get(key);
   507→    if (!cluster) {
   508→      cluster = { key, ccx, ccz, chunks: new Set(), meshes: null, dirty: true, allTrees: [] };
   509→      this.clusters.set(key, cluster);
   510→    }
   511→    cluster.chunks.add(chunkKey);
   512→    cluster.dirty = true;
   513→  }
   514→
   515→  private markClusterDirtyByKey(chunkKey: string): void {
   516→    const [cx, cz] = chunkKey.split(',').map(Number);
   517→    const ccx = Math.floor(cx / CLUSTER_CHUNKS);
   518→    const ccz = Math.floor(cz / CLUSTER_CHUNKS);
   519→    const key = this.clusterKey(ccx, ccz);
   520→    const cluster = this.clusters.get(key);
   521→    if (!cluster) return;
   522→    cluster.chunks.delete(chunkKey);
   523→    cluster.dirty = true;
   524→    if (cluster.chunks.size === 0 && cluster.meshes) {
   525→      this.disposeClusterMeshes(cluster.meshes);
   526→      cluster.meshes = null;
   527→      this.clusters.delete(key);
   528→    }
   529→  }
   530→
   531→  private disposeClusterMeshes(meshes: ClusterMeshes): void {
   532→    for (const mesh of meshes.trees) {
   533→      this.sceneRef.remove(mesh);
   534→      mesh.dispose();
   535→    }
   536→    this.sceneRef.remove(meshes.billboards);
   537→    meshes.billboards.dispose();
   538→    this.sceneRef.remove(meshes.rock);
   539→    meshes.rock.dispose();
   540→  }

   542→  /** (Re)build the instanced meshes of one cluster from its member chunks. */
   543→  private buildCluster(cluster: ClusterData): void {
   544→    if (!cluster.meshes) {
   545→      const trees: THREE.InstancedMesh[] = [];
   546→      for (let s = 0; s < SPECIES_COUNT; s++) {
   547→        // Fase 12 — every species is a two-group geometry (bark + leaf);
   548→        // the dead tree keeps bark on BOTH groups (bare branches, no foliage)
   549→        const mesh = new THREE.InstancedMesh(
   550→          this.treeGeos[s],
   551→          s === 5 ? [this.barkMat, this.barkMat] : [this.barkMat, this.leafMat],
   552→          this.speciesCapacity,
   553→        );
   554→        mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
   555→        mesh.castShadow = true;
   556→        mesh.receiveShadow = false;
   557→        mesh.count = 0;
   558→        mesh.frustumCulled = true;
   559→        this.sceneRef.add(mesh);
   560→        trees.push(mesh);
   561→      }
   562→      const billboards = new THREE.InstancedMesh(this.billboardGeo, this.billboardMat, this.billboardCapacity);
   563→      billboards.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
   564→      billboards.castShadow = false;
   565→      billboards.receiveShadow = false;
   566→      billboards.count = 0;
   567→      billboards.frustumCulled = true;
   568→      billboards.visible = false;
   569→      this.sceneRef.add(billboards);
   570→      const rock = new THREE.InstancedMesh(this.rockGeo, this.rockMat, this.rockCapacity);
   571→      rock.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
   572→      rock.castShadow = true;
   573→      rock.receiveShadow = true;
   574→      rock.count = 0;
   575→      rock.frustumCulled = true;
   576→      this.sceneRef.add(rock);
   577→      cluster.meshes = {
   578→        trees, billboards, rock,
   579→        centerX: (cluster.ccx + 0.5) * CLUSTER_SIZE,
   580→        centerZ: (cluster.ccz + 0.5) * CLUSTER_SIZE,
   581→        lod: 0,
   582→      };
   583→    }
   584→    const meshes = cluster.meshes!;
   585→    const speciesCount = new Array(SPECIES_COUNT).fill(0);
   586→    let rockCount = 0;
   587→    const allTrees: TreeInstance[] = [];
   588→    for (const chunkKey of cluster.chunks) {
   589→      const data = this.chunkData.get(chunkKey);
   590→      if (!data) continue;
   591→      for (const t of data.trees) {
   592→        allTrees.push(t);
   593→        if (speciesCount[t.species] < this.speciesCapacity) {
   594→          meshes.trees[t.species].setMatrixAt(speciesCount[t.species], t.matrix);
   595→          meshes.trees[t.species].setColorAt(speciesCount[t.species], t.color);
   596→          speciesCount[t.species]++;
   597→        }
   598→      }
   599→      for (const r of data.rocks) {
   600→        if (rockCount >= this.rockCapacity) continue;
   601→        meshes.rock.setMatrixAt(rockCount, r.matrix);
   602→        meshes.rock.setColorAt(rockCount, r.color);
   603→        rockCount++;
   604→      }
   605→    }
   606→    cluster.allTrees = allTrees;
   607→    meshes.trees.forEach((mesh, i) => {
   608→      mesh.count = speciesCount[i];
   609→      mesh.visible = mesh.count > 0;
   610→      mesh.instanceMatrix.needsUpdate = true;
   611→      if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
   612→      if (mesh.count > 0) mesh.computeBoundingSphere();
   613→    });
   614→    // Billboard impostors: one plane per tree (used when far away)
   615→    let bb = 0;
   616→    for (const t of allTrees) {
   617→      if (bb >= this.billboardCapacity) break;
   618→      meshes.billboards.setMatrixAt(bb, t.matrix);
   619→      meshes.billboards.setColorAt(bb, TREE_BILLBOARD_COLOR[t.species] ?? TREE_BILLBOARD_COLOR[0]);
   620→      bb++;
   621→    }
   622→    meshes.billboards.count = bb;
   623→    meshes.billboards.instanceMatrix.needsUpdate = true;
   624→    if (meshes.billboards.instanceColor) meshes.billboards.instanceColor.needsUpdate = true;
   625→    if (bb > 0) meshes.billboards.computeBoundingSphere();
   626→    meshes.rock.count = rockCount;
   627→    meshes.rock.visible = rockCount > 0;
   628→    meshes.rock.instanceMatrix.needsUpdate = true;
   629→    if (meshes.rock.instanceColor) meshes.rock.instanceColor.needsUpdate = true;
   630→    if (rockCount > 0) meshes.rock.computeBoundingSphere();
   631→    cluster.dirty = false;
   632→  }

   636→  /** Rebuild at most 2 dirty clusters per frame + LOD swap by distance.
   637→   *  Fase 6: el prado de briznas se actualiza aquí también (envuelve al jugador). */
   638→  update(px = NaN, pz = NaN): void {
   639→    let budget = 2;
   640→    for (const cluster of this.clusters.values()) {
   641→      if (!cluster.dirty) continue;
   642→      this.buildCluster(cluster);
   643→      budget--;
   644→      if (budget <= 0) break;
   645→    }
   646→    // LOD: full detail near, billboards far (Fase 4)
   647→    if (!Number.isNaN(px)) {
   648→      for (const cluster of this.clusters.values()) {
   649→        const meshes = cluster.meshes;
   650→        if (!meshes) continue;
   651→        const d = Math.hypot(meshes.centerX - px, meshes.centerZ - pz);
   652→        const wantLod = d > LOD_IMPOSTOR_DIST ? 1 : 0;
   653→        if (wantLod !== meshes.lod) {
   654→          meshes.lod = wantLod;
   655→          for (const t of meshes.trees) t.visible = wantLod === 0 && t.count > 0;
   656→          meshes.billboards.visible = wantLod === 1 && meshes.billboards.count > 0;
   657→        }
   658→      }
   659→      // Fase 6 — el prado sigue al jugador (ventana envolvente de tiles)
   660→      this.grassField.update(px, pz);
   661→      // Fase 9 — la pradera lejana también (ventana de 1.1 km)
   662→      this.farGrass.update(px, pz);
   663→    }
   664→  }
   665→
   666→  dispose(): void {
   667→    for (const cluster of this.clusters.values()) {
   668→      if (cluster.meshes) this.disposeClusterMeshes(cluster.meshes);
   669→    }
   670→    this.clusters.clear();
   671→    this.treeGeos.forEach((g) => g.dispose());
   672→    for (const [t, c] of this.speciesParts) { t.dispose(); c.dispose(); }
   673→    this.billboardGeo.dispose();
   674→    this.rockGeo.dispose();
   675→    this.treeMat.dispose();
   676→    this.barkMat.dispose();
   677→    this.leafMat.dispose();
   678→    (this.billboardMat.map as THREE.Texture)?.dispose();
   679→    this.billboardMat.dispose();
   680→    this.rockMat.dispose();
   681→    this.grassField.dispose();
   682→    this.farGrass.dispose();
   683→    this.chunkData.clear();
   684→  }

   686→  /** Debug/stats: clusters + el prado de briznas + la pradera lejana. */
   687→  get stats(): { clusters: number; drawCalls: number; blades: number } {
   688→    let meshes = 0;
   689→    for (const cluster of this.clusters.values()) {
   690→      if (!cluster.meshes) continue;
   691→      meshes += 2 + cluster.meshes.trees.length;
   692→    }
   693→    return {
   694→      clusters: this.clusters.size,
   695→      drawCalls: meshes,
   696→      blades: this.grassField.stats.blades + this.farGrass.stats.clumps,
   697→    };
   698→  }
