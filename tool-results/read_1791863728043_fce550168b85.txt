    42→function box(w: number, h: number, d: number, color: number, rough = 0.85): THREE.Mesh {
    43→  const m = new THREE.Mesh(
    44→    new THREE.BoxGeometry(w, h, d),
    45→    new THREE.MeshStandardMaterial({ color, roughness: rough, metalness: 0.02 }),
    46→  );
    47→  m.castShadow = true;
    48→  return m;
    49→}

    51→/**
    52→ * Common visual surface every character body implements (procedural
    53→ * CharacterModel, GLB-skinned KnightModelGLB, ...). Lets Player / NPCs /
    54→ * crews swap bodies without touching gameplay code.
    55→ */
    56→export interface CharacterVisual {
    57→  readonly root: THREE.Group;
    58→  update(p: AnimParams): void;
    59→  flash(): void;
    60→  setWeapon(kind: WeaponKind): void;
    61→  setShield(visible: boolean): void;
    62→  setEquippedState(equipped: boolean): void;
    63→  faceYaw?(yaw: number): void;
    64→  setSwordEquipped?(equipped: boolean): void;
    65→  /** Fase 7 — guard deflected a hit: recoil impulse on arms/torso. */
    66→  blockImpact?(): void;
    67→  /** Fase 8 — dusk lights the night torch in place of the shield. */
    68→  setNightTorch?(on: boolean): void;
    69→  dispose(): void;

    75→export interface AnimParams {
    76→  dt: number;

    85→  attackKind?: 'light1' | 'light2' | 'heavy' | 'thrust' | 'bowRelease'
    86→    | 'spin' | 'uppercut' | 'plunge' | 'aircut';

    88→  blockT?: number;
    89→  /** Dodge roll progress 0..1 (drives the whole-body forward roll). */
    90→  rollT?: number;
    91→  /** Fase 11 — quickstep dodge (low dash pose instead of the somersault tuck). */
    92→  dodgeStep?: boolean;
    93→  /** Bow draw blend 0..1 (aim pose while drawing). */
    94→  bowDraw?: number;
    95→  /** Fase 4 — riding pose on horseback (legs bent around the saddle, reins). */
    96→  riding?: boolean;
    97→  /** Fase 4 — riding bounce phase (0..1), driven by the horse gait. */
    98→  rideBounce?: number;

   274→  private buildWeapon(kind: WeaponKind, hip: THREE.Group, hand: THREE.Group): void {
   275→    const steel = new THREE.MeshStandardMaterial({ color: 0xaab0b6, roughness: 0.35, metalness: 0.85 });
   276→    const darkSteel = new THREE.MeshStandardMaterial({ color: 0x767c82, roughness: 0.45, metalness: 0.75 });
   277→    const guardM = new THREE.MeshStandardMaterial({ color: 0xc9a227, roughness: 0.4, metalness: 0.7 });
   278→    const woodM = new THREE.MeshStandardMaterial({ color: 0x4a3624, roughness: 0.85 });
   279→    const gripM = new THREE.MeshStandardMaterial({ color: 0x3a2a1a, roughness: 0.9 });
   280→
   281→    if (kind === 'sword') {
   282→      const mk = (parent: THREE.Group) => {
   283→        const blade = new THREE.Mesh(new THREE.BoxGeometry(0.045, 0.78, 0.012), steel);
   284→        blade.position.y = 0.52;
   285→        blade.castShadow = true;
   286→        parent.add(blade);
   287→        const tip = new THREE.Mesh(new THREE.ConeGeometry(0.033, 0.12, 4), steel);
   288→        tip.position.y = 0.96;
   289→        parent.add(tip);
   290→        const guard = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.035, 0.045), guardM);
   291→        guard.position.y = 0.13;
   292→        parent.add(guard);
   293→        const grip = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.022, 0.22, 6), gripM);
   294→        grip.position.y = 0.0;
   295→        parent.add(grip);
   296→        const pommel = new THREE.Mesh(new THREE.SphereGeometry(0.032, 6, 5), guardM);
   297→        pommel.position.y = -0.12;
   298→        parent.add(pommel);
   299→      };
   300→      mk(hip);
   301→      mk(hand);
   302→      return;
   303→    }
   304→
   305→    if (kind === 'axe') {
   306→      const mk = (parent: THREE.Group) => {
   307→        const haft = new THREE.Mesh(new THREE.CylinderGeometry(0.024, 0.028, 0.92, 7), woodM);
   308→        haft.position.y = 0.42;
   309→        haft.castShadow = true;
   310→        parent.add(haft);
   311→        // Axe head: two flaring blades joined at the socket
   312→        const headCore = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.16, 0.1), darkSteel);
   313→        headCore.position.set(0, 0.82, 0);
   314→        parent.add(headCore);
   315→        for (const s of [-1, 1]) {
   316→          const blade = new THREE.Mesh(new THREE.CylinderGeometry(0.15, 0.05, 0.03, 3), darkSteel);
   317→          blade.rotation.set(0, 0, -s * Math.PI / 2);
   318→          blade.rotation.x = Math.PI / 2;
   319→          blade.position.set(s * 0.13, 0.82, 0);
   320→          blade.castShadow = true;
   321→          parent.add(blade);
   322→        }
   323→        const grip = new THREE.Mesh(new THREE.CylinderGeometry(0.026, 0.026, 0.18, 7), gripM);
   324→        grip.position.y = 0.02;
   325→        parent.add(grip);
   326→        const pommel = new THREE.Mesh(new THREE.SphereGeometry(0.03, 6, 5), guardM);
   327→        pommel.position.y = -0.06;
   328→        parent.add(pommel);
   329→      };
   330→      mk(hip);
   331→      mk(hand);
   332→      return;
   333→    }
   334→
   335→    // spear
   336→    const mk = (parent: THREE.Group) => {
   337→      const shaft = new THREE.Mesh(new THREE.CylinderGeometry(0.021, 0.023, 1.72, 7), woodM);
   338→      shaft.position.y = 0.55;
   339→      shaft.castShadow = true;
   340→      parent.add(shaft);
   341→      const socket = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.024, 0.1, 7), guardM);
   342→      socket.position.y = 1.42;
   343→      parent.add(socket);
   344→      const tip = new THREE.Mesh(new THREE.ConeGeometry(0.045, 0.3, 6), steel);
   345→      tip.position.y = 1.6;
   346→      tip.castShadow = true;
   347→      parent.add(tip);
   348→      const butt = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.016, 0.12, 6), darkSteel);
   349→      butt.position.y = -0.32;
   350→      parent.add(butt);
   351→    };
   352→    mk(hip);
   353→    mk(hand);
   354→  }

   356→  private buildBow(parent: THREE.Group): void {
   357→    const woodM = new THREE.MeshStandardMaterial({ color: 0x5a4128, roughness: 0.75 });
   358→    // Arc of the bow: 5 short segments following a curve
   359→    const segs = 6;
   360→    for (let i = 0; i < segs; i++) {
   361→      const k = i / (segs - 1);          // 0..1 along the bow
   362→      const ang = (k - 0.5) * 1.9;       // sweep
   363→      const seg = new THREE.Mesh(new THREE.CylinderGeometry(0.014, 0.017, 0.19, 5), woodM);
   364→      seg.position.set(Math.cos(ang) * 0.12 - 0.12, Math.sin(ang) * 0.52, 0);
   365→      seg.rotation.z = -ang + Math.PI / 2;
   366→      seg.castShadow = true;
   367→      parent.add(seg);
   368→    }
   369→    // String
   370→    const stringM = new THREE.MeshStandardMaterial({ color: 0xd8d2c0, roughness: 0.9 });
   371→    const str = new THREE.Mesh(new THREE.CylinderGeometry(0.004, 0.004, 1.04, 4), stringM);
   372→    str.position.set(-0.085, 0, 0);
   373→    parent.add(str);
   374→  }

   376→  private buildShield(parent: THREE.Group): void {
   377→    // Heater shield: extruded 2D silhouette + iron boss + scale bands
   378→    const shape = new THREE.Shape();
   379→    shape.moveTo(-0.26, 0.32);
   380→    shape.quadraticCurveTo(0, 0.4, 0.26, 0.32);
   381→    shape.quadraticCurveTo(0.3, 0.05, 0.2, -0.18);
   382→    shape.quadraticCurveTo(0.1, -0.36, 0, -0.44);
   383→    shape.quadraticCurveTo(-0.1, -0.36, -0.2, -0.18);
   384→    shape.quadraticCurveTo(-0.3, 0.05, -0.26, 0.32);
   385→    const geo = new THREE.ExtrudeGeometry(shape, { depth: 0.035, bevelEnabled: true, bevelThickness: 0.012, bevelSize: 0.012, bevelSegments: 1 });
   386→    const woodM = new THREE.MeshStandardMaterial({ color: 0x4c3a26, roughness: 0.8 });
   387→    const shield = new THREE.Mesh(geo, woodM);
   388→    shield.castShadow = true;
   389→    parent.add(shield);
   390→    // Iron boss
   391→    const boss = new THREE.Mesh(
   392→      new THREE.SphereGeometry(0.075, 10, 8, 0, Math.PI * 2, 0, Math.PI / 2),
   393→      new THREE.MeshStandardMaterial({ color: 0x8a9198, roughness: 0.35, metalness: 0.85 }),
   394→    );
   395→    boss.rotation.x = Math.PI / 2;
   396→    boss.position.z = 0.03;
   397→    boss.castShadow = true;
   398→    parent.add(boss);
   399→    // Scale bands (dragon-scale look, tide blue)
   400→    const scaleM = new THREE.MeshStandardMaterial({ color: 0x2e5a5e, roughness: 0.5, metalness: 0.35 });
   401→    for (let row = 0; row < 3; row++) {
   402→      for (let i = 0; i < 4; i++) {
   403→        const scale = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.055, 0.012, 6), scaleM);
   404→        scale.rotation.x = Math.PI / 2;
   405→        scale.position.set(-0.15 + i * 0.1 + (row % 2) * 0.05, 0.2 - row * 0.19, 0.032);
   406→        parent.add(scale);
   407→      }
   408→    }
   409→  }

   413→  /** Equip a weapon visual by kind (null/'none' hides all). */
   414→  setWeapon(kind: WeaponKind): void {
   415→    this.weapon = kind;
   416→    this.weaponEquippedVisible = kind !== 'none';
   417→    this.refreshWeaponVisibility();
   418→  }
   419→
   420→  /** Show the shield on the left forearm. */
   421→  setShield(visible: boolean): void {
   422→    this.shieldGroup.visible = visible;
   423→  }
   424→
   425→  get hasShieldVisible(): boolean {
   426→    return this.shieldGroup.visible;
   427→  }
   428→
   429→  /** Legacy API (NPCs, boarding crew): sword shown at the hip. */
   430→  setSwordEquipped(equipped: boolean): void {
   431→    this.weapon = equipped ? 'sword' : 'none';
   432→    this.weaponEquippedVisible = equipped;
   433→    this.refreshWeaponVisibility();
   434→  }
   435→
   436→  /** Legacy API used by the game: any weapon equipped? */
   437→  setEquippedState(equipped: boolean): void {
   438→    this.weaponEquippedVisible = equipped;
   439→    if (equipped && this.weapon === 'none') this.weapon = 'sword';
   440→    if (!equipped) this.weapon = 'none';
   441→    this.refreshWeaponVisibility();
   442→  }
   443→
   444→  /** Brief red flash on the body when the character takes damage. */
   445→  flash(): void {
   446→    this.flashT = 1;
   447→  }
   448→
   449→  private refreshWeaponVisibility(): void {
   450→    for (const kind of ['sword', 'axe', 'spear'] as WeaponKind[]) {
   451→      const hip = this.hipSlots[kind];
   452→      const hand = this.handSlots[kind];
   453→      if (hip) hip.visible = this.weapon === kind && this.weaponEquippedVisible;
   454→      if (hand) hand.visible = false; // shown only mid-swing by update()
   455→    }
   456→  }

   752→        this.armR.rotation.z = armZ!;
   753→        this.chest.rotation.y = chestY!;

   755→      }
   756→    }
   757→
   758→    this.updateWeaponPose(p, p.attackT > 0);
   759→  }

   761→  /** Show hip/hand weapon copies + bow according to the current context. */
   762→  private updateWeaponPose(p: AnimParams, attacking: boolean): void {
   763→    const hip = this.hipSlots[this.weapon];
   764→    const showHip = this.weaponEquippedVisible && this.weapon !== 'none';
   765→    if (hip) hip.visible = showHip && !attacking;
   766→    // Hand copy: visible only mid-swing for melee weapons
   767→    if (this.weapon !== 'none' && this.weapon !== 'bow') {
   768→      const hand = this.handSlots[this.weapon];
   769→      if (hand) hand.visible = attacking;
   770→    }
   771→    // Bow: visible only while aiming/drawing (held in the left hand)
   772→    const bowDraw = p.bowDraw ?? 0;
   773→    this.bowInHand.visible = this.weapon === 'bow' && (bowDraw > 0 || (p.attackKind === 'bowRelease' && p.attackT > 0));
   774→  }

   776→  dispose(): void {
   777→    this.root.traverse((obj) => {
   778→      if ((obj as THREE.Mesh).isMesh) {
   779→        const m = obj as THREE.Mesh;
   780→        m.geometry.dispose();
   781→        (m.material as THREE.Material).dispose();
   782→      }
   783→    });
   784→  }
