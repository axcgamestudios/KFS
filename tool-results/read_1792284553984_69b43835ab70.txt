     1→/** Dialogue trees — fully data-driven, resolved by the DialogueSystem. */
     2→
     3→export type DialogueActionSingle =
     4→  | { t: 'flag'; flag: string }
     5→  | { t: 'item'; id: string; count: number }
     6→  | { t: 'takeItem'; id: string; count: number }
     7→  | { t: 'quest'; id: string }
     8→  | { t: 'xp'; amount: number }
     9→  | { t: 'talk'; target: string }
    10→  /** Fase 16 — open a merchant's shop panel and close the dialogue. */
    11→  | { t: 'shop'; vendor: string }
    12→  | { t: 'close' };
    13→
    14→/** Actions can be a single action or a list applied in order. */
    15→export type DialogueAction = DialogueActionSingle | DialogueActionSingle[];
    16→
    17→export interface DialogueOption {
    18→  text: string;
    19→  next?: string;
    20→  action?: DialogueAction;
    21→  cond?: string;   // flag must be set to show
    22→  notCond?: string; // flag must NOT be set to show
    23→  /** Fase 15 — inventory requirement: option only shows while the player
    24→   *  carries every listed item (Erica's sea-goods hub, Bram's forge). */
    25→  requires?: Array<{ id: string; count: number }>;
    26→}
    27→
    28→export interface DialogueNode {
    29→  text: string;
    30→  options: DialogueOption[];
    31→}
    32→
    33→export interface DialogueTree {
    34→  /** First matching tree for an NPC is used. */
    35→  cond?: string;
    36→  notCond?: string;
    37→  /** Marks a required item count (id>=count) for this tree to trigger. */
    38→  requireItem?: { id: string; count: number };
    39→  /** Fase 16 — player level gates (inclusive). Used by the smiths. */
    40→  minLevel?: number;
    41→  maxLevel?: number;
    42→  nodes: Record<string, DialogueNode>;
    43→  start: string;
    44→}
    45→
    46→export interface NPCDialogue {
    47→  npcId: string;
    48→  name: string;
    49→  trees: DialogueTree[];
    50→}
