     8→/**
     9→ * Fase 5 — KnightModelGLB.
    10→ * The player's body: the repository's medieval knight, rigged offline into
    11→ * knight_rigged.glb (real glTF skin + 15 landmark bones + the statue sword
    12→ * and shield as hideable rigid props hanging from the wrists).
    13→ *
    14→ * This class drives the skeleton procedurally — a full animation layer:
    15→ *   locomotion (idle / walk / run with counter-rotation, bob and knee
    16→ *   extension), jump / fall / landing absorb, dodge roll (full somersault),
    17→ *   surface swim, horse riding, block (shield brace), bow aim + release,
    18→ *   five attack curves (diagonal, backhand, overhead smash, thrust, bow
    19→ *   snap) and a hit-reaction recoil. The statue sword rests in the right
    20→ *   hand; when the game equips a weapon it swaps for the kit model, and the
    21→ *   statue shield yields to the kit shield on the left forearm.
    22→ *
    23→ * Convention: character faces +Z. RIGHT side = -X (sword arm, j_*R bones),
    24→ * LEFT side = +X (shield arm, j_*L bones). Limb rotation about +X moves a
    25→ * hanging limb backward (-Z); about -Z the right arm raises outward.
    26→ */
    27→export interface KnightVisualDeps {
    28→  scene: THREE.Object3D;
    29→  weaponFactory?: (kind: Exclude<WeaponKind, 'none'>) => THREE.Object3D | null;
    30→  shieldFactory?: () => THREE.Object3D | null;
    31→  /** Villager variant: re-tints the armour into clothes + headwear variety.
    32→   *  When set, the statue sword & shield NEVER show. */
    33→  villager?: { palette: import('./CharacterModel').CharacterPalette; seed: number; role?: string };
    34→  /** Fase 10 — bandit variant: RED knights (dark crimson cloth + oxidised
    35→   *  red-steel armour). When set the statue props never show. Fase 17:
    36→   *  `shield: true` arms heavies with a kit shield on the left forearm.
    37→   *  Fase 22: `tint`/`eyes` recolour legends and temple guardians. */
    38→  bandit?: {
    39→    seed: number; captain?: boolean; shield?: boolean;
    40→    tint?: { h: number; s: number; l: number };
    41→    eyes?: number;
    42→  };
    43→}

   104→  constructor(deps: KnightVisualDeps) {
   105→    this.weaponFactory = deps.weaponFactory;
   106→    this.shieldFactory = deps.shieldFactory;
   107→    const scene = deps.scene;
   108→    this.isVillager = !!deps.villager;
   109→    this.isBandit = !!deps.bandit;
   110→    this.banditShield = !!deps.bandit?.shield;
   111→
   112→    // collect bones + meshes
   113→    for (const name of BONE_NAMES) {
   114→      const b = scene.getObjectByName(name);
   115→      if (b) this.bones.set(name, b);
   116→    }
   117→    this.swordProp = scene.getObjectByName('j_sword_held') ?? null;
   118→    this.shieldProp = scene.getObjectByName('j_shield_held') ?? null;
   119→    // Fase 22 — the cape chain (absent in pre-F22 rigs: dynamics no-ops)
   120→    this.capeBone1 = scene.getObjectByName('j_cape1') ?? null;
   121→    this.capeBone2 = scene.getObjectByName('j_cape2') ?? null;
   122→
   123→    if (deps.villager) this.dressAsVillager(scene, deps.villager);
   124→    if (deps.bandit) this.dressAsBandit(scene, deps.bandit);
   125→
   126→    scene.traverse((o) => {
   127→      const mesh = o as THREE.Mesh;
   128→      if (mesh.isMesh) {
   129→        mesh.castShadow = true;
   130→        mesh.receiveShadow = true;
   131→        mesh.frustumCulled = false; // skeleton moves beyond the bind bbox
   132→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   133→        for (const m of mats) {
   134→          const std = m as THREE.MeshStandardMaterial;
   135→          if (std.isMeshStandardMaterial && !this.flashMats.includes(std)) this.flashMats.push(std);
   136→        }
   137→      }
   138→    });
   139→
   140→    // attach slots
   141→    const wristR = this.bones.get('j_wristR');
   142→    if (wristR) { wristR.add(this.handSlot); }
   143→    const elbowL = this.bones.get('j_elbowL');
   144→    if (elbowL) { elbowL.add(this.armSlot); this.armSlot.position.set(0, -0.12, 0.05); }
   145→
   146→    // The GLB clone root can arrive with visible=false (something in the boot
   147→    // pipeline toggles loaded model roots); the knight must always render.
   148→    // GLTF scene roots can arrive with visible=false after loader round-trips;
   149→    // the player's body must always render.
   150→    scene.visible = true;
   151→
   152→    this.inner.add(scene);
   153→    this.root.add(this.inner);
   154→    this.refreshProps();
   155→  }

   193→  /**
   194→   * Fase 8 — at dusk the shield gives way to a lit torch: a wooden brand
   195→   * with a real FlameFX flame and a warm flickering light, riding the left
   196→   * forearm slot. At dawn the torch goes out and the shield returns.

   198→  setNightTorch(on: boolean): void {
   199→    if (on === this.torchOn || this.isVillager) return;
   200→    this.torchOn = on;
   201→    if (on) {
   202→      const g = new THREE.Group();
   203→      g.name = 'night_torch';
   204→      const wood = new THREE.MeshStandardMaterial({ color: 0x5a4028, roughness: 0.9 });
   205→      const stick = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.042, 0.52, 6), wood);
   206→      stick.position.y = 0.16;
   207→      stick.castShadow = true;
   208→      g.add(stick);
   209→      const wrap = new THREE.Mesh(
   210→        new THREE.CylinderGeometry(0.062, 0.05, 0.15, 8),
   211→        new THREE.MeshStandardMaterial({ color: 0x3a2c1c, roughness: 1 }),
   212→      );
   213→      wrap.position.y = 0.46;
   214→      g.add(wrap);
   215→      this.torchLight = new THREE.PointLight(0xff9a3c, 2.4, 20, 1.7);
   216→      this.torchLight.position.y = 0.55;
   217→      g.add(this.torchLight);
   218→      this.armSlot.add(g);
   219→      g.rotation.set(-0.35, 0, 0);
   220→      g.position.set(-0.02, -0.05, 0.1);
   221→      this.torchProp = g;

   223→      this.torchFlame = getFlameFX().addFlame(new THREE.Vector3(0, -100, 0), {
   224→        scale: 0.17, height: 0.42, heat: 0.5,
   225→      });

   227→      if (this.activeShield) this.activeShield.visible = false;
   228→      if (this.shieldProp) this.shieldProp.visible = false;
   229→    } else {
   230→      if (this.torchProp) { this.armSlot.remove(this.torchProp); this.torchProp = null; }
   231→      if (this.torchFlame) { this.torchFlame.dispose(); this.torchFlame = null; }
   232→      this.torchLight = null;
   233→      this.torchOn = false;
   234→      // the shield returns with the sun
   235→      this.refreshProps();
   236→    }

   293→  private weaponTier = 0;
   294→  private weaponElement: SwordElement | null = null;

   296→  private applyWeaponGlow(): void {
   297→    const w = this.activeWeapon ?? this.swordProp;
   298→    if (!w) return;
   299→    // Fase 11 — the master-forged sword carries its own rune-glow materials
   300→    if (w.userData?.heroSword) {
   301→      applyHeroSwordTier(w, this.weaponTier, false);
   302→      if (this.weaponElement) applyHeroSwordElement(w, this.weaponElement);
   303→      const s = 1 + this.weaponTier * 0.05 + (this.weaponElement ? 0.06 : 0);
   304→      w.scale.set(s, s, s);
   305→      return;
   306→    }
   307→    const glows = [null, 0xff9a3c, 0xe8c96a, 0x7ec8ff] as const;
   308→    const glow = glows[this.weaponTier];
   309→    w.traverse((o) => {
   310→      const mesh = o as THREE.Mesh;
   311→      if (!mesh.isMesh) return;
   312→      const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   313→      for (const m of mats) {
   314→        const std = m as THREE.MeshStandardMaterial;
   315→        if (!std.isMeshStandardMaterial) continue;
   316→        if (glow === null) {
   317→          std.emissive?.setHex(0x000000);
   318→        } else {
   319→          std.emissive?.setHex(glow);
   320→          std.emissiveIntensity = 0.55 + this.weaponTier * 0.3;
   321→        }
   322→        std.needsUpdate = true;
   323→      }
   324→    });
   325→    // each tier forges a slightly longer, prouder blade
   326→    const s = 1 + this.weaponTier * 0.06;
   327→    w.scale.set(s, s, s);
   328→  }

   330→  private refreshProps(): void {
   331→    // Kit weapon in the right hand (detaches the previous one first)
   332→    if (this.activeWeapon) {
   333→      this.handSlot.remove(this.activeWeapon);
   334→      if (this.activeWeapon.parent) this.activeWeapon.parent.remove(this.activeWeapon);
   335→      this.activeWeapon = null;
   336→    }
   337→    const showKitWeapon = !this.isVillager && this.weaponKind !== 'none';
   338→    if (showKitWeapon && this.weaponFactory) {
   339→      const w = this.weaponFactory(this.weaponKind as Exclude<WeaponKind, 'none'>);
   340→      if (w) {
   341→        if (this.weaponKind === 'bow') {
   342→          // bows live permanently in the LEFT hand (aim code extends the arm)
   343→          const wristL = this.bones.get('j_wristL');
   344→          if (wristL) {
   345→            wristL.add(w);
   346→            w.position.set(0.02, -0.08, 0.05);
   347→            w.rotation.set(0, 0, 0.08);
   348→          }
   349→        } else {
   350→          this.handSlot.add(w);
   351→        }
   352→        this.activeWeapon = w;
   353→        this.tuneWeaponPose(this.weaponKind);
   354→        this.applyWeaponGlow();
   355→      }
   356→    }
   357→    // shields: the kit shield on the left forearm
   358→    if (this.activeShield) {
   359→      this.armSlot.remove(this.activeShield);
   360→      this.activeShield = null;
   361→    }
   362→    if (this.wantsShield && !this.isVillager && (!this.isBandit || this.banditShield) && this.shieldFactory) {
   363→      const s = this.shieldFactory();
   364→      if (s) {
   365→        this.armSlot.add(s);
   366→        this.activeShield = s;
   367→        this.tuneShieldPose();
   368→      }
   369→    }
   370→    // Fase 7 — NOTHING is shown until equipped:
   371→    //   · bare hands → no statue sword, no statue shield
   372→    //   · the statue props only appear as FALLBACKS when the matching kit
   373→    //     object could not be produced (kit GLB missing)
   374→    if (this.swordProp) {
   375→      this.swordProp.visible = !this.isVillager && !this.isBandit
   376→        && this.weaponKind !== 'none'
   377→        && this.weaponKind !== 'bow'
   378→        && !this.activeWeapon;
   379→    }
   380→    if (this.shieldProp) {
   381→      this.shieldProp.visible = !this.isVillager && this.wantsShield && !this.activeShield
   382→        && (!this.isBandit || this.banditShield);
   383→    }
   384→  }
   385→
   386→  /**
   387→   * Fase 7 — villager variant: the repository knight's body re-dressed as a
   388→   * commoner. Materials are CLONED first (the AssetSystem shares them with
   389→   * the player), the armour is re-tinted to cloth/leather tones and each NPC
   390→   * gets seeded headwear (hood / straw hat / cap) + a work tool by role.
   391→   */
   392→  private dressAsVillager(scene: THREE.Object3D, v: { palette: import('./CharacterModel').CharacterPalette; seed: number; role?: string }): void {
   393→    const r = (k: number) => {
   394→      const x = Math.sin(v.seed * 37.71 + k * 91.13) * 43758.5453;
   395→      return x - Math.floor(x);
   396→    };
   397→    const tint = new THREE.Color();
   398→    scene.traverse((o) => {
   399→      const mesh = o as THREE.Mesh;
   400→      if (!mesh.isMesh) return;
   401→      const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   402→      const clones = mats.map((m) => (m as THREE.Material).clone());
   403→      for (const m of clones) {
   404→        const std = m as THREE.MeshStandardMaterial;
   405→        if (!std.isMeshStandardMaterial) continue;
   406→        const isUpper = /upper/i.test(std.name ?? '');
   407→        const base = isUpper ? v.palette.tunic : v.palette.pants;
   408→        tint.setHex(base);
   409→        // seeded shade variety per NPC (avoid clone-army look)
   410→        const shade = 0.82 + r(9) * 0.36;
   411→        tint.multiplyScalar(shade);
   412→        std.color.copy(tint);
   413→        std.roughness = 0.92;
   414→        std.metalness = 0.04;
   415→        std.emissive?.setHex(0x000000);
   416→        std.needsUpdate = true;
   417→      }
   418→      mesh.material = clones.length === 1 ? clones[0] : clones;
   419→    });
   420→
   421→    // --- headwear on the head bone (rigid props ride the skeleton) ---
   422→    const head = this.bones.get('j_head') ?? scene.getObjectByName('j_head');
   423→    if (head) {
   424→      const cloth = (c: number, rough = 0.95) =>
   425→        new THREE.MeshStandardMaterial({ color: c, roughness: rough, metalness: 0.02 });
   426→      const wear = r(2);
   427→      if (wear < 0.34) {
   428→        // hood
   429→        const hood = new THREE.Mesh(new THREE.ConeGeometry(0.155, 0.3, 8, 1, true), cloth(v.palette.tunic));
   430→        hood.position.y = 0.22;
   431→        hood.rotation.x = -0.14;
   432→        hood.castShadow = true;
   433→        hood.userData.villagerProp = true;
   434→        head.add(hood);
   435→      } else if (wear < 0.64) {
   436→        // straw hat (brim + crown)
   437→        const g = new THREE.Group();
   438→        const brim = new THREE.Mesh(new THREE.CylinderGeometry(0.26, 0.28, 0.025, 12), cloth(0xb59a58));
   439→        const crown = new THREE.Mesh(new THREE.CylinderGeometry(0.13, 0.15, 0.13, 10), cloth(0xa8894a));
   440→        crown.position.y = 0.07;
   441→        brim.castShadow = crown.castShadow = true;
   442→        g.add(brim, crown);
   443→        g.userData.villagerProp = true;
   444→        g.position.y = 0.26;
   445→        g.rotation.z = (r(3) - 0.5) * 0.24;
   446→        head.add(g);
   447→      } else if (wear < 0.85) {
   448→        // soft cap
   449→        const cap = new THREE.Mesh(
   450→          new THREE.SphereGeometry(0.15, 10, 8, 0, Math.PI * 2, 0, Math.PI * 0.52),
   451→          cloth(v.palette.sleeves),
   452→        );
   453→        cap.position.y = 0.17;
   454→        cap.castShadow = true;
   455→        cap.userData.villagerProp = true;
   456→        head.add(cap);
   457→      }
   458→    }
   459→
   460→    // --- role tool in the right hand ---
   461→    const wristR = scene.getObjectByName('j_wristR');
   462→    if (wristR && v.role) {
   463→      const wood = new THREE.MeshStandardMaterial({ color: 0x6a4e2e, roughness: 0.9, metalness: 0.02 });
   464→      const steel = new THREE.MeshStandardMaterial({ color: 0x767c82, roughness: 0.42, metalness: 0.78 });
   465→      if (v.role === 'smith') {
   466→        const hammer = new THREE.Group();
   467→        const haft = new THREE.Mesh(new THREE.CylinderGeometry(0.016, 0.02, 0.34, 6), wood);
   468→        haft.position.y = 0.14;
   469→        const hm = new THREE.Mesh(new THREE.BoxGeometry(0.075, 0.055, 0.05), steel);
   470→        hm.position.y = 0.3;
   471→        hammer.add(haft, hm);
   472→        hammer.rotation.x = 1.35;
   473→        hammer.userData.villagerProp = true;
   474→        wristR.add(hammer);
   475→      } else if (v.role === 'merchant') {
   476→        const sack = new THREE.Mesh(
   477→          new THREE.SphereGeometry(0.09, 8, 6),
   478→          new THREE.MeshStandardMaterial({ color: 0x9a8a66, roughness: 0.98, metalness: 0.0 }),
   479→        );

   500→   * Fase 10 — BANDIT variant: the same repository knight re-forged as a RED
   501→   * KNIGHT — dark crimson cloth, oxidised red-steel armour, ember eyes and
   502→   * a torn red cloak. Materials are CLONED (they are shared with the player).

   504→  private dressAsBandit(scene: THREE.Object3D, v: { seed: number; captain?: boolean; tint?: { h: number; s: number; l: number }; eyes?: number }): void {
   505→    const r = (k: number) => {
   506→      const x = Math.sin(v.seed * 37.71 + k * 91.13) * 43758.5453;
   507→      return x - Math.floor(x);
   508→    };
   509→    const tint = new THREE.Color();
   510→    scene.traverse((o) => {
   511→      const mesh = o as THREE.Mesh;
   512→      if (!mesh.isMesh) return;
   513→      const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   514→      const clones = mats.map((m) => (m as THREE.Material).clone());
   515→      for (const m of clones) {
   516→        const std = m as THREE.MeshStandardMaterial;
   517→        if (!std.isMeshStandardMaterial) continue;
   518→        const name = (std.name ?? '').toLowerCase();
   519→        const isUpper = /upper/i.test(name);
   520→        if (isUpper) {
   521→          // oxidised red steel with seeded shade variety
   522→          // (Fase 22 — legends/guardians wear their own HSL instead)
   523→          if (v.tint) {
   524→            tint.setHSL(v.tint.h, v.tint.s, v.tint.l);
   525→            std.metalness = 0.62;
   526→            std.roughness = 0.42;
   527→          } else {
   528→            tint.setHSL(0.985 + r(3) * 0.012, 0.52 + r(4) * 0.2, 0.26 + r(5) * 0.08);
   529→            std.metalness = 0.55;
   530→            std.roughness = 0.5;
   531→          }
   532→          std.color.copy(tint);
   533→        } else {
   534→          // dark crimson trousers / leather (or the legend's own shade)
   535→          if (v.tint) {
   536→            tint.setHSL(v.tint.h, Math.min(1, v.tint.s + 0.1), v.tint.l * 0.72);
   537→          } else {
   538→            tint.setHSL(0.99, 0.42, 0.16 + r(6) * 0.07);
   539→          }
   540→          std.color.copy(tint);
   541→          std.metalness = 0.12;
   542→          std.roughness = 0.85;
   543→        }
   544→        std.needsUpdate = true;
   545→      }
   546→      mesh.material = clones.length === 1 ? clones[0] : clones;
   547→    });
   548→    // ember eyes — a tiny malevolent glow under the helm
   549→    // (Fase 22 — legends and guardians choose their own colour)
   550→    const head = this.bones.get('j_head') ?? scene.getObjectByName('j_head');
   551→    if (head) {
   552→      const eyeMat = new THREE.MeshStandardMaterial({
   553→        color: 0x220000, emissive: v.eyes ?? 0xff2a1a, emissiveIntensity: 2.4, roughness: 0.4,
   554→      });

   564→      const torso = this.bones.get('j_torso');

   566→        const cloakMat = new THREE.MeshStandardMaterial({
   567→          color: 0x5a1216, roughness: 0.95, side: THREE.DoubleSide,

   570→        cloak.position.set(0, -0.28, -0.16);

   572→        cloak.userData.villagerProp = true;
   573→        torso.add(cloak);
   574→      }
   575→      this.root.scale.setScalar(1.08); // captains stand taller
   576→    } else {
   577→      this.root.scale.setScalar(0.97 + r(7) * 0.07);
   578→    }

   612→  /** Fase 14 — scratch reutilizable: zero() creaba un objeto + 15 arrays por
   613→   *  frame POR MODELO (17 caballeros = 270 arrays/frame de presión de GC). */
   614→  private zeroScratch: PoseTarget = {};
   615→  private zero(): PoseTarget {
   616→    const p = this.zeroScratch;
   617→    for (const n of BONE_NAMES) {
   618→      const e = p[n];
   619→      if (e) { e[0] = 0; e[1] = 0; e[2] = 0; }
   620→      else p[n] = [0, 0, 0];
   621→    }
   622→    return p;
   623→  }

   625→  update(p: AnimParams): void {
   626→    // Fase 14 — culling de animación: lejos del jugador el sistema dueño
   627→    // pasa dt = 0 y la pose queda congelada (invisible a esa distancia,
   628→    // todo el álgebra de huesos ahorrada). El propio modelo sigue en su sitio.
   629→    if (p.dt <= 0) return;
   630→    this.t += p.dt;
   631→    const dt = Math.max(1e-4, p.dt);
   632→    const target = this.zero();
   633→
   634→    // speed smoothing + phase
   635→    this.speedSm = damp(this.speedSm, p.speed, 8, dt);
   636→    const moving = p.moving && this.speedSm > 0.25;
   637→    if (moving) this.phase += dt * (this.speedSm > 4.4 ? 9.4 : 6.6);
   638→    const ph = this.phase;
   639→
   640→    // landing detection
   641→    if (p.grounded && !this.wasGrounded) this.landT = p.attackT > 0 || (p.rollT ?? 0) > 0 ? 0 : 0.34;
   642→    if (!p.grounded) this.airT += dt; else this.airT = 0;
   643→    this.wasGrounded = p.grounded;
   644→    if (this.landT > 0) this.landT = Math.max(0, this.landT - dt);
   645→
   646→    // ---- roll/swim/ride overrides ----
   647→    const rollT = p.rollT ?? 0;
   648→    if (rollT > 0) { this.poseRoll(target, rollT, moving, ph, p.dodgeStep === true); this.apply(target, dt, true); this.post(p); return; }
   649→    if (p.swimming) { this.poseSwim(target, ph, p); this.apply(target, dt, true); this.post(p); return; }
   650→    if (p.riding) { this.poseRide(target, p); this.apply(target, dt, true); this.post(p); return; }
   651→
   652→    // ---- Fase 17 — the HOOK DASH flying kick: the rope carries the body
   653→    //      and both heels lead the way. Held while the kick lingers. ----
   654→    if (p.dashKick && p.attackT === 0) {
   655→      this.poseHookKick(target);
   656→      this.apply(target, dt, true);
   657→      this.post(p);
   658→      return;
   659→    }
   660→
   661→    // ---- Fase 11 — LAYERED base: air OR locomotion (no early return any
   662→    // more: the attack/block overlays now layer over the airborne pose too,
   663→    // which is what makes mid-air slashes and plunges readable) ----
   664→    if (!p.grounded) {
   665→      this.poseAir(target, this.airT);
   666→    } else {
   667→      this.poseLocomotion(target, p, moving, ph);
   668→    }
   669→
   670→    // ---- landing absorb (grounded only) ----
   671→    if (p.grounded) {
   672→      if (this.landT > 0) {
   673→        const k = this.landT / 0.34;
   674→        const e = k * k;
   675→        target.j_kneeR = [target.j_kneeR[0] + 0.85 * e, 0, 0];
   676→        target.j_kneeL = [target.j_kneeL[0] + 0.85 * e, 0, 0];
   677→        target.j_hipR = [target.j_hipR[0] - 0.5 * e, 0, 0];
   678→        target.j_hipL = [target.j_hipL[0] - 0.5 * e, 0, 0];
   679→        target.j_torso = [target.j_torso[0] + 0.28 * e, target.j_torso[1], 0];
   680→        this.inner.position.y = -0.09 * e;
   681→      } else if (this.inner.position.y !== 0) {
   682→        this.inner.position.y = damp(this.inner.position.y, 0, 10, dt);
   683→      }
   684→    }
   685→
   686→    // ---- bow aim (overrides arms) ----
   687→    const bowDraw = p.bowDraw ?? 0;
   688→    if (bowDraw > 0) this.poseBowAim(target, p, bowDraw, moving, ph);
   689→
   690→    // ---- block overlay (grounded stance only) ----
   691→    const blockT = p.blockT ?? 0;
   692→    if (blockT > 0 && p.grounded) this.poseBlock(target, blockT, this.activeShield !== null || (!!this.shieldProp && this.shieldProp.visible));
   693→
   694→    // ---- attack overlays (timed curves — ground OR air) ----
   695→    if (p.attackT > 0) {
   696→      this.poseAttack(target, p.attackT, p.attackKind ?? 'light1', bowDraw);
   697→      this.apply(target, dt, true);
   698→      this.post(p);
   699→      return;
   700→    }
   701→
   702→    // Fase 17 — the sweep-kick spin unwinds when the kick isn't driving it
   703→    this.inner.rotation.y = damp(this.inner.rotation.y, 0, 14, dt);
   704→
   705→    // ---- hit reaction ----
   706→    if (this.hitT > 0) {
   707→      const k = this.hitT;
   708→      // Fase 22 — the flinch reads through the WHOLE upper body now: chest
   709→      // caves, head snaps down and rolls, both shoulders recoil off the line
   710→      // of the blow — a reaction with mass, not a red flash.
   711→      target.j_torso = [target.j_torso[0] - 0.22 * k, target.j_torso[1], 0.05 * k];
   712→      target.j_head = [target.j_head[0] - 0.18 * k, target.j_head[1] * (1 - k), -0.08 * k];
   713→      target.j_shoulderR = [target.j_shoulderR[0], target.j_shoulderR[1], target.j_shoulderR[2] - 0.16 * k];
   714→      target.j_shoulderL = [target.j_shoulderL[0], target.j_shoulderL[1], target.j_shoulderL[2] + 0.12 * k];
   715→    }
   716→
   717→    this.apply(target, dt, false);
   718→    this.post(p);
   719→  }

   721→  /** Per-frame post steps (flash decay, prop visibility by context). */
   722→  private post(p: AnimParams): void {
   723→    // Fase 22 — THE CAPE CLOTH LAYER runs on every path (locomotion, attack
   724→    // early-returns included — the cape answers to the swing's coil-whip)
   725→    if (p.dt > 0) this.updateCape(p, Math.max(1e-4, p.dt));
   726→    if (this.flashT > 0) {
   727→      this.flashT = Math.max(0, this.flashT - p.dt * 3.2);
   728→      const e = this.flashT * 0.6;
   729→      for (const m of this.flashMats) m.emissive.setRGB(e, e * 0.12, e * 0.08);
   730→    }
   731→    if (this.hitT > 0) this.hitT = Math.max(0, this.hitT - p.dt * 2.6);
   732→    // the statue shield (fallback only) tucks away while aiming a bow
   733→    if (this.shieldProp && !this.isVillager) {
   734→      this.shieldProp.visible = this.wantsShield && !this.activeShield && (p.bowDraw ?? 0) < 0;
   735→    }
   736→    // shield impact recoil decay
   737→    if (this.blockHitT > 0) this.blockHitT = Math.max(0, this.blockHitT - p.dt * 4.2);
   738→    // Fase 8 — the night torch rides the left forearm: sync the flame to the
   739→    // brand head and flicker the light with the same rhythm
   740→    if (this.torchOn && this.torchProp && this.torchFlame) {
   741→      this.torchT += p.dt;
   742→      this.torchProp.updateWorldMatrix(true, false);
   743→      this.torchFlame.position.setFromMatrixPosition(this.torchProp.matrixWorld);
   744→      this.torchFlame.position.y += 0.55;
   745→      if (this.torchLight) {
   746→        const fl = 0.8 + Math.sin(this.torchT * 11.3) * 0.14 + Math.sin(this.torchT * 23.7) * 0.07;
   747→        this.torchLight.intensity = gfxRuntime.pointLights ? 2.4 * fl : 0;
   748→      }
   749→    }
   750→    // the kit weapon stays in the hand; while swimming it is shouldered up
   751→    if (this.activeWeapon) {
   752→      const swim = p.swimming;
   753→      this.activeWeapon.rotation.x = damp(
   754→        this.activeWeapon.rotation.x, swim ? -1.2 : this.activeWeapon.userData.baseRotX ?? this.activeWeapon.rotation.x, 6, Math.max(1e-4, p.dt),
   755→      );
   756→    }
   757→  }

   761→  private poseLocomotion(t: PoseTarget, p: AnimParams, moving: boolean, ph: number): void {
   762→    const runK = Math.min(1, this.speedSm / 6.2);
   763→    if (moving) {
   764→      // ---- Fase 17 — the charge blend: past ~65% of top speed the knight
   765→      // drops into the SHIELD CHARGE — shield forward, body low, blade
   766→      // trailing. It eases in with speed so walking never triggers it. ----
   767→      const hasShield = this.wantsShield
   768→        && (this.activeShield !== null || (this.shieldProp?.visible ?? false));
   769→      const charge = hasShield
   770→        ? THREE.MathUtils.smoothstep(runK, 0.55, 0.88)
   771→        : THREE.MathUtils.smoothstep(runK, 0.68, 0.92);
   772→      const armed = this.weaponKind !== 'none' && this.weaponKind !== 'bow';
   773→
   774→      const amp = 0.52 + runK * 0.5 + charge * 0.12;
   775→      const legPhL = Math.sin(ph);
   776→      const legPhR = Math.sin(ph + Math.PI);
   777→      // Fase 12 — pelvis rocks over the stance leg (secondary motion: the
   778→      // walk stops reading as two stiff pistons and starts swinging from
   779→      // the hips like a real gait).
   780→      const hipRoll = Math.sin(ph) * (0.035 + runK * 0.05);
   781→      // right leg (x<0) — j_hipR: -X = forward
   782→      t.j_hipR = [-legPhR * amp, 0, -hipRoll];
   783→      t.j_kneeR = [Math.max(0, -Math.sin(ph + Math.PI - 0.7)) * (0.55 + runK * 0.75 + charge * 0.22), 0, 0];
   784→      t.j_ankleR = [-0.1 + legPhR * 0.18, 0, Math.max(0, legPhR) * 0.08];
   785→      // left leg
   786→      t.j_hipL = [-legPhL * amp, 0, hipRoll];
   787→      t.j_kneeL = [Math.max(0, -Math.sin(ph - 0.7)) * (0.55 + runK * 0.75 + charge * 0.22), 0, 0];
   788→      t.j_ankleL = [-0.1 + legPhL * 0.18, 0, Math.max(0, legPhL) * 0.08];
   789→
   790→      // ---- ARMS ----
   791→      // Fase 17 — the sword arm never pendulums while armed (the old full
   792→      // counter-swing with a blade in hand read as broken). It holds a low
   793→      // READY CARRY: elbow bent, blade angled down-forward, a hint of swing
   794→      // so the gait still breathes. Charging, the arm trails back like a
   795→      // lance at rest.
   796→      const armAmp = 0.34 + runK * 0.5;
   797→      if (armed) {
   798→        const ready = -0.22 + legPhR * 0.13;
   799→        const trailing = 0.55 + legPhR * 0.08;
   800→        t.j_shoulderR = [THREE.MathUtils.lerp(ready, trailing, charge), 0, THREE.MathUtils.lerp(-0.1, -0.18, charge)];
   801→        t.j_elbowR = [THREE.MathUtils.lerp(-0.62, -0.3, charge), 0, 0];
   802→      } else {
   803→        t.j_shoulderR = [legPhR * armAmp * (1 - charge * 0.5), 0, -0.06];
   804→        t.j_elbowR = [-0.18 - runK * 0.35 - Math.max(0, -legPhR) * (0.3 + runK * 0.3), 0, 0];
   805→      }
   806→      // Fase 10/17 — SHIELD CARRY → SHIELD CHARGE: a knight does NOT swing
   807→      // the shield arm like a pendulum. Walking: elbow bent, shield held low
   808→      // across the thigh with a touch of bob. Charging: the shield snaps UP
   809→      // AND FORWARD — the battering-ram brace.
   810→      const shieldCarry = hasShield && (p.blockT ?? 0) < 0.1;
   811→      if (shieldCarry) {
   812→        const bob = Math.sin(ph * 2) * 0.045;
   813→        const walkPose: [number, number, number] = [-0.36 + bob - runK * 0.12, 0.18, 0.32];
   814→        const chargePose: [number, number, number] = [-1.12, 0.28, 0.3];
   815→        t.j_shoulderL = [
   816→          THREE.MathUtils.lerp(walkPose[0], chargePose[0], charge),
   817→          THREE.MathUtils.lerp(walkPose[1], chargePose[1], charge),
   818→          THREE.MathUtils.lerp(walkPose[2], chargePose[2], charge),
   819→        ];
   820→        t.j_elbowL = [THREE.MathUtils.lerp(-0.94, -1.02, charge), 0, 0.16];
   821→      } else {
   822→        t.j_shoulderL = [legPhL * armAmp * 0.8 * (1 - charge * 0.5), 0, 0.06];
   823→        t.j_elbowL = [-0.25 - runK * 0.3 - Math.max(0, -legPhL) * (0.28 + runK * 0.28), 0, 0];
   824→      }
   825→
   826→      // ---- TORSO & HEAD ----
   827→      // The charge crouches into the run: deep lean, eyes UP over the rim.
   828→      const lean = 0.05 + runK * 0.2 + charge * 0.16;
   829→      t.j_torso = [lean, Math.sin(ph) * (0.05 + runK * 0.08), 0];
   830→      t.j_head = [
   831→        -0.04 - runK * 0.08 - charge * 0.2,
   832→        -Math.sin(ph) * 0.06,
   833→        -Math.sin(ph) * 0.02,
   834→      ];
   835→      // Fase 21 — the whole body ROLLS over the stance foot (weight shift in
   836→      // the ribcage, not just a positional sway: kills the stiff-plank read)
   837→      this.inner.rotation.z = damp(this.inner.rotation.z, Math.sin(ph) * (0.02 + runK * 0.035), 8, 1 / 60);
   838→      this.inner.position.y = Math.abs(Math.sin(ph)) * 0.028 * (0.5 + runK + charge * 0.5) - charge * 0.05;
   839→      // Fase 12 — subtle lateral sway of the whole body over the stance foot
   840→      this.inner.position.x = Math.sin(ph) * 0.012 * (0.6 + runK);
   841→      // Fase 21 — WRIST follow-through: the hands lag the arm swing a beat
   842→      // (relaxed wrists are half of "not rigid"); armed, the grip stays quiet
   843→      if (!armed) {
   844→        t.j_wristR = [legPhR * 0.14 * (0.4 + runK), 0, -0.05 + Math.sin(ph * 2) * 0.03];
   845→        t.j_wristL = [legPhL * 0.12 * (0.4 + runK), 0, 0.05 - Math.sin(ph * 2) * 0.03];
   846→      } else {
   847→        t.j_wristR = [-0.06, 0, Math.sin(ph) * 0.025];
   848→        t.j_wristL = [-0.05, 0, -Math.sin(ph) * 0.025];
   849→      }
   850→    } else {
   851→      // idle: breathing, weight shift, relaxed sword arm
   852→      const br = Math.sin(this.t * 1.7);
   853→      const sway = Math.sin(this.t * 0.33);
   854→      t.j_torso = [0.02 + br * 0.012, Math.sin(this.t * 0.4) * 0.02, sway * 0.012];
   855→      // Fase 21 — idle head GLANCES: the knight scans the horizon now and
   856→      // then instead of staring dead ahead
   857→      const glance = Math.sin(this.t * 0.21) * Math.sin(this.t * 0.07 + 1.3);
   858→      t.j_head = [0.0, Math.sin(this.t * 0.5) * 0.13 + glance * 0.22, Math.sin(this.t * 0.33) * 0.05];
   859→      t.j_shoulderR = [br * 0.03 - 0.04, 0, -0.05];
   860→      t.j_shoulderL = [-br * 0.03, 0, 0.06];
   861→      t.j_elbowR = [-0.15 - br * 0.02, 0, 0];
   862→      t.j_elbowL = [-0.22, 0, 0];
   863→      // Fase 21 — relaxed wrists at rest (a touch of life in the hands)
   864→      t.j_wristR = [-0.04 + br * 0.015, 0, -0.04];
   865→      t.j_wristL = [-0.05 - br * 0.015, 0, 0.05];
   866→      // Fase 12 — the knees breathe with the weight shift
   867→      t.j_kneeR = [-0.04 + Math.max(0, sway) * 0.05, 0, 0];
   868→      t.j_kneeL = [-0.06 + Math.max(0, -sway) * 0.05, 0, 0];
   869→      t.j_hipR = [0, 0, -sway * 0.018];
   870→      t.j_hipL = [0, 0, sway * 0.018];

  1077→  /** Attack curves. attackT: 0→1 across the whole swing. */
  1078→  private poseAttack(t: PoseTarget, k: number, kind: string, bowDraw: number): void {
  1079→    const L = THREE.MathUtils.lerp;
  1080→    const ease = (u: number) => u * u * (3 - 2 * u);
  1081→    // Fase 22 — the AAA easing set:
  1082→    //  · wuEase (quadratic-OUT): launches into the gather, then settles into
  1083→    //    the hold. The windup reads as WEIGHT TRANSFER — anticipation — not a
  1084→    //    linear wind-up clock.
  1085→    //  · strikeEase (cubic-OUT): the blade EXPLODES out of the windup and
  1086→    //    decelerates through the target. A whip, not a ramp.
  1087→    const wuEase = (u: number) => 1 - (1 - u) * (1 - u);
  1088→    const strikeEase = (u: number) => 1 - Math.pow(1 - u, 3);
  1089→    if (kind === 'heavy') {
  1090→      // overhead smash: windup up-back (0..0.42), slam (0.42..0.68), recover
  1091→      if (k < 0.42) {
  1092→        const u = wuEase(k / 0.42);
  1093→        t.j_shoulderR = [L(0, -0.52, u), 0, L(0, -2.95, u)];
  1094→        t.j_elbowR = [-0.6, 0, 0];
  1095→        t.j_torso = [0.1, L(0, -0.5, u), 0];
  1096→        t.j_head = [L(0, -0.3, u), 0, 0];
  1097→        this.inner.position.y = -0.09 * u;
  1098→      } else if (k < 0.68) {
  1099→        const u = strikeEase((k - 0.42) / 0.26);
  1100→        t.j_shoulderR = [L(-0.52, 1.08, u), 0, L(-2.95, -0.42, u)];
  1101→        t.j_elbowR = [L(-0.6, -0.1, u), 0, 0];
  1102→        // Fase 21 — the wrist WHIP of the overhead smash
  1103→        t.j_wristR = [Math.sin(u * Math.PI) * -0.4, 0, 0];
  1104→        t.j_torso = [L(0.1, 0.5, u), L(-0.5, 0.5, u), 0];
  1105→        t.j_head = [L(-0.3, 0.34, u), 0, 0];
  1106→        t.j_kneeR = [0.5 * u, 0, 0];
  1107→        t.j_kneeL = [0.55 * u, 0, 0];
  1108→        this.inner.position.y = -0.07 * (1 - u) - 0.05 * u;
  1109→      } else {
  1110→        const u = (k - 0.68) / 0.32;
  1111→        t.j_shoulderR = [L(0.95, 0, u), 0, L(-0.35, -0.06, u)];
  1112→        t.j_elbowR = [L(-0.1, -0.25, u), 0, 0];

  1116→        t.j_kneeL = [0.55 * (1 - u), 0, 0];
  1117→        this.inner.position.y = -0.05 * (1 - u);
  1118→      }
  1119→      t.j_shoulderL = [-0.35, 0, 0.25];
  1120→      t.j_elbowL = [-0.9, 0, 0];
  1121→    } else if (kind === 'thrust') {

  1372→  /**
  1373→   * THE CAPE CLOTH LAYER — spring-damper secondary motion for the cloak.
  1374→   *
  1375→   * Two chain bones (j_cape1 shoulder-line, j_cape2 mid-panel) are skinned to
  1376→   * the real cape sheet of the GLB. Every frame we drive them from the
  1377→   * character's motion state: the cape streams back at a run, billows in
  1378→   * long falls, tucks on the rise, trails flat in the water, lags a beat
  1379→   * behind turns, and answers the coil-whip of an attack with a pulse.
  1380→   * j_cape1 is a stiff spring (settles fast); j_cape2 follows cape1 with a
  1381→   * beat of lag and more overshoot — the wave travels down the fabric.
  1382→   */
  1383→  private updateCape(p: AnimParams, dt: number): void {
  1384→    if (!this.capeBone1 || !this.capeBone2) return;
  1385→    const sm = (x: number, lo: number, hi: number) => THREE.MathUtils.smoothstep(x, lo, hi);
  1386→    const clamp = THREE.MathUtils.clamp;
  1387→    const speed = this.speedSm;
  1388→    const vy = p.vy ?? 0;
  1389→    const yawRate = p.yawRate ?? 0;
  1390→
  1391→    // idle flutter: two sines, quiet at speed (the air is doing the talking)
  1392→    const flutter = (Math.sin(this.t * 3.1) * 0.028 + Math.sin(this.t * 7.7) * 0.014)
  1393→      * (1 - Math.min(1, speed / 5.2));
  1394→
  1395→    // pitch target: hang → stream back at a run → flare in long falls
  1396→    let pitch = 0.05
  1397→      + sm(speed, 0.6, 6.4) * 0.44        // streams back at a run
  1398→      + (vy < -3 ? Math.min(0.38, (-vy - 3) * 0.05) : 0)  // skydive flare
  1399→      - (vy > 3 ? 0.10 : 0)               // tucked on the rise of a jump
  1400→      + (p.swimming ? 0.5 : 0)            // trails flat in the water
  1401→      + flutter;
  1402→    // the coil-whip of a swing shoves the fabric back through the cut
  1403→    if (p.attackT > 0) pitch += Math.sin(Math.min(1, p.attackT) * Math.PI) * 0.16;
  1404→    pitch = clamp(pitch, -0.15, 1.05);
  1405→
  1406→    // yaw target: the cape lags the turn (counter-rotation = secondary action)
  1407→    const yawT = clamp(-yawRate * 0.055, -0.30, 0.30) + flutter * 0.4;
  1408→
  1409→    // integrate: critically-snappy springs, cape1 stiff / cape2 loose
  1410→    const spring = (st: { r: number[]; v: number[] }, tx: number, ty: number, K: number, D: number) => {
  1411→      for (let a = 0; a < 2; a++) {
  1412→        const target = a === 0 ? tx : ty;
  1413→        const acc = (target - st.r[a]) * K - st.v[a] * D;
  1414→        st.v[a] += acc * dt;
  1415→        st.r[a] = clamp(st.r[a] + st.v[a] * dt, -0.5, 1.2);
  1416→      }
  1417→    };
  1418→    spring(this.cape1st, pitch, yawT, 26, 8.0);
  1419→    // cape2 chases cape1's CURRENT pose ×1.28 — the wave travels down the hem
  1420→    spring(this.cape2st, this.cape1st.r[0] * 1.28 + 0.03, this.cape1st.r[1] * 1.35 + flutter * 0.5, 11, 4.2);
  1421→
  1422→    this.capeBone1.rotation.set(this.cape1st.r[0], this.cape1st.r[1], 0);
  1423→    this.capeBone2.rotation.set(this.cape2st.r[0], this.cape2st.r[1], 0);
  1424→  }
  1425→
  1426→  // ------------------------------------------------------------ apply
  1427→
  1428→  /** Write target eulers onto the bones. `direct` skips damping (attacks,
  1429→   * rolls, swims) so the curves stay crisp. */
  1430→  private apply(target: PoseTarget, dt: number, direct: boolean): void {
  1431→    for (const name of BONE_NAMES) {
  1432→      const b = this.bones.get(name);
  1433→      if (!b) continue;
  1434→      const tv = target[name];
  1435→      if (!tv) continue;
  1436→      if (direct) {
  1437→        b.rotation.set(tv[0], tv[1], tv[2]);
  1438→      } else {
  1439→        const c = this.cur[name] ?? [0, 0, 0] as Euler3;
  1440→        c[0] = damp(c[0], tv[0], 14, dt);
  1441→        c[1] = damp(c[1], tv[1], 14, dt);
  1442→        c[2] = damp(c[2], tv[2], 14, dt);
  1443→        this.cur[name] = c;
  1444→        b.rotation.set(c[0], c[1], c[2]);
  1445→      }
  1446→      if (direct) this.cur[name] = [tv[0], tv[1], tv[2]];
  1447→    }
  1448→  }
  1449→
  1450→  dispose(): void {
  1451→    if (this.isVillager) {
  1452→      // Villager clones SHARE geometry with the player's knight (SkeletonUtils
  1453→      // clone) — only the per-instance headwear may be freed.
  1454→      this.root.traverse((o) => {
  1455→        const mesh = o as THREE.Mesh;
  1456→        if (mesh.isMesh && mesh.userData.villagerProp) {
  1457→          mesh.geometry?.dispose();
  1458→          const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
  1459→          for (const m of mats) (m as THREE.Material).dispose();
  1460→        }
  1461→      });
  1462→      return;
  1463→    }
  1464→    this.root.traverse((o) => {
  1465→      const mesh = o as THREE.Mesh;
  1466→      if (mesh.isMesh) {
  1467→        mesh.geometry?.dispose();
  1468→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
  1469→        for (const m of mats) (m as THREE.Material).dispose();
  1470→      }
  1471→    });
  1472→  }
