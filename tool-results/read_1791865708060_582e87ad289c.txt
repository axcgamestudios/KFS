   429→   429→  private executeBroadside(ship: Ship, shots: CannonShot[], fromPlayer: boolean): void {
   430→   430→    for (let i = 0; i < shots.length; i++) {
   431→   431→      const shot = shots[i];
   432→   432→      const spread = fromPlayer ? 0.035 : 0.085;
   433→   433→      const dir = shot.dir.clone();
   434→   434→      dir.x += (Math.random() - 0.5) * spread * 2;
   435→   435→      dir.z += (Math.random() - 0.5) * spread * 2;
   436→   436→      dir.y += 0.05 + Math.random() * 0.05;
   437→   437→      dir.normalize();
   438→   438→      const speed = 32 + Math.random() * 3;
   439→   439→      this.spawnBall(shot.from, dir.multiplyScalar(speed), fromPlayer);
   440→   440→      // Muzzle flash + gun smoke (Fase 7: real flame burst at the muzzle)
   441→   441→      getFlameFX().burst(shot.from, 0.9, 3, 0.75);
   442→   442→      this.flashes.spawn(shot.from, 1.2, 0.14, undefined, { grow: 3 });
   443→   443→      this.smoke.spawn(shot.from.clone().add(new THREE.Vector3(0, 0.4, 0)), 1.8, 1.6,
   444→   444→        new THREE.Vector3(dir.x * 2, 0.6, dir.z * 2), { grow: 1.4 });
   445→   445→      if (i === 0) this.cb.sound('cannon');
   446→   446→    }
   447→   447→  }
