    47→/** Bar with a delayed pale "ghost" trail that shows recent damage. */
    48→function GhostBar({
    49→  value, max, kind, low,
    50→}: { value: number; max: number; kind: 'hp' | 'st'; low?: boolean }) {
    51→  const [ghost, setGhost] = useState(value);
    52→  useEffect(() => {
    53→    // heals follow instantly (next tick); damage trails behind (420 ms)
    54→    const t = setTimeout(() => setGhost(value), value < ghost ? 420 : 0);
    55→    return () => clearTimeout(t);
    56→  }, [value, ghost]);
    57→  const pct = Math.max(0, Math.min(100, (value / max) * 100));
    58→  const ghostPct = Math.max(pct, Math.min(100, (ghost / max) * 100));
    59→  return (
    60→    <div className={`kfs-bar ${low ? 'kfs-bar-low' : ''}`}>
    61→      <div className="kfs-bar-ghost" style={{ width: `${ghostPct}%` }} />
    62→      <div className={`kfs-bar-fill ${kind === 'hp' ? 'kfs-bar-fill-hp' : 'kfs-bar-fill-st'}`} style={{ width: `${pct}%` }} />
    63→      <div className="kfs-bar-ticks" />
    64→    </div>
    65→  );
    66→}

    98→        const a = (i * 36 - 90) * (Math.PI / 180);
    99→        return (
   100→          <line key={i}

   122→/** Compass strip with 15° ticks, cardinal letters and a degree readout. */
   123→function CompassStrip() {
   124→  const [yaw, setYaw] = useState(0);
   125→  useEffect(() => {
   126→    const iv = setInterval(() => {
   127→      const y = (window as unknown as { __kotfsYaw?: number }).__kotfsYaw;
   128→      if (typeof y === 'number') setYaw(y);
   129→    }, 80);
   130→    return () => clearInterval(iv);
   131→  }, []);
   132→  const deg = Math.round(((-yaw * 180) / Math.PI + 360) % 360);
   133→  const CARDINALS: Record<number, string> = { 0: 'N', 90: 'E', 180: 'S', 270: 'O' };
   134→  const HALVES: Record<number, string> = { 45: 'NE', 135: 'SE', 225: 'SO', 315: 'NO' };
   135→
   136→  const marks: React.ReactNode[] = [];
   137→  for (let d = 0; d < 360; d += 15) {
   138→    let delta = d - deg;
   139→    if (delta > 180) delta -= 360;
   140→    if (delta < -180) delta += 360;
   141→    const px = delta * 2.7;
   142→    if (Math.abs(px) > 158) continue;
   143→    const cardinal = CARDINALS[d];
   144→    const half = HALVES[d];
   145→    if (cardinal) {
   146→      marks.push(
   147→        <span key={d} className="kotfs-title absolute text-[0.86rem] font-bold text-[color:#e8c96a]"
   148→          style={{
   149→            left: `calc(50% + ${px}px)`, top: '50%', transform: 'translate(-50%,-54%)',
   150→            textShadow: '0 0 10px rgba(232,201,106,0.75), 0 2px 4px #000',
   151→            opacity: 1 - Math.abs(px) / 210,
   152→          }}>
   153→          {cardinal}
   154→        </span>,
   155→      );
   156→    } else if (half) {
   157→      marks.push(
   158→        <span key={d} className="kotfs-title absolute text-[0.56rem] text-[color:var(--kotfs-parchment-dim)]"
   159→          style={{
   160→            left: `calc(50% + ${px}px)`, top: '56%', transform: 'translateX(-50%)',
   161→            opacity: 0.85 - Math.abs(px) / 220,
   162→          }}>
   163→          {half}
   164→        </span>,
   165→      );
   166→    } else {
   167→      marks.push(
   168→        <span key={d} className="absolute w-px bg-[color:rgba(201,162,39,0.5)]"
   169→          style={{
   170→            left: `calc(50% + ${px}px)`, top: 7, height: d % 45 === 0 ? 8 : 5,
   171→            opacity: 0.8 - Math.abs(px) / 220,
   172→          }} />,
   173→      );
   174→    }
   175→  }
   176→
   177→  return (
   178→    <div className="flex flex-col items-center gap-0.5">
   179→      <div className="kfs-compass kotfs-compass-mask relative w-[330px]">
   180→        {marks}
   181→        <div className="kfs-compass-vignette" />
   182→      </div>
   183→      <p className="kotfs-title text-[0.56rem] tracking-[0.3em] text-[color:var(--kotfs-parchment-dim)] opacity-75">
   184→        {String(deg).padStart(3, '0')}°
   185→      </p>
   186→    </div>
   187→  );
   188→}
