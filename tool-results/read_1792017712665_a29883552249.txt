     8→/**
     9→ * Fase 5 — KnightModelGLB.
    10→ * The player's body: the repository's medieval knight, rigged offline into
    11→ * knight_rigged.glb (real glTF skin + 15 landmark bones + the statue sword
    12→ * and shield as hideable rigid props hanging from the wrists).
    13→ *
    14→ * This class drives the skeleton procedurally — a full animation layer:
    15→ *   locomotion (idle / walk / run with counter-rotation, bob and knee
    16→ *   extension), jump / fall / landing absorb, dodge roll (full somersault),
    17→ *   surface swim, horse riding, block (shield brace), bow aim + release,
    18→ *   five attack curves (diagonal, backhand, overhead smash, thrust, bow
    19→ *   snap) and a hit-reaction recoil. The statue sword rests in the right
    20→ *   hand; when the game equips a weapon it swaps for the kit model, and the
    21→ *   statue shield yields to the kit shield on the left forearm.
    22→ *
    23→ * Convention: character faces +Z. RIGHT side = -X (sword arm, j_*R bones),
    24→ * LEFT side = +X (shield arm, j_*L bones). Limb rotation about +X moves a
    25→ * hanging limb backward (-Z); about -Z the right arm raises outward.
    26→ */
    27→export interface KnightVisualDeps {
    28→  scene: THREE.Object3D;
    29→  weaponFactory?: (kind: Exclude<WeaponKind, 'none'>) => THREE.Object3D | null;
    30→  shieldFactory?: () => THREE.Object3D | null;
    31→  /** Villager variant: re-tints the armour into clothes + headwear variety.
    32→   *  When set, the statue sword & shield NEVER show. */
    33→  villager?: { palette: import('./CharacterModel').CharacterPalette; seed: number; role?: string };
    34→  /** Fase 10 — bandit variant: RED knights (dark crimson cloth + oxidised
    35→   *  red-steel armour). When set the statue props never show. Fase 17:
    36→   *  `shield: true` arms heavies with a kit shield on the left forearm.
    37→   *  Fase 22: `tint`/`eyes` recolour legends and temple guardians. */
    38→  bandit?: {
    39→    seed: number; captain?: boolean; shield?: boolean;
    40→    tint?: { h: number; s: number; l: number };
    41→    eyes?: number;
    42→  };
    43→}

   104→  constructor(deps: KnightVisualDeps) {
   105→    this.weaponFactory = deps.weaponFactory;
   106→    this.shieldFactory = deps.shieldFactory;
   107→    const scene = deps.scene;
   108→    this.isVillager = !!deps.villager;
   109→    this.isBandit = !!deps.bandit;
   110→    this.banditShield = !!deps.bandit?.shield;
   111→
   112→    // collect bones + meshes
   113→    for (const name of BONE_NAMES) {
   114→      const b = scene.getObjectByName(name);
   115→      if (b) this.bones.set(name, b);
   116→    }
   117→    this.swordProp = scene.getObjectByName('j_sword_held') ?? null;
   118→    this.shieldProp = scene.getObjectByName('j_shield_held') ?? null;
   119→    // Fase 22 — the cape chain (absent in pre-F22 rigs: dynamics no-ops)
   120→    this.capeBone1 = scene.getObjectByName('j_cape1') ?? null;
   121→    this.capeBone2 = scene.getObjectByName('j_cape2') ?? null;
   122→
   123→    if (deps.villager) this.dressAsVillager(scene, deps.villager);
   124→    if (deps.bandit) this.dressAsBandit(scene, deps.bandit);
   125→
   126→    scene.traverse((o) => {
   127→      const mesh = o as THREE.Mesh;
   128→      if (mesh.isMesh) {
   129→        mesh.castShadow = true;
   130→        mesh.receiveShadow = true;
   131→        mesh.frustumCulled = false; // skeleton moves beyond the bind bbox
   132→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   133→        for (const m of mats) {
   134→          const std = m as THREE.MeshStandardMaterial;
   135→          if (std.isMeshStandardMaterial && !this.flashMats.includes(std)) this.flashMats.push(std);
   136→        }
   137→      }
   138→    });
   139→
   140→    // attach slots
   141→    const wristR = this.bones.get('j_wristR');
   142→    if (wristR) { wristR.add(this.handSlot); }
   143→    const elbowL = this.bones.get('j_elbowL');
   144→    if (elbowL) { elbowL.add(this.armSlot); this.armSlot.position.set(0, -0.09, 0.045); }
   145→
   146→    // The GLB clone root can arrive with visible=false (something in the boot
   147→    // pipeline toggles loaded model roots); the knight must always render.
   148→    // GLTF scene roots can arrive with visible=false after loader round-trips;
   149→    // the player's body must always render.
   150→    scene.visible = true;
   151→
   152→    this.inner.add(scene);
   153→    this.root.add(this.inner);
   154→    this.refreshProps();
   155→  }

   193→  /**
   194→   * Fase 8 — at dusk the shield gives way to a lit torch: a wooden brand
   195→   * with a real FlameFX flame and a warm flickering light, riding the left
   196→   * forearm slot. At dawn the torch goes out and the shield returns.

   198→  setNightTorch(on: boolean): void {
   199→    if (on === this.torchOn || this.isVillager) return;
   200→    this.torchOn = on;
   201→    if (on) {
   202→      const g = new THREE.Group();
   203→      g.name = 'night_torch';
   204→      const wood = new THREE.MeshStandardMaterial({ color: 0x5a4028, roughness: 0.9 });
   205→      const stick = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.042, 0.52, 6), wood);
   206→      stick.position.y = 0.16;
   207→      stick.castShadow = true;
   208→      g.add(stick);
   209→      const wrap = new THREE.Mesh(
   210→        new THREE.CylinderGeometry(0.062, 0.05, 0.15, 8),
   211→        new THREE.MeshStandardMaterial({ color: 0x3a2c1c, roughness: 1 }),
   212→      );
   213→      wrap.position.y = 0.46;
   214→      g.add(wrap);
   215→      this.torchLight = new THREE.PointLight(0xff9a3c, 2.4, 20, 1.7);
   216→      this.torchLight.position.y = 0.55;
   217→      g.add(this.torchLight);
   218→      this.armSlot.add(g);
   219→      g.rotation.set(-0.35, 0, 0);
   220→      g.position.set(-0.02, -0.05, 0.1);
   221→      this.torchProp = g;
   222→      // the flame mesh lives at the scene root (additive) — synced every frame
   223→      this.torchFlame = getFlameFX().addFlame(new THREE.Vector3(0, -100, 0), {
   224→        scale: 0.17, height: 0.42, heat: 0.5,
   225→      });

   227→      if (this.activeShield) this.activeShield.visible = false;
   228→      if (this.shieldProp) this.shieldProp.visible = false;
   229→    } else {
   230→      if (this.torchProp) { this.armSlot.remove(this.torchProp); this.torchProp = null; }
   231→      if (this.torchFlame) { this.torchFlame.dispose(); this.torchFlame = null; }
   232→      this.torchLight = null;
   233→      this.torchOn = false;
   234→      // the shield returns with the sun
   235→      this.refreshProps();
   236→    }

   293→  private weaponTier = 0;
   294→  private weaponElement: SwordElement | null = null;

   296→  private applyWeaponGlow(): void {
   297→    const w = this.activeWeapon ?? this.swordProp;
   298→    if (!w) return;
   299→    // Fase 11 — the master-forged sword carries its own rune-glow materials
   300→    if (w.userData?.heroSword) {
   301→      applyHeroSwordTier(w, this.weaponTier, false);
   302→      if (this.weaponElement) applyHeroSwordElement(w, this.weaponElement);
   303→      const s = 1 + this.weaponTier * 0.05 + (this.weaponElement ? 0.06 : 0);
   304→      w.scale.set(s, s, s);
   305→      return;
   306→    }
   307→    const glows = [null, 0xff9a3c, 0xe8c96a, 0x7ec8ff] as const;
   308→    const glow = glows[this.weaponTier];
   309→    w.traverse((o) => {
   310→      const mesh = o as THREE.Mesh;
   311→      if (!mesh.isMesh) return;
   312→      const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   313→      for (const m of mats) {
   314→        const std = m as THREE.MeshStandardMaterial;
   315→        if (!std.isMeshStandardMaterial) continue;
   316→        if (glow === null) {
   317→          std.emissive?.setHex(0x000000);
   318→        } else {
   319→          std.emissive?.setHex(glow);
   320→          std.emissiveIntensity = 0.55 + this.weaponTier * 0.3;
   321→        }
   322→        std.needsUpdate = true;
   323→      }
   324→    });
   325→    // each tier forges a slightly longer, prouder blade
   326→    const s = 1 + this.weaponTier * 0.06;
   327→    w.scale.set(s, s, s);
   328→  }

   330→  private refreshProps(): void {
   331→    // Kit weapon in the right hand (detaches the previous one first)
   332→    if (this.activeWeapon) {
   333→      this.handSlot.remove(this.activeWeapon);
   334→      if (this.activeWeapon.parent) this.activeWeapon.parent.remove(this.activeWeapon);
   335→      this.activeWeapon = null;
   336→    }
   337→    const showKitWeapon = !this.isVillager && this.weaponKind !== 'none';
   338→    if (showKitWeapon && this.weaponFactory) {
   339→      const w = this.weaponFactory(this.weaponKind as Exclude<WeaponKind, 'none'>);
   340→      if (w) {
   341→        if (this.weaponKind === 'bow') {
   342→          // bows live permanently in the LEFT hand (aim code extends the arm)
   343→          const wristL = this.bones.get('j_wristL');
   344→          if (wristL) {
   345→            wristL.add(w);
   346→            w.position.set(0.02, -0.08, 0.05);
   347→            w.rotation.set(0, 0, 0.08);
   348→          }
   349→        } else {
   350→          this.handSlot.add(w);
   351→        }
   352→        this.activeWeapon = w;
   353→        this.tuneWeaponPose(this.weaponKind);
   354→        this.applyWeaponGlow();
   355→      }
   356→    }
   357→    // shields: the kit shield on the left forearm
   358→    if (this.activeShield) {
   359→      this.armSlot.remove(this.activeShield);
   360→      this.activeShield = null;
   361→    }
   362→    if (this.wantsShield && !this.isVillager && (!this.isBandit || this.banditShield) && this.shieldFactory) {
   363→      const s = this.shieldFactory();
   364→      if (s) {
   365→        this.armSlot.add(s);
   366→        this.activeShield = s;
   367→        this.tuneShieldPose();
   368→      }
   369→    }
   370→    // Fase 7 — NOTHING is shown until equipped:
   371→    //   · bare hands → no statue sword, no statue shield
   372→    //   · the statue props only appear as FALLBACKS when the matching kit
   373→    //     object could not be produced (kit GLB missing)
   374→    if (this.swordProp) {
   375→      this.swordProp.visible = !this.isVillager && !this.isBandit
   376→        && this.weaponKind !== 'none'
   377→        && this.weaponKind !== 'bow'
   378→        && !this.activeWeapon;
   379→    }
   380→    if (this.shieldProp) {
   381→      this.shieldProp.visible = !this.isVillager && this.wantsShield && !this.activeShield
   382→        && (!this.isBandit || this.banditShield);
   383→    }
   384→  }
   385→
   386→  /**
   387→   * Fase 7 — villager variant: the repository knight's body re-dressed as a
   388→   * commoner. Materials are CLONED first (the AssetSystem shares them with
   389→   * the player), the armour is re-tinted to cloth/leather tones and each NPC
   390→   * gets seeded headwear (hood / straw hat / cap) + a work tool by role.
   391→   */
   392→  private dressAsVillager(scene: THREE.Object3D, v: { palette: import('./CharacterModel').CharacterPalette; seed: number; role?: string }): void {
   393→    const r = (k: number) => {
   394→      const x = Math.sin(v.seed * 37.71 + k * 91.13) * 43758.5453;
   395→      return x - Math.floor(x);
   396→    };
   397→    const tint = new THREE.Color();
   398→    scene.traverse((o) => {
   399→      const mesh = o as THREE.Mesh;
   400→      if (!mesh.isMesh) return;
   401→      const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   402→      const clones = mats.map((m) => (m as THREE.Material).clone());
   403→      for (const m of clones) {
   404→        const std = m as THREE.MeshStandardMaterial;
   405→        if (!std.isMeshStandardMaterial) continue;
   406→        const isUpper = /upper/i.test(std.name ?? '');
   407→        const base = isUpper ? v.palette.tunic : v.palette.pants;
   408→        tint.setHex(base);
   409→        // seeded shade variety per NPC (avoid clone-army look)
   410→        const shade = 0.82 + r(9) * 0.36;
   411→        tint.multiplyScalar(shade);
   412→        std.color.copy(tint);
   413→        std.roughness = 0.92;
   414→        std.metalness = 0.04;
   415→        std.emissive?.setHex(0x000000);
   416→        std.needsUpdate = true;
   417→      }
   418→      mesh.material = clones.length === 1 ? clones[0] : clones;
   419→    });
   420→
   421→    // --- headwear on the head bone (rigid props ride the skeleton) ---
   422→    const head = this.bones.get('j_head') ?? scene.getObjectByName('j_head');
   423→    if (head) {
   424→      const cloth = (c: number, rough = 0.95) =>
   425→        new THREE.MeshStandardMaterial({ color: c, roughness: rough, metalness: 0.02 });
   426→      const wear = r(2);
   427→      if (wear < 0.34) {
   428→        // hood
   429→        const hood = new THREE.Mesh(new THREE.ConeGeometry(0.155, 0.3, 8, 1, true), cloth(v.palette.tunic));
   430→        hood.position.y = 0.22;
   431→        hood.rotation.x = -0.14;
   432→        hood.castShadow = true;
   433→        hood.userData.villagerProp = true;
   434→        head.add(hood);
   435→      } else if (wear < 0.64) {
   436→        // straw hat (brim + crown)
   437→        const g = new THREE.Group();
   438→        const brim = new THREE.Mesh(new THREE.CylinderGeometry(0.26, 0.28, 0.025, 12), cloth(0xb59a58));
   439→        const crown = new THREE.Mesh(new THREE.CylinderGeometry(0.13, 0.15, 0.13, 10), cloth(0xa8894a));
   440→        crown.position.y = 0.07;
   441→        brim.castShadow = crown.castShadow = true;
   442→        g.add(brim, crown);
   443→        g.userData.villagerProp = true;
   444→        g.position.y = 0.26;
   445→        g.rotation.z = (r(3) - 0.5) * 0.24;
   446→        head.add(g);
   447→      } else if (wear < 0.85) {
   448→        // soft cap
   449→        const cap = new THREE.Mesh(
   450→          new THREE.SphereGeometry(0.15, 10, 8, 0, Math.PI * 2, 0, Math.PI * 0.52),
   451→          cloth(v.palette.sleeves),
   452→        );
   453→        cap.position.y = 0.17;
   454→        cap.castShadow = true;
   455→        cap.userData.villagerProp = true;
   456→        head.add(cap);
   457→      }
   458→    }
   459→
   460→    // --- role tool in the right hand ---
   461→    const wristR = scene.getObjectByName('j_wristR');
   462→    if (wristR && v.role) {
   463→      const wood = new THREE.MeshStandardMaterial({ color: 0x6a4e2e, roughness: 0.9, metalness: 0.02 });
   464→      const steel = new THREE.MeshStandardMaterial({ color: 0x767c82, roughness: 0.42, metalness: 0.78 });
   465→      if (v.role === 'smith') {
   466→        const hammer = new THREE.Group();
   467→        const haft = new THREE.Mesh(new THREE.CylinderGeometry(0.016, 0.02, 0.34, 6), wood);
   468→        haft.position.y = 0.14;
   469→        const hm = new THREE.Mesh(new THREE.BoxGeometry(0.075, 0.055, 0.05), steel);
   470→        hm.position.y = 0.3;
   471→        hammer.add(haft, hm);
   472→        hammer.rotation.x = 1.35;
   473→        hammer.userData.villagerProp = true;
   474→        wristR.add(hammer);
   475→      } else if (v.role === 'merchant') {
   476→        const sack = new THREE.Mesh(
   477→          new THREE.SphereGeometry(0.09, 8, 6),
   478→          new THREE.MeshStandardMaterial({ color: 0x9a8a66, roughness: 0.98, metalness: 0.0 }),
   479→        );

   499→  /**

   529→            std.metalness = 0.55;
   530→            std.roughness = 0.5;
   531→          }
   532→          std.color.copy(tint);
   533→        } else {
   534→          // dark crimson trousers / leather (or the legend's own shade)
   535→          if (v.tint) {
   536→            tint.setHSL(v.tint.h, Math.min(1, v.tint.s + 0.1), v.tint.l * 0.72);
   537→          } else {
   538→            tint.setHSL(0.99, 0.42, 0.16 + r(6) * 0.07);

   581→  /** Per-weapon grip orientation in the hand.
   582→   *
   583→   * Fase 23 — THE GRIP, rebuilt from the fist geometry: the wrist bone sits
   584→   * at the anatomical joint and the gauntlet fist hangs a few cm BELOW it,
   585→   * palm facing the thigh. A held sword therefore crosses that fist
   586→   * diagonally — grip centred inside the fist, pommel at the pinky edge,
   587→   * blade sweeping up-forward-outward at roughly 45 degrees. That single

   630→  /** Fase 14 — scratch reutilizable: zero() creaba un objeto + 15 arrays por
   631→   *  frame POR MODELO (17 caballeros = 270 arrays/frame de presión de GC). */
   632→  private zeroScratch: PoseTarget = {};
   633→  private zero(): PoseTarget {
   634→    const p = this.zeroScratch;
   635→    for (const n of BONE_NAMES) {
   636→      const e = p[n];
   637→      if (e) { e[0] = 0; e[1] = 0; e[2] = 0; }
   638→      else p[n] = [0, 0, 0];
   639→    }
   640→    return p;
   641→  }

   643→  update(p: AnimParams): void {
   644→    // Fase 14 — culling de animación: lejos del jugador el sistema dueño
   645→    // pasa dt = 0 y la pose queda congelada (invisible a esa distancia,
   646→    // todo el álgebra de huesos ahorrada). El propio modelo sigue en su sitio.
   647→    if (p.dt <= 0) return;
   648→    this.t += p.dt;
   649→    const dt = Math.max(1e-4, p.dt);
   650→    const target = this.zero();
   651→
   652→    // speed smoothing + phase
   653→    this.speedSm = damp(this.speedSm, p.speed, 8, dt);
   654→    const moving = p.moving && this.speedSm > 0.25;
   655→    // Fase 23 — CONTINUOUS STRIDE CADENCE: the old two-step switch (6.6 walk
   656→    // / 9.4 run at 4.4 m/s) made the gait hitch the instant the threshold
   657→    // was crossed. The stride rate now glides with speed — no seam.
   658→    const cadence = 6.6 + THREE.MathUtils.clamp((this.speedSm - 3.0) / 3.6, 0, 1) * 3.4;
   659→    if (moving) this.phase += dt * cadence;
   660→    const ph = this.phase;
   661→
   662→    // landing detection
   663→    if (p.grounded && !this.wasGrounded) this.landT = p.attackT > 0 || (p.rollT ?? 0) > 0 ? 0 : 0.34;
   664→    if (!p.grounded) this.airT += dt; else this.airT = 0;
   665→    this.wasGrounded = p.grounded;
   666→    if (this.landT > 0) this.landT = Math.max(0, this.landT - dt);
   667→
   668→    // ---- roll/swim/ride overrides ----
   669→    const rollT = p.rollT ?? 0;
   670→    if (rollT > 0) { this.poseRoll(target, rollT, moving, ph, p.dodgeStep === true); this.apply(target, dt, true); this.post(p); return; }
   671→    if (p.swimming) { this.poseSwim(target, ph, p); this.apply(target, dt, true); this.post(p); return; }
   672→    if (p.riding) { this.poseRide(target, p); this.apply(target, dt, true); this.post(p); return; }
   673→
   674→    // ---- Fase 17 — the HOOK DASH flying kick: the rope carries the body
   675→    //      and both heels lead the way. Held while the kick lingers. ----
   676→    if (p.dashKick && p.attackT === 0) {
   677→      this.poseHookKick(target);
   678→      this.apply(target, dt, true);
   679→      this.post(p);
   680→      return;
   681→    }
   682→
   683→    // ---- Fase 11 — LAYERED base: air OR locomotion (no early return any
   684→    // more: the attack/block overlays now layer over the airborne pose too,
   685→    // which is what makes mid-air slashes and plunges readable) ----
   686→    if (!p.grounded) {
   687→      this.poseAir(target, this.airT);
   688→    } else {
   689→      this.poseLocomotion(target, p, moving, ph);
   690→    }
   691→
   692→    // ---- landing absorb (grounded only) ----
   693→    if (p.grounded) {
   694→      if (this.landT > 0) {
   695→        const k = this.landT / 0.34;
   696→        const e = k * k;
   697→        target.j_kneeR = [target.j_kneeR[0] + 0.85 * e, 0, 0];
   698→        target.j_kneeL = [target.j_kneeL[0] + 0.85 * e, 0, 0];
   699→        target.j_hipR = [target.j_hipR[0] - 0.5 * e, 0, 0];
   700→        target.j_hipL = [target.j_hipL[0] - 0.5 * e, 0, 0];
   701→        target.j_torso = [target.j_torso[0] + 0.28 * e, target.j_torso[1], 0];
   702→        this.inner.position.y = -0.09 * e;
   703→      } else if (this.inner.position.y !== 0) {
   704→        this.inner.position.y = damp(this.inner.position.y, 0, 10, dt);
   705→      }
   706→    }
   707→
   708→    // ---- bow aim (overrides arms) ----
   709→    const bowDraw = p.bowDraw ?? 0;
   710→    if (bowDraw > 0) this.poseBowAim(target, p, bowDraw, moving, ph);
   711→
   712→    // ---- block overlay (grounded stance only) ----
   713→    const blockT = p.blockT ?? 0;
   714→    if (blockT > 0 && p.grounded) this.poseBlock(target, blockT, this.activeShield !== null || (!!this.shieldProp && this.shieldProp.visible));
   715→
   716→    // ---- attack overlays (timed curves — ground OR air) ----
   717→    if (p.attackT > 0) {
   718→      this.poseAttack(target, p.attackT, p.attackKind ?? 'light1', bowDraw);
   719→      this.apply(target, dt, true);
   720→      this.post(p);
   721→      return;
   722→    }
   723→
   724→    // Fase 17 — the sweep-kick spin unwinds when the kick isn't driving it
   725→    this.inner.rotation.y = damp(this.inner.rotation.y, 0, 14, dt);
   726→
   727→    // ---- hit reaction ----
   728→    if (this.hitT > 0) {
   729→      const k = this.hitT;
   730→      // Fase 22 — the flinch reads through the WHOLE upper body now: chest
   731→      // caves, head snaps down and rolls, both shoulders recoil off the line
   732→      // of the blow — a reaction with mass, not a red flash.
   733→      target.j_torso = [target.j_torso[0] - 0.22 * k, target.j_torso[1], 0.05 * k];
   734→      target.j_head = [target.j_head[0] - 0.18 * k, target.j_head[1] * (1 - k), -0.08 * k];
   735→      target.j_shoulderR = [target.j_shoulderR[0], target.j_shoulderR[1], target.j_shoulderR[2] - 0.16 * k];
   736→      target.j_shoulderL = [target.j_shoulderL[0], target.j_shoulderL[1], target.j_shoulderL[2] + 0.12 * k];
   737→    }
   738→
   739→    this.apply(target, dt, false);
   740→    this.post(p);
   741→  }

   743→  /** Per-frame post steps (flash decay, prop visibility by context). */
   744→  private post(p: AnimParams): void {
   745→    // Fase 22 — THE CAPE CLOTH LAYER runs on every path (locomotion, attack
   746→    // early-returns included — the cape answers to the swing's coil-whip)
   747→    if (p.dt > 0) this.updateCape(p, Math.max(1e-4, p.dt));
   748→    if (this.flashT > 0) {
   749→      this.flashT = Math.max(0, this.flashT - p.dt * 3.2);
   750→      const e = this.flashT * 0.6;
   751→      for (const m of this.flashMats) m.emissive.setRGB(e, e * 0.12, e * 0.08);
   752→    }
   753→    if (this.hitT > 0) this.hitT = Math.max(0, this.hitT - p.dt * 2.6);
   754→    // the statue shield (fallback only) tucks away while aiming a bow
   755→    if (this.shieldProp && !this.isVillager) {
   756→      this.shieldProp.visible = this.wantsShield && !this.activeShield && (p.bowDraw ?? 0) < 0;
   757→    }
   758→    // shield impact recoil decay
   759→    if (this.blockHitT > 0) this.blockHitT = Math.max(0, this.blockHitT - p.dt * 4.2);
   760→    // Fase 8 — the night torch rides the left forearm: sync the flame to the
   761→    // brand head and flicker the light with the same rhythm
   762→    if (this.torchOn && this.torchProp && this.torchFlame) {
   763→      this.torchT += p.dt;
   764→      this.torchProp.updateWorldMatrix(true, false);
   765→      this.torchFlame.position.setFromMatrixPosition(this.torchProp.matrixWorld);
   766→      this.torchFlame.position.y += 0.55;
   767→      if (this.torchLight) {
   768→        const fl = 0.8 + Math.sin(this.torchT * 11.3) * 0.14 + Math.sin(this.torchT * 23.7) * 0.07;
   769→        this.torchLight.intensity = gfxRuntime.pointLights ? 2.4 * fl : 0;
   770→      }
   771→    }
   772→    // the kit weapon stays in the hand; while swimming it is shouldered up
   773→    if (this.activeWeapon) {
   774→      const swim = p.swimming;
   775→      this.activeWeapon.rotation.x = damp(
   776→        this.activeWeapon.rotation.x, swim ? -1.2 : this.activeWeapon.userData.baseRotX ?? this.activeWeapon.rotation.x, 6, Math.max(1e-4, p.dt),
   777→      );
   778→    }
   779→  }

   783→  private poseLocomotion(t: PoseTarget, p: AnimParams, moving: boolean, ph: number): void {
   784→    const runK = Math.min(1, this.speedSm / 6.2);
   785→    if (moving) {
   786→      // ---- Fase 17 — the charge blend: past ~65% of top speed the knight
   787→      // drops into the SHIELD CHARGE — shield forward, body low, blade
   788→      // trailing. It eases in with speed so walking never triggers it. ----
   789→      const hasShield = this.wantsShield
   790→        && (this.activeShield !== null || (this.shieldProp?.visible ?? false));
   791→      const charge = hasShield
   792→        ? THREE.MathUtils.smoothstep(runK, 0.55, 0.88)
   793→        : THREE.MathUtils.smoothstep(runK, 0.68, 0.92);
   794→      const armed = this.weaponKind !== 'none' && this.weaponKind !== 'bow';
   795→
   796→      const amp = 0.52 + runK * 0.5 + charge * 0.12;
   797→      const legPhL = Math.sin(ph);
   798→      const legPhR = Math.sin(ph + Math.PI);
   799→      // Fase 12 — pelvis rocks over the stance leg (secondary motion: the
   800→      // walk stops reading as two stiff pistons and starts swinging from
   801→      // the hips like a real gait).
   802→      const hipRoll = Math.sin(ph) * (0.035 + runK * 0.05);
   803→      // right leg (x<0) — j_hipR: -X = forward
   804→      t.j_hipR = [-legPhR * amp, 0, -hipRoll];
   805→      t.j_kneeR = [Math.max(0, -Math.sin(ph + Math.PI - 0.7)) * (0.55 + runK * 0.75 + charge * 0.22), 0, 0];
   806→      t.j_ankleR = [-0.1 + legPhR * 0.18, 0, Math.max(0, legPhR) * 0.08];
   807→      // left leg
   808→      t.j_hipL = [-legPhL * amp, 0, hipRoll];
   809→      t.j_kneeL = [Math.max(0, -Math.sin(ph - 0.7)) * (0.55 + runK * 0.75 + charge * 0.22), 0, 0];
   810→      t.j_ankleL = [-0.1 + legPhL * 0.18, 0, Math.max(0, legPhL) * 0.08];
   811→
   812→      // ---- ARMS ----
   813→      // Fase 23 — the armed arm LIVES: the old ready-carry was a frozen
   814→      // statue arm (the literal "predetermined pose" read). The sword arm
   815→      // now rides the gait — a modest counter-swing with the elbow breathing
   816→      // and the wrist rocking the blade — while charging it still settles
   817→      // back into the lance-at-rest trail.
   818→      const armAmp = 0.34 + runK * 0.5;
   819→      if (armed) {
   820→        const gait = Math.sin(ph + 0.35);                  // arm lags the leg a touch
   821→        const ready = -0.3 + gait * (0.13 + runK * 0.1);   // low ready, gently swinging
   822→        const trailing = 0.55 + gait * 0.08;
   823→        t.j_shoulderR = [THREE.MathUtils.lerp(ready, trailing, charge), -gait * 0.04, THREE.MathUtils.lerp(-0.12, -0.18, charge)];
   824→        t.j_elbowR = [THREE.MathUtils.lerp(-0.72 - Math.max(0, -gait) * 0.16, -0.3, charge), 0, 0];
   825→        t.j_wristR = [-0.1 + gait * 0.07, 0, 0.1 + Math.sin(ph * 2) * 0.03];
   826→      } else {
   827→        t.j_shoulderR = [legPhR * armAmp * (1 - charge * 0.5), 0, -0.06];
   828→        t.j_elbowR = [-0.18 - runK * 0.35 - Math.max(0, -legPhR) * (0.3 + runK * 0.3), 0, 0];
   829→      }
   830→      // Fase 10/17 — SHIELD CARRY → SHIELD CHARGE: a knight does NOT swing
   831→      // the shield arm like a pendulum. Walking: elbow bent, shield held low
   832→      // across the thigh with a touch of bob. Charging: the shield snaps UP
   833→      // AND FORWARD — the battering-ram brace.
   834→      const shieldCarry = hasShield && (p.blockT ?? 0) < 0.1;
   835→      if (shieldCarry) {
   836→        const bob = Math.sin(ph * 2) * 0.045;
   837→        const walkPose: [number, number, number] = [-0.36 + bob - runK * 0.12, 0.18, 0.32];
   838→        const chargePose: [number, number, number] = [-1.12, 0.28, 0.3];
   839→        t.j_shoulderL = [
   840→          THREE.MathUtils.lerp(walkPose[0], chargePose[0], charge),
   841→          THREE.MathUtils.lerp(walkPose[1], chargePose[1], charge),
   842→          THREE.MathUtils.lerp(walkPose[2], chargePose[2], charge),
   843→        ];
   844→        t.j_elbowL = [THREE.MathUtils.lerp(-0.94, -1.02, charge), 0, 0.16];
   845→      } else {
   846→        t.j_shoulderL = [legPhL * armAmp * 0.8 * (1 - charge * 0.5), 0, 0.06];
   847→        t.j_elbowL = [-0.25 - runK * 0.3 - Math.max(0, -legPhL) * (0.28 + runK * 0.28), 0, 0];
   848→      }
   849→
   850→      // ---- TORSO & HEAD ----
   851→      // The charge crouches into the run: deep lean, eyes UP over the rim.
   852→      const lean = 0.05 + runK * 0.2 + charge * 0.16;
   853→      t.j_torso = [lean, Math.sin(ph) * (0.05 + runK * 0.08), 0];
   854→      t.j_head = [
   855→        -0.04 - runK * 0.08 - charge * 0.2,
   856→        -Math.sin(ph) * 0.06,
   857→        -Math.sin(ph) * 0.02,
   858→      ];
   859→      // Fase 21 — the whole body ROLLS over the stance foot (weight shift in
   860→      // the ribcage, not just a positional sway: kills the stiff-plank read)
   861→      this.inner.rotation.z = damp(this.inner.rotation.z, Math.sin(ph) * (0.02 + runK * 0.035), 8, 1 / 60);
   862→      this.inner.position.y = Math.abs(Math.sin(ph)) * 0.028 * (0.5 + runK + charge * 0.5) - charge * 0.05;
   863→      // Fase 12 — subtle lateral sway of the whole body over the stance foot
   864→      this.inner.position.x = Math.sin(ph) * 0.012 * (0.6 + runK);
   865→      // Fase 21 — WRIST follow-through: the hands lag the arm swing a beat
   866→      // (relaxed wrists are half of "not rigid"). Fase 23: the armed wrist
   867→      // is now authored inside the armed-arm block above — it rocks the
   868→      // BLADE with the gait instead of a fixed grip.
   869→      if (!armed) {
   870→        t.j_wristR = [legPhR * 0.14 * (0.4 + runK), 0, -0.05 + Math.sin(ph * 2) * 0.03];
   871→        t.j_wristL = [legPhL * 0.12 * (0.4 + runK), 0, 0.05 - Math.sin(ph * 2) * 0.03];
   872→      } else {

  1118→  private poseAttack(t: PoseTarget, k: number, kind: string, bowDraw: number): void {
  1119→    const L = THREE.MathUtils.lerp;
  1120→    const ease = (u: number) => u * u * (3 - 2 * u);
  1121→    // Fase 22 — the AAA easing set:
  1122→    //  · wuEase (quadratic-OUT): launches into the gather, then settles into
  1123→    //    the hold. The windup reads as WEIGHT TRANSFER — anticipation — not a
  1124→    //    linear wind-up clock.
  1125→    //  · strikeEase (cubic-OUT): the blade EXPLODES out of the windup and
  1126→    //    decelerates through the target. A whip, not a ramp.
  1127→    const wuEase = (u: number) => 1 - (1 - u) * (1 - u);
  1128→    const strikeEase = (u: number) => 1 - Math.pow(1 - u, 3);
  1129→    if (kind === 'heavy') {
  1130→      // overhead smash: windup up-back (0..0.42), slam (0.42..0.68), recover
  1131→      if (k < 0.42) {
  1132→        const u = wuEase(k / 0.42);
  1133→        t.j_shoulderR = [L(0, -0.52, u), 0, L(0, -2.95, u)];
  1134→        t.j_elbowR = [-0.6, 0, 0];
  1135→        t.j_torso = [0.1, L(0, -0.5, u), 0];
  1136→        t.j_head = [L(0, -0.3, u), 0, 0];
  1137→        this.inner.position.y = -0.09 * u;
  1138→      } else if (k < 0.68) {
  1139→        const u = strikeEase((k - 0.42) / 0.26);
  1140→        t.j_shoulderR = [L(-0.52, 1.08, u), 0, L(-2.95, -0.42, u)];
  1141→        t.j_elbowR = [L(-0.6, -0.1, u), 0, 0];
  1142→        // Fase 21 — the wrist WHIP of the overhead smash
  1143→        t.j_wristR = [Math.sin(u * Math.PI) * -0.4, 0, 0];
  1144→        t.j_torso = [L(0.1, 0.5, u), L(-0.5, 0.5, u), 0];
  1145→        t.j_head = [L(-0.3, 0.34, u), 0, 0];
  1146→        t.j_kneeR = [0.5 * u, 0, 0];
  1147→        t.j_kneeL = [0.55 * u, 0, 0];
  1148→        this.inner.position.y = -0.07 * (1 - u) - 0.05 * u;
  1149→      } else {

  1151→        t.j_shoulderR = [L(0.95, 0, u), 0, L(-0.35, -0.06, u)];
  1152→        t.j_elbowR = [L(-0.1, -0.25, u), 0, 0];

  1154→        t.j_head = [L(0.3, 0, u), 0, 0];
  1155→        t.j_kneeR = [0.5 * (1 - u), 0, 0];
  1156→        t.j_kneeL = [0.55 * (1 - u), 0, 0];
  1157→        this.inner.position.y = -0.05 * (1 - u);
  1158→      }
  1159→      t.j_shoulderL = [-0.35, 0, 0.25];
  1160→      t.j_elbowL = [-0.9, 0, 0];
  1161→    } else if (kind === 'thrust') {
  1162→      if (k < 0.35) {
  1163→        const u = wuEase(k / 0.35);
  1164→        t.j_shoulderR = [L(0, -0.58, u), 0, L(0, -0.18, u)];
  1165→        t.j_elbowR = [L(0, -1.95, u), 0, 0];

  1167→        t.j_hipR = [L(0, 0.15, u), 0, 0];

  1175→        t.j_torso = [0.16, L(0.4, -0.32, u), 0];

  1177→        t.j_kneeR = [0.25, 0, 0];

  1412→  /**
  1413→   * THE CAPE CLOTH LAYER — spring-damper secondary motion for the cloak.
  1414→   *
  1415→   * Two chain bones (j_cape1 shoulder-line, j_cape2 mid-panel) are skinned to
  1416→   * the real cape sheet of the GLB. Every frame we drive them from the
  1417→   * character's motion state: the cape streams back at a run, billows in
  1418→   * long falls, tucks on the rise, trails flat in the water, lags a beat
  1419→   * behind turns, and answers the coil-whip of an attack with a pulse.
  1420→   * j_cape1 is a stiff spring (settles fast); j_cape2 follows cape1 with a
  1421→   * beat of lag and more overshoot — the wave travels down the fabric.
  1422→   */
  1423→  private updateCape(p: AnimParams, dt: number): void {
  1424→    if (!this.capeBone1 || !this.capeBone2) return;
  1425→    const sm = (x: number, lo: number, hi: number) => THREE.MathUtils.smoothstep(x, lo, hi);
  1426→    const clamp = THREE.MathUtils.clamp;
  1427→    const speed = this.speedSm;
  1428→    const vy = p.vy ?? 0;
  1429→    const yawRate = p.yawRate ?? 0;
  1430→
  1431→    // idle flutter: two sines, quiet at speed (the air is doing the talking)
  1432→    const flutter = (Math.sin(this.t * 3.1) * 0.028 + Math.sin(this.t * 7.7) * 0.014)
  1433→      * (1 - Math.min(1, speed / 5.2));
  1434→
  1435→    // pitch target: hang → stream back at a run → flare in long falls
  1436→    let pitch = 0.05
  1437→      + sm(speed, 0.6, 6.4) * 0.44        // streams back at a run
  1438→      + (vy < -3 ? Math.min(0.38, (-vy - 3) * 0.05) : 0)  // skydive flare
  1439→      - (vy > 3 ? 0.10 : 0)               // tucked on the rise of a jump
  1440→      + (p.swimming ? 0.5 : 0)            // trails flat in the water
  1441→      + flutter;
  1442→    // the coil-whip of a swing shoves the fabric back through the cut
  1443→    if (p.attackT > 0) pitch += Math.sin(Math.min(1, p.attackT) * Math.PI) * 0.16;
  1444→    pitch = clamp(pitch, -0.15, 1.05);
  1445→
  1446→    // yaw target: the cape lags the turn (counter-rotation = secondary action)
  1447→    const yawT = clamp(-yawRate * 0.055, -0.30, 0.30) + flutter * 0.4;
  1448→
  1449→    // integrate: critically-snappy springs, cape1 stiff / cape2 loose
  1450→    const spring = (st: { r: number[]; v: number[] }, tx: number, ty: number, K: number, D: number) => {
  1451→      for (let a = 0; a < 2; a++) {
  1452→        const target = a === 0 ? tx : ty;
  1453→        const acc = (target - st.r[a]) * K - st.v[a] * D;
  1454→        st.v[a] += acc * dt;
  1455→        st.r[a] = clamp(st.r[a] + st.v[a] * dt, -0.5, 1.2);
  1456→      }
  1457→    };
  1458→    spring(this.cape1st, pitch, yawT, 26, 8.0);
  1459→    // cape2 chases cape1's CURRENT pose ×1.28 — the wave travels down the hem
  1460→    spring(this.cape2st, this.cape1st.r[0] * 1.28 + 0.03, this.cape1st.r[1] * 1.35 + flutter * 0.5, 11, 4.2);
  1461→
  1462→    this.capeBone1.rotation.set(this.cape1st.r[0], this.cape1st.r[1], 0);
  1463→    this.capeBone2.rotation.set(this.cape2st.r[0], this.cape2st.r[1], 0);
  1464→  }
  1465→
  1466→  // ------------------------------------------------------------ apply
  1467→
  1468→  /** Write target eulers onto the bones. `direct` skips damping (attacks,
  1469→   * rolls, swims) so the curves stay crisp.
  1470→   *
  1471→   * Fase 23 — SETTLE CASCADE: instead of one uniform rate, the legs track
  1472→   * the ground fast (they must — the foot plants), the arms settle a beat
  1473→   * later and the torso/head last. The staggered rates are what make a
  1474→   * stop from a sprint READ as weight transferring up the body instead of
  1475→   * a whole puppet snapping to rest. */
  1476→  private static readonly DAMP_RATE: Record<string, number> = {
  1477→    j_hipR: 20, j_kneeR: 20, j_ankleR: 22,
  1478→    j_hipL: 20, j_kneeL: 20, j_ankleL: 22,
  1479→    j_shoulderR: 11, j_elbowR: 13, j_wristR: 15,
  1480→    j_shoulderL: 11, j_elbowL: 13, j_wristL: 15,
  1481→    j_root: 12, j_torso: 9, j_head: 8,
  1482→  };
  1483→  private apply(target: PoseTarget, dt: number, direct: boolean): void {
  1484→    for (const name of BONE_NAMES) {
  1485→      const b = this.bones.get(name);
  1486→      if (!b) continue;
  1487→      const tv = target[name];
  1488→      if (!tv) continue;
  1489→      if (direct) {
  1490→        b.rotation.set(tv[0], tv[1], tv[2]);
  1491→      } else {
  1492→        const rate = KnightModelGLB.DAMP_RATE[name] ?? 14;
  1493→        const c = this.cur[name] ?? [0, 0, 0] as Euler3;
  1494→        c[0] = damp(c[0], tv[0], rate, dt);
  1495→        c[1] = damp(c[1], tv[1], rate, dt);
  1496→        c[2] = damp(c[2], tv[2], rate, dt);
  1497→        this.cur[name] = c;
  1498→        b.rotation.set(c[0], c[1], c[2]);
  1499→      }
  1500→      if (direct) this.cur[name] = [tv[0], tv[1], tv[2]];
  1501→    }
  1502→  }

  1504→  dispose(): void {

  1517→    }
  1518→    this.root.traverse((o) => {
  1519→      const mesh = o as THREE.Mesh;
  1520→      if (mesh.isMesh) {
  1521→        mesh.geometry?.dispose();
  1522→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
  1523→        for (const m of mats) (m as THREE.Material).dispose();
  1524→      }
  1525→    });
  1526→  }
